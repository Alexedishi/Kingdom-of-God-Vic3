kog_country_is_christian_trigger = {
	OR = {
		kog_country_is_western_christian_trigger = yes
		kog_country_is_eastern_christian_trigger = yes
	}
}

kog_country_is_western_christian_trigger = {
	OR = {
		kog_country_is_roman_catholic_trigger = yes
		kog_country_is_protestant_trigger = yes
	}
}

kog_country_is_eastern_christian_trigger = {
	OR = {
		kog_country_is_eastern_orthodox_trigger = yes
		kog_country_is_oriental_orthodox_trigger = yes
		kog_country_is_rus_orthodox_trigger = yes # Basileia Romaion
		kog_country_is_elysian_orthodox_trigger = yes # Basileia Romaion
	}
}

####---------- Religious Capacity Trigger ----------####
# Be careful invoking this trigger, it calls the script value directly
# This is gated behind the is_player trigger for performance- kog_capacity_value is very complicated
kog_capacity_trigger = {
	hidden_trigger = {
		save_temporary_scope_value_as = {
			name = kog_trigger_value
			value = $value$
		}
	}
	custom_tooltip = {
		text = kog_capacity_trigger
		kog_capacity_value >= $value$
	}
}
