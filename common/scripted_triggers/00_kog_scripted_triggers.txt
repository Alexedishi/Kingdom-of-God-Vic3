# Religion Group Triggers
kog_is_christian_trigger = {
	OR = {
		kog_is_western_christian_trigger = yes
		kog_is_eastern_christian_trigger = yes
	}
}

kog_is_western_christian_trigger = {
	OR = {
		country_has_state_religion = rel:catholic
		country_has_state_religion = rel:protestant
	}
}

kog_is_eastern_christian_trigger = {
	OR = {
		country_has_state_religion = rel:orthodox
		country_has_state_religion = rel:oriental_orthodox
	}
}

kog_is_roman_catholic_trigger = {
	country_has_state_religion = rel:catholic
}

kog_is_eastern_orthodox_trigger = {
	country_has_state_religion = rel:orthodox
}

kog_is_rus_orthodox_trigger = {
	country_has_state_religion = rel:orthodox
	var:kog_local.var:kog_liturgical_rite_var = flag:kog_russian_rite
}

kog_is_oriental_orthodox_trigger = {
	country_has_state_religion = rel:oriental_orthodox
}

kog_is_oriental_armenian_trigger = {
	country_has_state_religion = rel:oriental_orthodox
	var:kog_local.var:kog_liturgical_rite_var = flag:kog_armenian_rite
}

kog_is_elysian_orthodox_trigger = { # Basileia Romaion
	always = no
}

kog_is_mixtec_orthodox_trigger = { # Basileia Romaion
	kog_is_elysian_orthodox_trigger = yes
	var:kog_local.var:kog_liturgical_rite_var = flag:kog_mixtec_rite
}

kog_is_cahokian_orthodox_trigger = { # Basileia Romaion
	kog_is_elysian_orthodox_trigger = yes
	var:kog_local.var:kog_liturgical_rite_var = flag:kog_cahokian_rite
}
