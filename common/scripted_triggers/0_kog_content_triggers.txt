####---------- Religious Group Triggers ----------####
kog_content_scope_trigger = {
	OR = {
		kog_country_is_roman_catholic_trigger = yes
		kog_country_is_eastern_orthodox_trigger = yes
		# kog_country_is_oriental_orthodox_trigger = yes
		# kog_country_is_protestant_trigger = yes
		# kog_country_is_rus_orthodox_trigger = yes <-- currently unused in BR
		kog_country_is_elysian_orthodox_trigger = yes
	}
}

kog_country_is_christian_trigger = {
	OR = {
		kog_country_is_western_christian_trigger = yes
		kog_country_is_eastern_christian_trigger = yes
	}
}

kog_country_is_western_christian_trigger = {
	OR = {
		kog_country_is_roman_catholic_trigger = yes
		kog_country_is_protestant_trigger = yes
	}
}

kog_country_is_eastern_christian_trigger = {
	OR = {
		kog_country_is_eastern_orthodox_trigger = yes
		kog_country_is_oriental_orthodox_trigger = yes
		kog_country_is_rus_orthodox_trigger = yes
		kog_country_is_elysian_orthodox_trigger = yes
	}
}


####---------- Religion Denomination Triggers ----------####
kog_country_is_roman_catholic_trigger = {
	country_has_state_religion = rel:catholic
}

kog_country_is_eastern_orthodox_trigger = {
	country_has_state_religion = rel:orthodox
}

kog_country_is_oriental_orthodox_trigger = {
	country_has_state_religion = rel:oriental_orthodox
}

kog_country_is_protestant_trigger = {
	OR = {
		country_has_state_religion = rel:protestant
		country_has_state_religion = rel:lutheran
		country_has_state_religion = rel:calvinist
		country_has_state_religion = rel:anglican
		country_has_state_religion = rel:methodist
		country_has_state_religion = rel:baptist
		country_has_state_religion = rel:mormon
	}
}

kog_country_is_oriental_armenian_trigger = {
	country_has_state_religion = rel:oriental_orthodox
	var:kog_local.var:kog_liturgical_rite_var = flag:kog_armenian_rite
}

kog_religion_is_rus_orthodox_trigger = {
	trigger_if = {
		limit = {
			basileia_is_active = yes
		}
		religion = rel:br_religion_rus_orthodox
	}
	trigger_else = {
		always = no
	}
}
