kog_country_schism_threshold_trigger = {
	hidden_trigger = {
		save_temporary_scope_value_as = {
			name = kog_schism_threshold_value
			value = modifier:country_schism_threshold_add
		}
	}
	custom_tooltip = {
		text = kog_country_schism_threshold_trigger
		kog_capacity_value >= modifier:country_schism_threshold_add
	}
}

	custom_description = {
		text = kog_is_schismatic_trigger
		je:je_kog_local ?= {
			has_modifier = modifier_kog_schismatic
		}
	}
}
