kog_journal_has_council_map_trigger = {
	scope:journal_entry = {
		has_variable_map = kog_ecumenical_council_map
	}
}

kog_journal_has_canon_law_group_debate_trigger = {
	scope:journal_entry = {
		has_variable_map = kog_ecumenical_council_map
		is_key_in_variable_map = {
			name = kog_ecumenical_council_map
			target = flag:$canon_law_group$
		}
	}
}

kog_country_matches_journal_religion_trigger = {
	hidden_trigger = {
		scope:journal_entry = {
			var:kog_religion_var ?= root.religion
		}
	}
}

kog_council_button_validity_trigger = {
	kog_country_matches_journal_religion_trigger = yes
	kog_journal_has_council_map_trigger = yes
}

kog_country_has_ongoing_council_trigger = {
	custom_description = {
		text = kog_country_has_ongoing_council_trigger
		subject = this
		object = var:kog_global_faith_journal_var
		var:kog_global_faith_journal_var = {
			has_variable_map = kog_ecumenical_council_map
		}
	}
}

kog_country_own_council_is_debating_canon_lawgroup_trigger = {
	# TODO custom_description is currently broken; do not use parameterized custom descriptions
	custom_description = {
		text = $canon_law_group$_debate_trigger
		subject = this
		object = var:kog_global_faith_journal_var

			has_variable_map = kog_ecumenical_council_map
			is_key_in_variable_map = {

			}
		}
	}
}

kog_country_is_ecumenical_council_participant_trigger = {
	custom_description = {
		subject = this
		object = var:kog_mother_church_var
		text = kog_country_is_ecumenical_council_participant_trigger
		var:kog_local.var:kog_mother_church_var = {
			has_variable_map = kog_ecumenical_council_map
		}
		OR = {
			has_modifier = modifier_kog_convoked_council
			has_modifier = modifier_kog_participated_in_council
		}
	}
}
