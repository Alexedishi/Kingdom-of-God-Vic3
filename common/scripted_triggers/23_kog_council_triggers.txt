kog_journal_has_council_map_trigger = {
	scope:journal_entry = {
		has_variable_map = kog_ecumenical_council_map
	}
}

kog_country_matches_journal_religion_trigger = {
	scope:journal_entry = {
		var:kog_religion_var ?= root.religion
	}
}

kog_council_button_validity_trigger = {
	kog_country_matches_journal_religion_trigger = yes
	kog_journal_has_council_map_trigger = yes
}

kog_has_ongoing_council_trigger = {
	custom_description = {
		text = kog_has_ongoing_council_trigger
		subject = this
		object = var:kog_global_faith_journal_var
		var:kog_global_faith_journal_var = {
			has_variable = kog_ecumenical_council_var
			exists = var:kog_ecumenical_council_var
		}
	}
}

kog_council_is_debating_canon_lawgroup_trigger = {
	custom_description = {
		text = $canon_law_group$_debate_trigger
		scope:journal_entry = {
			has_variable = $canon_law_group$
		}
	}
}

kog_country_is_ecumenical_council_participant_trigger = {
	custom_description = {
		subject = this
		object = var:kog_mother_church_var.var:kog_ecumenical_council_var
		text = kog_country_is_ecumenical_council_participant_trigger
		var:kog_mother_church_var.var:kog_ecumenical_council_var = {
			any_in_list = {
				variable = kog_ecumenical_council_list
				root = this
			}
		}
	}
}
