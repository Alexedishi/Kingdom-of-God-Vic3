	custom_description = {
		text = kog_has_ongoing_council_trigger
		subject = this
		object = var:kog_local.var:kog_global_faith_journal_var
		var:kog_local.var:kog_global_faith_journal_var = {
			has_variable_map = kog_council_matters_varmap
		}
	}
}

	}
}
