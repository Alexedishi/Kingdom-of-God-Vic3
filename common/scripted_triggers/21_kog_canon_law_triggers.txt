kog_faith_has_canon_law_trigger = {
	custom_tooltip = {
		text = kog_faith_has_canon_law
		flag:$canon_law$ = {
			save_temporary_scope_as = canon_law
		}
		rel:$religion$ = {
			save_temporary_scope_as = religion
		}
		any_in_global_list = {
			variable = kog_global_journals_list
			var:kog_religion_var ?= $religion$
			OR = {
				var:kog_canon_lawgroup_apostolic_succession ?= flag:$canon_law$
				var:kog_canon_lawgroup_almsgiving ?= flag:$canon_law$
				var:kog_canon_lawgroup_iconography ?= flag:$canon_law$
				var:kog_canon_lawgroup_statism ?= flag:$canon_law$
				var:kog_canon_lawgroup_nationalism ?= flag:$canon_law$
				var:kog_canon_lawgroup_clerical_marriage ?= flag:$canon_law$
				var:kog_canon_lawgroup_social_role ?= flag:$canon_law$
				var:kog_canon_lawgroup_stance_on_sciences ?= flag:$canon_law$
				var:kog_canon_lawgroup_apostolic_succession ?= flag:$canon_law$
				var:kog_canon_lawgroup_crusades ?= flag:$canon_law$
			}
		}
	}
}
