kog_canon_lawgroup_debate_trigger = {
	custom_tooltip = {
		text = kog_canon_lawgroup_debate_trigger
		flag:$flag$ = {
			save_temporary_scope_as = kog_lawgroup_debate_trigger
		}
		scope:journal_entry = {
			has_variable_map = kog_ecumenical_council_map
			is_key_in_variable_map = {
				name = kog_ecumenical_council_map
				target = flag:$flag$
			}
		}
	}
}

kog_canon_lawgroup_debate_trigger_NOT = {
	custom_tooltip = {
		text = kog_canon_lawgroup_debate_trigger
		flag:$flag$ = {
			save_temporary_scope_as = kog_lawgroup_debate_trigger
		}
		scope:journal_entry = {
			has_variable_map = kog_ecumenical_council_map
			NOR = {
				is_key_in_variable_map = {
					name = kog_ecumenical_council_map
					target = flag:$flag$
				}
			}
		}
	}
}
