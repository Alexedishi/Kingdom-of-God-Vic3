####----------- Episcopal Seat Control Triggers ----------####
# THESE TRIGGERS EVALUATE FROM THE COUNTRY SCOPE!
kog_controls_seat_trigger = {
	custom_tooltip = {
		text = kog_controls_seat_trigger
		container:$seat$ = {
			save_temporary_scope_value_as = {
				name = kog_seat_name
				value = name
			}
			var:kog_seat_controller ?= root
		}
	}
}

kog_controls_mother_seat_trigger = {
	custom_tooltip = {
		text = kog_controls_mother_seat_trigger
		trigger_if = {
			limit = {
				kog_country_is_roman_catholic_trigger = yes
			}
			kog_controls_seat_trigger = {
				seat = kog_rome
			}
		}
		trigger_else_if = {
			limit = {
				kog_country_is_eastern_orthodox_trigger = yes
			}
			kog_controls_seat_trigger = {
				seat = kog_constantinople
			}
		}
		trigger_else_if = {
			limit = {
				kog_country_is_oriental_orthodox_trigger = yes
			}
			kog_controls_seat_trigger = {
				seat = kog_coptic_alexandria
			}
		}
		trigger_else_if = {
			limit = {
				kog_country_is_elysian_orthodox_trigger = yes
			}
			kog_controls_seat_trigger = {
				seat = elysium
			}
		}
		trigger_else = {
			always = no
		}
	}
}
