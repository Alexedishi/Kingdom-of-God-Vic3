####---------- Religious Capacity Trigger ----------####
kog_capacity_trigger = {
	custom_description = {
		text = kog_capacity_trigger
		subject = this
		value = $value$
		save_temporary_scope_value_as = {
			name = kog_capacity_temp_scope
			value = $value$
		}
		kog_capacity_value >= $value$
	}
}
