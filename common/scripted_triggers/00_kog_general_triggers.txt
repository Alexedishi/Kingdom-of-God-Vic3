####---------- Religious Capacity Trigger ----------####
# Be careful invoking this trigger, it calls kog_capacity_value directly
kog_capacity_trigger = {
	custom_description = {
		text = kog_capacity_trigger
		subject = this
		value = $value$
	}
	kog_capacity_value >= $value$
}
