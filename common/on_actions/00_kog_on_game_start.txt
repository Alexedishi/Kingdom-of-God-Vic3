####---------- Game Initialization Effect ----------####
on_game_started = {
	on_actions = {
		kog_game_start
	}
}

kog_game_start = {
	effect = {
		kog_initialize_content_effect = yes
	}
}

on_game_started_after_lobby = {
	on_actions = {
		kog_after_lobby
	}
}

kog_after_lobby = {
	effect = {
		every_country = {
			limit = {
				kog_content_scope_trigger = yes
			}
			save_scope_value_as = {
				name = kog_national_piety_value
				value = kog_national_piety_target_value
			}
			save_scope_value_as = {
				name = kog_national_piety_target_value
				value = kog_national_piety_target_value
			}
			save_scope_value_as = {
				name = kog_national_piety_progress_value
				value = kog_national_piety_progress_value
			}
			save_scope_value_as = {
				name = kog_canon_law_compliance_value
				value = kog_canon_law_compliance_value
			}
			var:kog_local = {
				set_variable = {
					name = kog_national_piety_var
					value = scope:kog_national_piety_value
				}
				set_variable = {
					name = kog_national_piety_target_var
					value = scope:kog_national_piety_target_value
				}
				set_variable = {
					name = kog_national_piety_progress_var
					value = scope:kog_root_scope.kog_national_piety_progress_value
				}
				set_variable = {
					name = kog_canon_law_compliance_var
					value = scope:kog_canon_law_compliance_value
				}
			}
		}
		kog_update_faith_journal_lists_effect = yes
	}
}

kog_initialize_values_in_variable_maps = {
	effect = {
		every_in_global_list = {
			variable = kog_global_seat_list
			com_update_value_in_variable_map = {
				name = kog_relations_map
				key = root
				value = 0
			}
			com_update_value_in_variable_map = {
				name = kog_opinions_map
				key = root
				value = 15
			}
			com_update_value_in_variable_map = {
				name = kog_diplomacy_map
				key = root
				value = 0
			}
		}
	}
}
