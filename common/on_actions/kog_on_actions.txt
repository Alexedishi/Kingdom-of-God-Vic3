kog_monthly_pulse = {
	trigger = {
		kog_content_scope_trigger = yes
	}
	on_actions = {
		kog_catholic_pulse
		kog_orthodox_pulse
		kog_oriental_pulse
		kog_elysian_pulse
#		kog_protestant_pulse
		kog_schism_pulse
		kog_integration_pulse
		kog_mending_pulse
	}
	events = {
		kog_general.121 # The Capture of the Great Mosque
	}
	random_events = { 
		chance_of_no_event = 95
		5 = kog_general.700 # Ecumenical Financing
		5 = kog_general.701 # Delegation Arrives from Seat
		5 = kog_general.702 # Patriarch Condemns Immorality
	}
	effect = {
		if = {
			limit = {
				NOT = {
					has_global_variable = kog_monthly_timeout_var
				}
			}
			set_global_variable = {
				name = kog_monthly_timeout_var
				days = 20
			}
			kog_update_holy_site_count_effect = yes
			kog_update_patriarch_variables_effect = yes
			kog_patriarch_diplomacy_effect = yes
			kog_update_opinion_variables = yes
			kog_update_rite_modifiers_effect = yes
			kog_update_seat_controllers_effect = yes
			kog_update_votes_effect = yes
		}
	}
}

	}

kog_yearly_pulse = {
	effect = {
		if = {
			limit = {
				NOT = {
					has_global_variable = kog_yearly_timeout_var
				}
			}
			set_global_variable = {
				name = kog_yearly_timeout_var
				days = 20 # Should be enough to get past all monthly pulses
			}
			kog_update_ranking_lists = yes
		}
	}
}

	on_actions = {

	}
}

kog_topbar_setup = {
	effect = {
		add_com_topbar_element = {
			element_name = kog_topbar_element_capacity
		}
		every_country = {
			limit = {
				kog_is_eastern_christian_trigger = yes
			}
			add_to_variable_list = {
				name = com_topbar_second_line
				target = scope:kog_topbar_element_capacity
			}
		}
	}
}
