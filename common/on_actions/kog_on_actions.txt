kog_christendom_pulse = {
	trigger = {
		kog_content_scope_trigger = yes
	}
	effect = {
		if = {
			limit = {
				NOT = {
					has_global_variable = kog_monthly_timeout_var
				}
			}
			set_global_variable = {
				name = kog_monthly_timeout_var
				days = 20
			}
			kog_update_patriarch_variables_effect = yes
			kog_update_seat_controllers_effect = yes
		}
	}
}

	on_actions = {
		kog_yearly_pulse
	}
}

kog_yearly_pulse = {
	effect = {
		if = {
			limit = {
				NOT = {
					has_global_variable = kog_yearly_timeout_var
				}
			}
			set_global_variable = {
				name = kog_yearly_timeout_var
				days = 20 # Should be enough to get past all monthly pulses
			}
			kog_update_ranking_lists = yes
		}
	}
}

on_game_started_after_lobby = {
	on_actions = {
		kog_topbar_setup
	}
}

kog_topbar_setup = {
	effect = {
		add_com_topbar_element = {
			element_name = kog_topbar_element_capacity
		}
		every_country = {
			limit = {
				kog_is_eastern_christian_trigger = yes
			}
			add_to_variable_list = {
				name = com_topbar_second_line
				target = scope:kog_topbar_element_capacity
			}
		}
	}
}
