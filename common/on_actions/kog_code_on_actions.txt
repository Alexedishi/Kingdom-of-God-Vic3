####---------- on_law_activated ----------####
on_law_activated = {
	on_actions = {
		kog_on_theocracy_enacted
	}
}

kog_on_theocracy_enacted = {
	trigger = {
		law_type = law_type:law_theocracy
		owner = {
			kog_content_scope_trigger = yes
		}
	}
	effect = {
		owner = {
			activate_law = law_type:law_theocratic_investiture
		}
	}
}

####---------- on_law_enactment_pass ----------####
on_law_enactment_pass = {
	on_actions = {
		kog_education_law_enactment_pass
	}
}

kog_education_law_enactment_pass = {
	trigger = {
		active_law:lawgroup_education_system ?= {
			has_amendment = amendment_type:amendment_kog_sunday_schools
		}
	}
	effect = {
		if = {
			limit = {
				OR = {
					ROOT.currently_enacting_law.type ?= law_type:law_religious_schools
					ROOT.currently_enacting_law.type ?= law_type:law_public_schools
					ROOT.currently_enacting_law.type ?= law_type:law_private_schools
				}
			}
			currently_enacting_law = {
				add_amendment = {
					type = amendment_kog_sunday_schools
					sponsor = prev.ig:ig_devout
				}
			}
		}
	}
}
