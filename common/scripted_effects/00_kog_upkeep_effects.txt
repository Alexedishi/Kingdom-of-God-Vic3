####---------- Patriarch Update Effects ---------- ####
# These update effects use the following flow control:
# Check if the character saved to kog_patriarch_var exists; this variable is set during initialization
# If the character exists, do nothing
# If the character does not exist, determine if the seat has a controller and then generate a character from the appropriate scope
# Update kog_patriarch_var in the appropriate struct

kog_update_rome_effect = {
	global_var:kog_global = {
		var:kog_rome = {
			if = {
				limit = {
					NOT = {
						exists = var:kog_patriarch_var
					}
				}
				# First, check for the papacy
				if = {
					limit = {
						c:PAP ?= {
							has_law = law_type:law_theocracy
						}
					}
					c:PAP ?= {
						ruler = {
							save_scope_as = kog_rome_patriarch_scope
						}
					}
				}
				# Failing that, attempt to find character in void
				else_if = {
					limit = {
						any_character_in_void = {
							has_template = template_kog_rome_patriarch
						}
					}
					random_character_in_void = {
						limit = {
							has_template = template_kog_rome_patriarch
						}
						save_scope_as = kog_rome_patriarch_scope
					}
				}
				# Failing that, generate a new character
				else = {
					# Attempt to generate using the Rome controller
					if = {
						limit = {
							has_variable = kog_controller_var
							exists = var:kog_controller_var
						}
						var:kog_controller_var = {
							create_character = {
								template = template_kog_rome_patriarch
								on_created = {
									place_character_in_void = 180 # 15 year shelf life
								}
							}
						}
					}
					# Fallback to an Italian Catholic
					else = {
						create_character = {
							template = template_kog_rome_patriarch
							culture = cu:south_italian
							religion = rel:catholic
							on_created = {
								place_character_in_void = 180 # 15 year shelf life
							}
						}
					}
				}
				# Save the new character
				random_character_in_void = {
					limit = {
						has_template = template_kog_rome_patriarch
					}
					save_scope_as = kog_rome_patriarch_scope
				}
				set_variable = {
					name = kog_patriarch_var
					value = scope:kog_rome_patriarch_scope
				}
				
				# Update Relations Variables
				every_country = {
					limit = {
						kog_content_scope_trigger = yes
					}
					var:kog_local = {
						# Sanity check in case initialization had a wobble
						if = {
							limit = {
								NOT = {
									has_variable = kog_rome_relations_var
								}
							}
							set_variable = {
								name = kog_rome_relations_var
								value = 0
							}
						}
						# Do not reduce relations if already at or below 0
						else_if = {
							limit = {
								var:kog_rome_relations_var > 0
							}
							change_variable = {
								name = kog_rome_relations_var
								add = -25
							}
							clamp_variable = {
								name = kog_rome_relations_var
								min = 0
								max = 50
							}
						}
					}
					# Notify all Christians of a new Pope
					post_notification = notification_kog_rome_replaced
				}
			}
		}
	}
}
