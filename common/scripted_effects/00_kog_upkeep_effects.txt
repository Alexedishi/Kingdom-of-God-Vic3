## Overview Effects
kog_validate_global_journal_involvement_effect = {
	every_country = {
		limit = {
			OR = {
				AND = {
					is_involved_in_journal_entry = $type$
					$trigger$ = no
				}
				AND = {
					$trigger$ = yes
					NOT = {
						is_involved_in_journal_entry = $type$
					}
				}
			}
		}
		if = {
			limit = {
				is_involved_in_journal_entry = $type$
				$trigger$ = no
			}
			global_var:$type$ = {
				remove_involved_country =  prev
			}
		}
		else_if = {
			limit = {
				$trigger$ = yes
				NOT = {
					is_involved_in_journal_entry = $type$
				}
			}
			global_var:$type$ = {
				add_involved_country =  prev
			}
		}
	}
}

kog_update_devout_member_traits_effect = {
	ig:ig_devout = {
		leader = {
			if = {
				limit = {
					NOR = {
						has_trait = kog_bishop
						has_trait = kog_cardinal
					}
				}
				add_trait = kog_bishop
			}
		}
	}
	every_scope_character = {
		limit = {
			is_interest_group_type = ig_devout
			has_trait = kog_bishop
			has_trait = kog_cardinal
		}
		remove_trait = kog_bishop
	}
}

kog_update_rome_patriatch_effect = {
	# Step 1, Ensure that there is a valid scope saved to the key flag:kog_seat_patriarch
	if = {
		limit = { # If this evaluates to true, proceed to patriarch regeneration
			OR = {
				AND = {
					is_key_in_global_variable_map = {
						name = kog_rome
						target = flag:kog_seat_patriarch
					}
					NOT = {
						exists = "global_variable_map(kog_rome|flag:kog_seat_patriarch)"
					}
				}
				NOT = {
					is_key_in_global_variable_map = {
						name = kog_rome
						target = flag:kog_seat_patriarch
					}
				}
			}
		}
		# First, check for the papacy
		if = {
			limit = {
				p:xF0A020 = {
					owner = {
						c:PAP ?= this
						has_law = law_type:law_theocracy
					}
				}
			}
			c:PAP ?= {
				ruler = {
					if = {
						limit = {
							NOT = {
								has_trait = kog_patriarch_of_rome
							}
						}
						add_trait = kog_patriarch_of_rome
					}
					save_scope_as = kog_rome_patriarch_scope
				}
			}
		}
		# Failing that, attempt to find character in void
		else_if = {
			limit = {
				any_character_in_void = {
					has_trait = kog_patriarch_of_rome
				}
			}
			random_character_in_void = {
				limit = {
					has_trait = kog_patriarch_of_rome
				}
				save_scope_as = kog_rome_patriarch_scope
			}
		}
		# Failing that, generate a new character
		else = {
			# Attempt to generate using the Rome Seat controller
			if = {
				limit = {
					AND = {
						is_key_in_global_variable_map = {
							name = kog_rome
							target = flag:kog_seat_controller
						}
						exists = "global_variable_map(kog_rome|flag:kog_seat_controller)"
					}
				}
				"global_variable_map(kog_rome|flag:kog_seat_controller)" = {
					create_character = {
						template = kog_patriarch_template
						on_created = {
							add_trait = kog_patriarch_of_rome
							place_character_in_void = 180 # 15 year shelf life
						}
					}
				}
			}
			# Attempt to generate using the Rome Province owner
			else_if = {
				limit = {
					p:xB5726E = {
						owner = {
							kog_content_scope_trigger = yes
						}
					}
				}
				p:xB5726E = {
					owner = {
						create_character = {
							template = kog_patriarch_template
							religion = rel:catholic
							on_created = {
								add_trait = kog_patriarch_of_rome
								place_character_in_void = 180 # 15 year shelf life
							}
						}
					}
				}
			}
			# Fall back to an Italian Catholic; create from scope of province owner
			else = {
				p:xB5726E = {
					owner = {
						create_character = {
							template = kog_patriarch_template
							culture = cu:south_italian
							religion = rel:catholic
							on_created = {
								add_trait = kog_patriarch_of_rome
								place_character_in_void = 180 # 15 year shelf life
							}
						}
					}
				}
			}
			# Capture the new character
			random_character_in_void = {
				limit = {
					has_trait = kog_patriarch_of_rome
				}
				save_scope_as = kog_rome_patriarch_scope
			}
		}
		# Update Relations Variables
		every_country = {
			limit = {
				kog_content_scope_trigger = yes
			}
			je:je_kog_local = {
				# Sanity check in case initialization had a wobble
				if = {
					limit = {
						NOT = {
							has_variable = kog_rome_relations_var
						}
					}
					set_variable = {
						name = kog_rome_relations_var
						value = 0
					}
				}
				# Do not reduce relations if already at or below 0
				else_if = {
					limit = {
						var:kog_rome_relations_var > 0
					}
					change_variable = {
						name = kog_rome_relations_var
						add = -25
					}
					clamp_variable = {
						name = kog_rome_relations_var
						min = 0
						max = 50
					}
				}
			}
			# Notify all Christians of a new Pope
			post_notification = notification_kog_rome_replaced
		}
	}
	# Step 2, Update the relevant variable map
	remove_from_global_variable_map = {
		name = kog_rome
		key = flag:kog_seat_patriarch
	}
	add_to_global_variable_map = {
		name = kog_rome
		key = flag:kog_seat_patriarch
		value = scope:kog_rome_patriarch_scope
	}
}
