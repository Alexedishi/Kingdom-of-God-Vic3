####---------- Patriarch Upkeep Effects ---------- ####
# These update effects use the following flow control:
# Attempt to locate the relevant variable map; regenerate if missing
# Check if the character saved to flag:kog_seat_patriarch exists; If the character exists, do nothing
# If the character does not exist, determine if the seat has a controller and then generate a character from the appropriate scope
# Clear the dead character scope from the map and insert the new scope

kog_update_patriarch_variables_effect = {
	kog_update_rome_patriatch_effect = yes
	kog_update_santiago_patriatch_effect = yes
	kog_update_paris_patriatch_effect = yes
	kog_update_koln_patriatch_effect = yes
	kog_update_vienna_patriatch_effect = yes
	kog_update_constantinople_patriatch_effect = yes
	kog_update_antioch_patriatch_effect = yes
	kog_update_jerusalem_patriatch_effect = yes
	kog_update_alexandria_patriatch_effect = yes
	kog_update_moscow_patriatch_effect = yes
	kog_update_coptic_alexandria_patriatch_effect = yes
	kog_update_syriac_antioch_patriatch_effect = yes
	kog_update_etchmiadzin_patriatch_effect = yes
	kog_update_addis_ababa_patriatch_effect = yes
	kog_update_konak_patriatch_effect = yes
	if = {
		limit = {
			basileia_is_active = yes
		}
		kog_update_elysium_patriatch_effect = yes
		kog_update_mixtec_patriatch_effect = yes
		kog_update_cahokia_patriatch_effect = yes
	}
	every_country = {
		limit = {
			kog_content_scope_trigger = yes
		}
		kog_update_metropolitan_bishop_patriatch_effect = yes
	}
}

kog_update_rome_patriatch_effect = {
	# Step 1, Ensure the relevant variable map exists
	if = {
		limit = {
			NOT = {
				has_global_variable_map = kog_rome
			}
		}
		kog_initialize_rome_variable_map_patriatch_effect = yes # Regenerate if missing
	}
	# Step 2, Ensure that there is a valid scope saved to the key flag:kog_seat_patriarch
	if = {
		limit = { # If this evaluates to true, proceed to patriarch regeneration
			OR = {
				AND = {
					is_key_in_variable_map = {
						name = kog_rome
						target = flag:kog_seat_patriarch
					}
					NOT = {
						exists = "variable_map(kog_rome|flag:kog_seat_patriarch)"
					}
				}
				NOT = {
					is_key_in_variable_map = flag:kog_seat_patriarch
				}
			}
		}
		# First, check for the papacy
		if = {
			limit = {
				c:PAP ?= {
					has_law = law_type:law_theocracy
				}
			}
			c:PAP ?= {
				ruler = {
					save_scope_as = kog_rome_patriarch_scope
				}
			}
		}
		# Failing that, attempt to find character in void
		else_if = {
			limit = {
				any_character_in_void = {
					has_modifier = modifier_kog_rome_patriarch
				}
			}
			random_character_in_void = {
				limit = {
					has_modifier = modifier_kog_rome_patriarch
				}
				save_scope_as = kog_rome_patriarch_scope
			}
		}
		# Failing that, generate a new character
		else = {
			# Attempt to generate using the Rome controller
			if = {
				limit = {
					AND = {
						is_key_in_variable_map = {
							name = kog_rome
							target = flag:kog_seat_controller
						}
						exists = "variable_map(kog_rome|flag:kog_seat_controller)"
					}
				}
				"variable_map(kog_rome|flag:kog_seat_controller)" = {
					create_character = {
						template = kog_patriarch_template
						on_created = {
							add_modifier = modifier_kog_rome_patriarch
							place_character_in_void = 180 # 15 year shelf life
						}
					}
				}
			}
			# Fallback to an Italian Catholic
			else = {
				create_character = {
					template = kog_patriarch_template
					culture = cu:south_italian
					religion = rel:catholic
					on_created = {
						place_character_in_void = 180 # 15 year shelf life
					}
				}
			}
		}
		# Update Relations Variables
		every_country = {
			limit = {
				kog_content_scope_trigger = yes
			}
			je:je_kog_local = {
				# Sanity check in case initialization had a wobble
				if = {
					limit = {
						NOT = {
							has_variable = kog_rome_relations_var
						}
					}
					set_variable = {
						name = kog_rome_relations_var
						value = 0
					}
				}
				# Do not reduce relations if already at or below 0
				else_if = {
					limit = {
						var:kog_rome_relations_var > 0
					}
					change_variable = {
						name = kog_rome_relations_var
						add = -25
					}
					clamp_variable = {
						name = kog_rome_relations_var
						min = 0
						max = 50
					}
				}
			}
			# Notify all Christians of a new Pope
			post_notification = notification_kog_rome_replaced
		}
	}
	# Step 3, Capture the Relevant Character and update the variable map
	random_character_in_void = {
		limit = {
			add_modifier = modifier_kog_rome_patriarch
		}
		save_scope_as = kog_rome_patriarch_scope
	}
	remove_from_global_variable_map = {
		name = kog_rome
		key = flag:kog_seat_patriarch
	}
	add_to_global_variable_map = {
		name = kog_rome
		key = flag:kog_seat_patriarch
		value = scope:kog_rome_patriarch_scope
	}
}
