				}
			}
			set_variable = {
				name = kog_controller_var
				value = scope:kog_elysium_controller
			}
			scope:kog_elysium_controller = {
				ruler = {
					save_scope_as = kog_elysium_patriarch
				}
			}
			set_variable = {
				name = kog_patriarch_var
				value = scope:kog_elysium_patriarch
			}
		}

		# Mixtec
		create_struct = {
			struct_type = kog_seat
			struct_scope = kog_mixtec
		}
		global_var:kog_global = {
			set_variable = {
				name = kog_mixtec
				value = scope:kog_mixtec
			}
		}
		global_var:kog_global.var:kog_mixtec = {
			set_variable = {
				name = seat_name
				value = flag:kog_mixtec_name
			}
			set_variable = {
				name = kog_rite_var
				value = flag:kog_mixtec_rite
			}
			p:xF0A020 = { # Tututepec
				owner = {
					set_variable = kog_controls_mixtec
					save_scope_as = kog_mixtec_controller
				}
			}
			set_variable = {
				name = kog_controller_var
				value = scope:kog_mixtec_controller
			}
			scope:kog_mixtec_controller = {
				ruler = {
					save_scope_as = kog_mixtec_patriarch
				}
			}
			set_variable = {
				name = kog_patriarch_var
				value = scope:kog_mixtec_patriarch
			}
		}
	}

kog_initialize_metropolis_effect = {
	create_struct = {
		struct_type = kog_local
		struct_scope = kog_local
	}
	set_variable = { 
		name = kog_local
		value = scope:kog_local
	}
	var:kog_local = {
		# First Initialize Liturgical Rites
		if = {
			limit = {
				kog_is_eastern_orthodox_trigger = yes
			}
			if = {
				limit = {
					kog_is_rus_orthodox_trigger = yes
				}
				set_variable = {
					name = kog_liturgical_rite_var
					value = flag:kog_russian_rite
				}
			}
			else = {
				set_variable = {
					name = kog_liturgical_rite_var
					value = flag:kog_byzantine_rite
				}
			}
		}
		else_if = {
			limit = {
				kog_is_oriental_orthodox_trigger = yes
			}
			if = {
				limit = {
					always = no # TODO set up Syriac Rites
				}
				set_variable = {
					name = kog_liturgical_rite_var
					value = flag:kog_syro_antiochian_rite
				}
			}
			else_if = {
				limit = {
					always = no # TODO set up Syriac Rites
				}
				set_variable = {
					name = kog_liturgical_rite_var
					value = flag:kog_nestorian_rite
				}
			}
			else_if = {
				limit = {
					kog_is_oriental_armenian_trigger = yes
				}
				set_variable = {
					name = kog_liturgical_rite_var
					value = flag:kog_armenian_rite
				}
			}
			else = {
				set_variable = {
					name = kog_liturgical_rite_var
					value = flag:kog_armenian_rite
				}
			}
		}
		else_if = { # Basileia Romaion Rites
			limit = {
				kog_is_elysian_orthodox_trigger = yes
			}
			if = {
				limit = {
					kog_is_mixtec_orthodox_trigger = yes
				}
				set_variable = {
					name = kog_liturgical_rite_var
					value = flag:kog_elysian_rite
				}
			}
			else = {
				set_variable = {
					name = kog_liturgical_rite_var
					value = flag:kog_mixtec_rite
				}
			}
		}
		else = {
			set_variable = {
				name = kog_liturgical_rite_var
				value = flag:kog_roman_rite
			}
		}

		# Second, Set Relations Variables
		## Catholic Seats
		set_variable = {
			name = kog_rome_relations_var
			value = 0
		}
		set_variable = {
			name = kog_santiago_relations_var
			value = 0
		}
		set_variable = {
			name = kog_paris_relations_var
			value = 0
		}
		set_variable = {
			name = kog_koln_relations_var
			value = 0
		}
		set_variable = {
			name = kog_vienna_relations_var
			value = 0
		}

		## Orthodox Seats
		set_variable = {
			name = kog_constantinople_relations_var
			value = 0
		}
		set_variable = {
			name = kog_antioch_relations_var
			value = 0
		}
		set_variable = {
			name = kog_jerusalem_relations_var
			value = 0
		}
		set_variable = {
			name = kog_alexandria_relations_var
			value = 0
		}
		set_variable = {
			name = kog_moscow_relations_var
			value = 0
		}

		## Oriental Orthodox Seats
		set_variable = {
			name = kog_coptic_alexandria_relations_var
			value = 0
		}
		set_variable = {
			name = kog_syriac_antioch_relations_var
			value = 0
		}
		set_variable = {
			name = kog_etchmiadzin_relations_var
			value = 0
		}
		set_variable = {
			name = kog_addis_ababa_relations_var
			value = 0
		}
		set_variable = {
			name = kog_konak_relations_var
			value = 0
		}

		## Basileia Romaion Seats
		if = {
			limit = {
				basileia_is_active = yes
			}
			set_variable = {
				name = kog_elysium_relations_var
				value = 0
			}
			set_variable = {
				name = kog_mixtec_relations_var
				value = 0
			}
		}

		# Lastly, set other variables
		if = {
			limit = {
				kog_is_roman_catholic_trigger = yes
			}
			set_variable = {
				name = kog_votes_var
				value = kog_cardinal_college_value
			}

				name = kog_national_piety_var
				value = kog_piety_baseline_value
			}
		}

			limit = {

			}

				name = kog_votes_var
				value = kog_ecumenical_weight_value
			}
		}

			limit = {

			}

				name = kog_votes_var
				value = kog_synodal_influence_value

		}
		if = {

			}
			set_variable = {
				name = kog_votes_var
				value = kog_elysian_synkletos_value
			}
		}
		else = {

				name = kog_votes_var
				value = 0
			}
		}
