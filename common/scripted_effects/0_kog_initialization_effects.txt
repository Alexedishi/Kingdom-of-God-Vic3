kog_initialize_content_effect = {
	# If you are making a submod: DO NOT override this effect- override the individual effects downstream
	
	#### Step 1, Create Global Struct
	kog_initialize_global_struct_effect = yes
	
	#### Step 2, Create Denomination Structs
	kog_intialize_denomination_structs_effect = yes
	
	#### Step 3, Create Seat Structs
	kog_initialize_episcopal_seats_effect = yes
	
	#### Step 4, Create Patriarchs and Assign to Seats
	kog_initialize_patriarchs_effect = yes
	
	#### Step 5, Create the College of Cardinals
	kog_initialize_collegium_effect = yes
	
	#### Step 6, Initialize the Ranking Lists
	kog_initialize_variable_lists_effect = yes
	
	#### Step 7, Initialize Local Structs
	every_country = {
		limit = {
			kog_content_scope_trigger = yes
		}
		kog_initialize_metropolis_effect = yes
	}
	
	#### Step 8, Apply Any Country-specific details
	kog_specific_country_setups_effect = yes
}

kog_initialize_global_struct_effect = {
	c:PAP = {
		create_struct = {
			struct_type = flag:kog_global
			struct_scope = kog_global
		}
		set_global_variable = {
			name = kog_global
			value = scope:kog_global
		}
	}
}
