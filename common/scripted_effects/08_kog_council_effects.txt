####---------- Ecumenical Council Effects ---------- ####
kog_update_votes_effect = {
	if = {
		limit = {
			kog_country_is_roman_catholic_trigger = yes
		}
		set_variable = {
			name = kog_votes_var
			value = kog_cardinal_college_value
		}
	}
	if = {
		limit = {
			kog_country_is_eastern_orthodox_trigger = yes
		}
		set_variable = {
			name = kog_votes_var
			value = kog_ecumenical_weight_value
		}
	}
	if = {
		limit = {
			kog_country_is_oriental_orthodox_trigger = yes
		}
		set_variable = {
			name = kog_votes_var
			value = kog_synodal_influence_value
		}
	}
	if = {
		limit = {
			kog_country_is_elysian_orthodox_trigger = yes
		}
		set_variable = {
			name = kog_votes_var
			value = kog_elysian_synkletos_value
		}
	}
	else = {
		set_variable = {
			name = kog_votes_var
			value = 0
		}
	}
}

kog_update_council_votes_effect = {
	# Set Timeout Variable
	set_global_variable = {
		name = kog_votes_timeout_var
		days = 20
	}
	
	# Check Catholic Variables
	if = {
		limit = {
			kog_has_catholic_council_trigger = yes
		}
		global_var:kog_global.var:kog_catholic = {
			set_variable = {
				name = kog_council_support_var
				value = kog_catholic_support_value
			}
			set_variable = {
				name = kog_council_opposition_var
				value = kog_catholic_opposition_value
			}
			set_variable = {
				name = kog_council_tally_var
				value = kog_catholic_council_value
			}
		}
	}
	
	# Check Orthodox Variables
	if = {
		limit = {
			kog_has_orthodox_council_trigger = yes
		}
		global_var:kog_global.var:kog_orthodox = {
			set_variable = {
				name = kog_council_support_var
				value = kog_catholic_support_value
			}
			set_variable = {
				name = kog_council_opposition_var
				value = kog_catholic_opposition_value
			}
			set_variable = {
				name = kog_council_tally_var
				value = kog_catholic_council_value
			}
		}
	}
}

kog_cleanup_council_effect = {
	if = {
		limit = {
			has_modifier = modifier_kog_hosting_council
		}
		remove_modifier = modifier_kog_hosting_council
		add_modifier = {
			name = modifier_kog_hosted_council
			months = long_modifier_time
		}
	}
}
