####---------- Ecumenical Council Effects ---------- ####
kog_convoke_council_effect = {
	custom_tooltip = {
		text = kog_convoke_council_effect
		add_contextless_journal_entry = je_kog_$type$_council
		global_var:je_kog_$type$_council = {
			set_variable = {
				name = kog_council_year_var
				value = year
			}
			set_variable = {
				name = kog_council_location_var
				value = global_var:je_kog_$type$.var:kog_location_var
			}
		}
		kog_add_council_matter_effect = {
			type = $type$
			flag = $flag$
		}
	}
	add_modifier = {
		name = modifier_kog_convoked_council
		days = kog_5_years
		is_decaying = yes
	}
}

kog_add_council_matter_effect = {
	custom_tooltip = {
		text = kog_add_council_matter_effect
		set_local_variable = {
			name = kog_council_matter_var
			value = flag:kog_canon_lawgroup_$flag$
		}
		global_var:je_kog_$type$_council ?= {
			if = {
				limit = {
					NOT = {
						is_key_in_variable_map = {
							name = kog_council_matters_varmap
							target = flag:kog_canon_lawgroup_$flag$
						}
					}
				}
				add_to_variable_map = {
					name = kog_council_matters_varmap
					key = flag:kog_canon_lawgroup_$flag$
					value = 0
				}
			}
		}
	}
}

kog_tally_canon_law_votes_effect = {
	set_variable = {
		name = $flag$
		value = {
			value = 0
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_kog_$type$_council
					var:kog_canon_lawgroup_apostolic_succession_var ?= flag:$flag$
				}
				add = var:kog_votes_var
			}
		}
	}
}
