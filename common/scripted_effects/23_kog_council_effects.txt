kog_convoke_council_effect = {
	custom_tooltip = {
		text = kog_convoke_council_effect
		save_scope_as = kog_council_caller
		var:kog_global_faith_journal_var = {
			add_to_variable_map = {
				name = kog_ecumenical_council_map
				key = flag:kog_council_year_var
				value = year
			}
			com_change_variable = {
				name = kog_council_counter_var
				value = 1
			}
		}
		if = {
			limit = {
				kog_is_autocephalous_trigger = yes
			}
			var:kog_global_faith_journal_var = {
				add_to_variable_map = {
					name = kog_ecumenical_council_map
					key = flag:kog_council_location_var
					value = scope:kog_council_caller.capital
				}
			}
		}
		else = {
			var:kog_global_faith_journal_var = {
				add_to_variable_map = {
					name = kog_ecumenical_council_map
					key = flag:kog_council_location_var
					value = var:kog_mother_church_var.var:kog_seat_state_region_var
				}
			}
		}
	}
	add_modifier = {
		name = modifier_kog_convoked_council
		days = kog_5_years
		is_decaying = yes
	}
}

kog_add_canon_law_group_to_council_effect = {
	custom_tooltip = {
		text = kog_add_canon_law_group_to_council_effect
		set_local_variable = {
			name = kog_council_matter_var
			value = flag:$canon_law_group$
		}
		var:kog_global_faith_journal_var = {
			if = {
				limit = {
					NOT = {
						is_key_in_variable_map = {
							name = kog_ecumenical_council_map
							target = flag:$canon_law_group$
						}
					}
				}
				add_to_variable_map = {
					name = kog_ecumenical_council_map
					key = flag:$canon_law_group$
					value = 0
				}
			}
		}
	}
}
