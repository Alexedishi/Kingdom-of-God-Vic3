kog_convoke_council_effect = {
	custom_tooltip = {
		text = kog_convoke_council_effect
		save_scope_as = kog_council_caller
		var:kog_global_faith_journal_var = {
			add_to_variable_map = {
				name = kog_ecumenical_council_map
				key = flag:kog_council_year_var
				value = year
			}
			com_change_variable = {
				name = kog_council_counter_var
				value = 1
			}
		}
		if = {
			limit = {
				kog_is_autocephalous_trigger = yes
			}
			var:kog_global_faith_journal_var = {
				add_to_variable_map = {
					name = kog_ecumenical_council_map
					key = flag:kog_council_location_var
					value = scope:kog_council_caller.capital
				}
			}
		}
		else = {
			var:kog_global_faith_journal_var = {
				add_to_variable_map = {
					name = kog_ecumenical_council_map
					key = flag:kog_council_location_var
					value = var:kog_mother_church_var.var:kog_seat_state_region_var
				}
			}
		}
	}
	add_modifier = {
		name = modifier_kog_convoked_council
		days = kog_5_years
		is_decaying = yes
	}
}

kog_add_canon_law_group_to_council_effect = {
	custom_tooltip = {
		text = kog_add_canon_law_group_to_council_effect
		set_local_variable = {
			name = kog_council_matter_var
			value = flag:$canon_law_group$
		}
		var:kog_global_faith_journal_var = {
			if = {
				limit = {
					NOT = {
						is_key_in_variable_map = {
							name = kog_ecumenical_council_map
							target = flag:$canon_law_group$
						}
					}
				}
				add_to_variable_map = {
					name = kog_ecumenical_council_map
					key = flag:$canon_law_group$
					value = var:$canon_law_group$
				}
			}
		}
	}
}

kog_determine_leading_council_positions_effect = {
	# This is split from kog_tally_council_votes_effect for readability; less efficient, but this is already hell to debug
	# Step 1, Check if the canon law group is up for debate
	# Step 2, Find the highest scoring canon law via ordered iterator and save its scope
	# Step 3, Compare that canon law's score to others in the map; if there is a tie, set the saved canon law scope to the active canon law in that group
	# Step 4, Update the leading canon law in the council map; *any* tie means the active canon law will not change
	if = {
		limit = {
			is_key_in_variable_map = {
				name = kog_ecumenical_council_map
				target = flag:kog_canon_lawgroup_apostolic_succession
			}
		}
		ordered_key_in_variable_map = {
			variable = kog_apostolic_succession_votes_map
			order_by = {
				value = "prev.variable_map(kog_apostolic_succession_votes_map|this)"
			}
			position = 0
			check_range_bounds = no
			save_scope_as = kog_canon_lawgroup_apostolic_succession_leader
		}
		if = {
			limit = {
				any_key_in_variable_map = {
					variable = kog_apostolic_succession_votes_map
					prev.var:kog_canon_lawgroup_apostolic_succession != this
					"prev.variable_map(kog_apostolic_succession_votes_map|this)" = "prev.variable_map(kog_apostolic_succession_votes_map|scope:kog_canon_lawgroup_apostolic_succession_leader)"
				}
			}
			var:kog_canon_lawgroup_apostolic_succession = {
				save_scope_as = kog_canon_lawgroup_apostolic_succession_leader
			}
		}
		com_update_value_in_variable_map = {
			name = kog_ecumenical_council_map
			key = flag:kog_canon_lawgroup_apostolic_succession
			value = scope:kog_canon_lawgroup_apostolic_succession_value
		}
	}
}
