####---------- Ecumenical Council Effects ---------- ####
kog_convoke_council_effect = { }

kog_add_council_matter_effect = {
	custom_tooltip = {
		text = kog_add_council_matter_effect
		flag:$flag$ = {
			save_temporary_scope_as = kog_council_matter_scope
		}
		if = {
			limit = {
				kog_country_is_roman_catholic_trigger = yes
			}
			global_var:je_kog_catholic = {
				save_scope_as = kog_target_denomination_scope
			}
		}
		else_if = {
			limit = {
				kog_country_is_eastern_orthodox_trigger = yes
			}
			global_var:je_kog_orthodox = {
				save_scope_as = kog_target_denomination_scope
			}
		}
		scope:kog_target_denomination_scope ?= {
			if = {
				limit = {
					NOT = {
						is_key_in_variable_map = {
							name = kog_council_matters_map
							target = flag:$flag$
						}
					}
				}
				add_to_variable_map = {
					name = kog_council_matters_map
					value = flag:$flag$
				}
			}
		}
	}
}

kog_tally_council_votes_effect = {
	# Set Timeout Variable
	set_global_variable = {
		name = kog_votes_timeout_var
		days = 20
	}
	
	# Check Catholic Variables
	if = {
		limit = {
			kog_has_catholic_council_trigger = yes
		}
		global_var:je_kog_catholic = {
			set_variable = {
				name = kog_council_support_var
				value = kog_catholic_support_value
			}
			set_variable = {
				name = kog_council_opposition_var
				value = kog_catholic_opposition_value
			}
			set_variable = {
				name = kog_council_tally_var
				value = kog_catholic_council_value
			}
		}
	}
	
	# Check Orthodox Variables
	if = {
		limit = {
			kog_has_orthodox_council_trigger = yes
		}
		global_var:je_kog_orthodox = {
			set_variable = {
				name = kog_council_support_var
				value = kog_catholic_support_value
			}
			set_variable = {
				name = kog_council_opposition_var
				value = kog_catholic_opposition_value
			}
			set_variable = {
				name = kog_council_tally_var
				value = kog_catholic_council_value
			}
		}
	}
}

kog_cleanup_council_effect = {
	if = {
		limit = {
			has_modifier = modifier_kog_hosting_council
		}
		remove_modifier = modifier_kog_hosting_council
		add_modifier = {
			name = modifier_kog_hosted_council
			months = long_modifier_time
		}
	}
}
