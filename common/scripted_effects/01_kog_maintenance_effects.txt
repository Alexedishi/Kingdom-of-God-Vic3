# Monthly Maintenance Effects
kog_update_ranking_lists = {
	# Set Timeout Variable
	set_global_variable = {
		name = kog_ranking_timeout_var
		days = 20 # Should be enough to get past all monthly pulses
	}

	# Clear All Lists
	clear_global_variable_list = kog_catholic_ranking
	clear_global_variable_list = kog_orthodox_ranking
	clear_global_variable_list = kog_oriental_ranking
	clear_global_variable_list = kog_elysian_ranking

	# Build & Sort Catholic List
	every_country = {
		limit = {
			kog_is_roman_catholic_trigger = yes
		}
		add_to_global_variable_list = {
			name = kog_catholic_ranking
			target = this
		}
	}
	sort_global_variable_list = {
		name = kog_catholic_ranking
		order = prestige
	}

	# Build & Sort Orthodox List
	every_country = {
		limit = {
			kog_is_eastern_orthodox_trigger = yes
		}
		add_to_global_variable_list = {
			name = kog_orthodox_ranking
			target = this
		}
	}
	sort_global_variable_list = {
		name = kog_orthodox_ranking
		order = prestige
	}

	# Build & Sort Oriental List
	every_country = {
		limit = {
			kog_is_oriental_orthodox_trigger = yes
		}
		add_to_global_variable_list = {
			name = kog_oriental_ranking
			target = this
		}
	}
	sort_global_variable_list = {
		name = kog_oriental_ranking
		order = prestige
	}

	# Build & Sort Elysian List
	if = {
		limit = {
			basileia_is_active = yes
		}
		every_country = {
			limit = {
				kog_is_elysian_orthodox_trigger = yes
			}
			add_to_global_variable_list = {
				name = kog_elysian_ranking
				target = this
			}
		}
		sort_global_variable_list = {
			name = kog_elysian_ranking
			order = prestige
		}
	}
}

# Council Votes Upkeep
kog_update_votes_effect = {
	if = {
		limit = {
			kog_is_roman_catholic_trigger = yes
		}
		set_variable = {
			name = kog_votes_var
			value = kog_cardinal_college_value
		}
	}
	if = {
		limit = {
			kog_is_eastern_orthodox_trigger = yes
		}
		set_variable = {
			name = kog_votes_var
			value = kog_ecumenical_weight_value
		}
	}
	if = {
		limit = {
			kog_is_oriental_orthodox_trigger = yes
		}
		set_variable = {
			name = kog_votes_var
			value = kog_synodal_influence_value
		}
	}
	if = {
		limit = {
			kog_is_elysian_orthodox_trigger = yes
		}
		set_variable = {
			name = kog_votes_var
			value = kog_elysian_synkletos_value
		}
	}
	else = {
		set_variable = {
			name = kog_votes_var
			value = 0
		}
	}
}

# Patriarch Update Effects
kog_update_rome_effect = {
	# First, check for the papacy
	if = {
		limit = {
			c:PAP ?= {
				has_law = law_type:law_theocracy
			}
		}
		c:PAP ?= {
			ruler = {
				save_scope_as = kog_rome_patriarch_scope
			}
		}
	}
	# Failing that, attempt to find character in void
	else_if = {
		limit = {
			any_character_in_void = {
				has_template = template_kog_rome_patriarch
			}
		}
		random_character_in_void = {
			limit = {
				has_template = template_kog_rome_patriarch
			}
			save_scope_as = kog_rome_patriarch_scope
		}
	}
	# Failing that, generate a new character
	else = {
		# Attempt to generate using the Rome controller
		if = {
			limit = {
				any_country = {
					has_variable = kog_controls_rome
					kog_is_christian_trigger = yes
				}
			}
			random_country = {
				limit = {
					has_variable = kog_controls_rome
				}
				create_character = {
					template = template_kog_rome_patriarch
					religion = rel:orthodox
					on_created = {
						place_character_in_void = 180 # 15 year shelf life
					}
				}
			}
		}
		# Fallback to an Italian Catholic
		else = {
			create_character = {
				template = template_kog_rome_patriarch
				culture = cu:south_italian
				religion = rel:catholic
				on_created = {
					place_character_in_void = 180 # 15 year shelf life
				}
			}
		}
		# Update Relations Variables
		every_country = {
			limit = {
				kog_is_christian_trigger = yes
			}
			if = {
				limit = {
					NOT = {
						has_variable = kog_rome_relations_var
					}
				}
				set_variable = {
					name = kog_rome_relations_var
					value = 0
				}
			}
			change_variable = {
				name = kog_rome_relations_var
				add = -25
			}
			clamp_variable = {
				name = kog_rome_relations_var
				min = 0
				max = 50
			}
			post_notification = kog_rome_replaced
		}
		# Save the new character
		random_character_in_void = {
			limit = {
				has_template = template_kog_rome_patriarch
			}
			save_scope_as = kog_rome_patriarch_scope
		}
	}
	# Finally, set the new variable
	global_var:kog_global.var:kog_rome = {
		set_variable = {
			name = kog_rome_patriarch_var
			value = scope:kog_rome_patriarch_scope
		}
	}
}

kog_update_konak_effect = {
	# First, try to find the character in the void
	if = {
		limit = {
			any_character_in_void = {
				has_template = template_kog_konak_patriarch
			}
		}
		random_character_in_void = {
			limit = {
				has_template = template_kog_konak_patriarch
			}
			save_scope_as = kog_konak_patriarch_scope
		}
	}
	# Failing that, generate a new character
	else = {
		# Attempt to generate using the konak controller
		if = {
			limit = {
				any_country = {
					has_variable = kog_controls_konak
					kog_is_christian_trigger = yes
				}
			}
			random_country = {
				limit = {
					has_variable = kog_controls_konak
				}
				create_character = {
					template = template_kog_konak_patriarch
					on_created = {
						place_character_in_void = 180 # 15 year shelf life
					}
				}
			}
		}
		# Fallback to Assyrian Oriental Orthodox
		else = {
			create_character = {
				template = template_kog_konak_patriarch
				culture = cu:assyrian
				religion = rel:oriental_orthodox # This is inaccurate, I know
				on_created = {
					place_character_in_void = 180 # 15 year shelf life
				}
			}
		}
		# Update Relations Variables
		every_country = {
			limit = {
				kog_is_christian_trigger = yes
			}
			if = {
				limit = {
					NOT = {
						has_variable = kog_konak_relations_var
					}
				}
				set_variable = {
					name = kog_konak_relations_var
					value = 0
				}
			}
			change_variable = {
				name = kog_konak_relations_var
				add = -25
			}
			clamp_variable = {
				name = kog_konak_relations_var
				min = 0
				max = 50
			}
			post_notification = kog_konak_replaced
		}
		# Save the new character
		random_character_in_void = {
			limit = {
				has_template = template_kog_konak_patriarch
			}
			save_scope_as = kog_konak_patriarch_scope
		}
	}
	# Finally, set the new variable
	global_var:kog_global.var:kog_konak = {
		set_variable = {
			name = kog_konak_patriarch_var
			value = scope:kog_konak_patriarch_scope
		}
	}
}
