kog_schism_monthly_progress_effect = {
	var:kog_local = {
		set_variable = {
			name = kog_schism_var
			value = {
				add = var:kog_schism_var
				add = owner.modifier:country_monthly_schism_progress_add
				max = 120
				min = 0
			}
		}
	}
}

kog_add_schism_progress_effect = {
	custom_tooltip = {
		text = kog_add_schism_progress_effect
		set_local_variable = {
			name = kog_schism_progress
			value = $value$
		}
		set_variable = {
			name = kog_schism_var
			value = {
				add = var:kog_schism_var
				add = $value$
				max = 120
				min = 0
			}
		}
	}
}
