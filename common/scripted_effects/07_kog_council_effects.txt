####---------- Ecumenical Council Effects ---------- ####
kog_update_votes_effect = {
	if = {
		limit = {
			kog_is_roman_catholic_trigger = yes
		}
		set_variable = {
			name = kog_votes_var
			value = kog_cardinal_college_value
		}
	}
	if = {
		limit = {
			kog_is_eastern_orthodox_trigger = yes
		}
		set_variable = {
			name = kog_votes_var
			value = kog_ecumenical_weight_value
		}
	}
	if = {
		limit = {
			kog_is_oriental_orthodox_trigger = yes
		}
		set_variable = {
			name = kog_votes_var
			value = kog_synodal_influence_value
		}
	}
	if = {
		limit = {
			kog_is_elysian_orthodox_trigger = yes
		}
		set_variable = {
			name = kog_votes_var
			value = kog_elysian_synkletos_value
		}
	}
	else = {
		set_variable = {
			name = kog_votes_var
			value = 0
		}
	}
}

kog_update_council_votes_effect = {
	# Set Timeout Variable
	set_global_variable = {
		name = kog_votes_timeout_var
		days = 20
	}
	
	# Check Catholic Variables
	if = {
		limit = {
			has_global_variable = kog_catholic_matter_var
		}
		set_global_variable = {
			name = kog_catholic_support_var
			value = kog_catholic_support_value
		}
		set_global_variable = {
			name = kog_catholic_opposition_var
			value = kog_catholic_opposition_value
		}
		set_global_variable = {
			name = kog_catholic_council_var
			value = kog_catholic_council_value
		}
	}
	else = {
		remove_global_variable = kog_catholic_support_var
		remove_global_variable = kog_catholic_opposition_var
		remove_global_variable = kog_catholic_council_var

		}
		set_global_variable = {
			name = kog_orthodox_council_var
			value = kog_orthodox_council_value
		}
	}
	else = {
		remove_global_variable = kog_orthodox_support_var
		remove_global_variable = kog_orthodox_opposition_var
		remove_global_variable = kog_orthodox_council_var

	# Check Oriental Variables

			has_global_variable = kog_oriental_matter_var

		set_global_variable = {
			name = kog_oriental_opposition_var
			value = kog_oriental_opposition_value
		}

			name = kog_oriental_council_var

kog_cleanup_council_effect = {
	if = {
		limit = {
			has_modifier = modifier_kog_hosting_council
		}
		remove_modifier = modifier_kog_hosting_council
		add_modifier = {
			name = modifier_kog_hosted_council
			months = long_modifier_time
		}
	}
}
