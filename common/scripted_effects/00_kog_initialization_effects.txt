kog_initialize_content_effect = {
	# Executed in kog_game_start via on_game_started
	# If you are making a sub-mod: DO NOT override anything in this effect- override the individual effects downstream
	
	#### Step 1, Activate Contextless Journals
	kog_activate_contextless_journals_effect = yes
	
	#### Step 2, Initialize Compatibility Variable Map
	kog_initialize_database_variable_map_effect = yes
	
	#### Step 3, Initialize Denomination Journal Variables
	kog_initialize_roman_catholic_variables_effect = yes
	kog_initialize_eastern_orthodox_variables_effect = yes
	#	kog_initialize_oriental_orthodox_variables_effect = yes
	#	kog_initialize_lutheran_protestant_variables_effect = yes
	#	kog_initialize_anglican_protestant_variables_effect = yes
	if = {
		limit = {
			basileia_is_active = yes
		}
		kog_initialize_elysian_orthodox_variables_effect = yes
	}
	
	#### Step 4, Create Script Containers
	kog_initialize_rome_container_effect = yes
	kog_initialize_santiago_container_effect = yes
	kog_initialize_paris_container_effect = yes
	kog_initialize_koln_container_effect = yes
	kog_initialize_vienna_container_effect = yes
	kog_initialize_constantinople_container_effect = yes
	kog_initialize_antioch_container_effect = yes
	kog_initialize_jerusalem_container_effect = yes
	kog_initialize_alexandria_container_effect = yes
	kog_initialize_moscow_container_effect = yes
	if = {
		limit = {
			basileia_is_active = yes
		}
		kog_initialize_elysium_container_effect = yes
		kog_initialize_onondaga_container_effect = yes
		kog_initialize_tututepec_container_effect = yes
		kog_initialize_cahokia_container_effect = yes
	}
	kog_initialize_container_lists_effect = yes
	
	#### Step 5, Create Patriarchs, Capture Heads of Faith, and Assign Seat Controllers
	if = {
		limit = {
			basileia_is_active = yes
		}
		kog_initialize_1736_patriarchs_effect = yes
	}
	else = {
		kog_initialize_1836_patriarchs_effect = yes
	}
	kog_initialize_seat_controllers_effect = yes
	
	#### Step 6, Create the College of Cardinals
	kog_initialize_collegium_effect = yes
	
	#### Step 7, Initialize the Faith Ranking Lists
	kog_initialize_variable_lists_effect = yes
	
	#### Step 8, Initialize Local Script Container and Variables
	every_country = {
		limit = {
			kog_content_scope_trigger = yes
		}
		kog_initialize_metropolis_effect = yes
	}
	
	#### Step 9, Apply Any Country-specific details
	kog_specific_country_setups_effect = yes
	
	#### Step 10, Apply Final Effects
	kog_initialization_cleanup_effect = yes
}

kog_initialize_container_lists_effect = {
	# Build Global Seat List
	every_container = {
		limit = {
			has_tag = episcopal_seat
		}
		add_to_global_variable_list = {
			name = kog_global_seat_list
			target = this
		}
	}
	
	# Build Catholic Seat List
	every_container = {
		limit = {
			has_tag = episcopal_seat
			has_tag = roman_catholic
		}
		add_to_global_variable_list = {
			name = kog_catholic_seat_list
			target = this
		}
		global_var:je_kog_catholic = {
			add_to_variable_list = {
				name = kog_seats_list
				target = prev
			}
		}
	}
	
	# Build Orthodox Seat List
	every_container = {
		limit = {
			has_tag = episcopal_seat
			has_tag = eastern_orthodox
		}
		add_to_global_variable_list = {
			name = kog_orthodox_seat_list
			target = this
		}
		global_var:je_kog_orthodox = {
			add_to_variable_list = {
				name = kog_seats_list
				target = prev
			}
		}
	}
	
	# Build Oriental Seat List
	every_container = {
		limit = {
			has_tag = episcopal_seat
			has_tag = oriental_orthodox
		}
		add_to_global_variable_list = {
			name = kog_oriental_seat_list
			target = this
		}
		global_var:je_kog_oriental = {
			add_to_variable_list = {
				name = kog_seats_list
				target = prev
			}
		}
	}
	
	# Build Elysian Seat List
	every_container = {
		limit = {
			has_tag = episcopal_seat
			has_tag = elysian_orthodox
		}
		add_to_global_variable_list = {
			name = kog_elysian_seat_list
			target = this
		}
		global_var:je_kog_elysian = {
			add_to_variable_list = {
				name = kog_seats_list
				target = prev
			}
		}
	}
}

		}
	}
	# Sort the list
	sort_global_variable_list = {
		name = kog_cardinals_list
		order = kog_character_gravitas_value
	}
}

kog_initialize_metropolis_effect = {
	# Step 0: Save Country Scope; this might not always have the country as root
	save_scope_as = kog_root_scope
	
	# Step 1: Create Script Container
	create_container = {
		name = kog_local
		tags = { local_metropolis }
		parent = scope:kog_root_scope
		save_scope_as = kog_local
	}
	set_variable = {
		name = kog_local
		value = scope:kog_local
	}
	
	
	# Step 2: Initialize Liturgical Rites & Languages
	if = {
		limit = {
			kog_country_is_eastern_orthodox_trigger = yes
		}
		var:kog_local = {
			set_variable = {
				name = kog_liturgical_rite_var
				value = flag:kog_byzantine_rite
			}
			set_variable = {
				name = kog_mother_church_var
				value = flag:kog_constantinople
			}
		}
		if = {
			limit = {
				country_has_primary_culture = cu:russian
			}
			var:kog_local = {
				set_variable = {
					name = kog_liturgical_language_var
					value = flag:kog_language_slavonic
				}
			}
		}
		else = {
			var:kog_local = {
				set_variable = {
					name = kog_liturgical_language_var
					value = flag:kog_language_khoine
				}
			}
		}
	}
	else_if = {
		limit = {
			kog_country_is_oriental_orthodox_trigger = yes
		}
		var:kog_local = {
			set_variable = {
				name = kog_liturgical_language_var
				value = flag:kog_language_coptic # TODO replace this once Oriental Churches are added
			}
			set_variable = {
				name = kog_mother_church_var
				value = flag:kog_coptic_alexandria
			}
		}
		if = {
			limit = {
				country_has_primary_culture = cu:kurdish
			}
			var:kog_local = {
				set_variable = {
					name = kog_liturgical_rite_var
					value = flag:kog_syro_antiochian_rite
				}
			}
		}
		else_if = {
			limit = {
				country_has_primary_culture = cu:assyrian
			}
			var:kog_local = {
				set_variable = {
					name = kog_liturgical_rite_var
					value = flag:kog_nestorian_rite
				}
			}
		}
		else_if = {
			limit = {
				country_has_primary_culture = cu:armenian
			}
			var:kog_local = {
				set_variable = {
					name = kog_liturgical_rite_var
					value = flag:kog_armenian_rite
				}
			}
		}
		else = {
			var:kog_local = {
				set_variable = {
					name = kog_liturgical_rite_var
					value = flag:kog_alexandrian_rite
				}
			}
		}
	}
	else_if = {
		# Basileia Romaion Rites
		limit = {
			kog_country_is_elysian_orthodox_trigger = yes
		}
		if = {
			limit = {
				country_has_primary_culture = cu:mixtec
			}
			var:kog_local = {
				set_variable = {
					name = kog_liturgical_rite_var
					value = flag:kog_mixtec_rite
				}
				set_variable = {
					name = kog_mother_church_var
					value = flag:kog_elysium
				}
			}
		}
		else = {
			var:kog_local = {
				set_variable = {
					name = kog_liturgical_rite_var
					value = flag:kog_elysian_rite
				}
			}
		}
		var:kog_local = {
			set_variable = {
				name = kog_liturgical_language_var
				value = flag:kog_language_khoine # TODO replace this once Elysian Churches are added
			}
		}
	}
	else = {
		var:kog_local = {
			set_variable = {
				name = kog_liturgical_rite_var
				value = flag:kog_roman_rite
			}
			set_variable = {
				name = kog_liturgical_language_var
				value = flag:kog_language_latin
			}
			set_variable = {
				name = kog_mother_church_var
				value = flag:kog_rome
			}
		}
	}
	
	# Step 3: Set Relations Variables
	save_scope_as = kog_country_scope
	every_in_global_list = {
		variable = kog_global_seat_list
		add_to_variable_map = {
			name = kog_relations_map
			key = scope:kog_country_scope
			value = kog_opinion_start_value
		}
		add_to_variable_map = {
			name = kog_opinions_map
			key = scope:kog_country_scope
			value = kog_opinion_start_value
		}
		add_to_variable_map = {
			name = kog_diplomacy_map
			key = scope:kog_country_scope
			value = 0
		}
	}
	
	# Step 4, Set Up Country Modifiers & Variables
	var:kog_local = {
		set_variable = {
			name = kog_canon_law_compliance_var
			value = root.kog_canon_law_compliance_value
		}
		set_variable = {
			name = kog_national_piety_var
			value = root.kog_national_piety_target_value
		}
		set_variable = {
			name = kog_votes_var
			value = root.kog_votes_value
		}
	}
	kog_update_national_piety_effect = yes
}
