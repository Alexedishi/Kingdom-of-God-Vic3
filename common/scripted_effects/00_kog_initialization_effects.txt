kog_initialize_rome_variable_map_effect = {
	p:xF0A020 = {
		owner = {
			save_scope_as = kog_rome_controller
		}
	}
	add_to_global_variable_map = {
		name = kog_rome
		key = flag:kog_seat_id
		value = flag:kog_rome
	}
	add_to_global_variable_map = {
		name = kog_rome
		key = flag:kog_seat_name
		value = flag:kog_rome_name
	}
	add_to_global_variable_map = {
		name = kog_tempkey
		key = flag:kog_seat_rank
		value = flag:kog_major_seat
	}
	add_to_global_variable_map = {
		name = kog_rome
		key = flag:kog_seat_religion
		value = rel:catholic
	}
	add_to_global_variable_map = {
		name = kog_rome
		key = flag:kog_seat_liturgical_language
		value = flag:kog_language_latin
	}
	add_to_global_variable_map = {
		name = kog_rome
		key = flag:kog_seat_liturgical_rite
		value = flag:kog_roman_rite
	}
	add_to_global_variable_map = {
		name = kog_rome
		key = flag:kog_seat_location
		value = s:STATE_LAZIO
	}
	add_to_global_variable_map = {
		name = kog_rome
		key = flag:kog_seat_controller
		value = scope:kog_rome_controller
	}
	scope:kog_rome_controller = {
		ruler = {
			save_scope_as = kog_rome_patriarch
		}
	}
	add_to_global_variable_map = {
		name = kog_rome
		key = flag:kog_seat_patriarch
		value = scope:kog_rome_patriarch
	}
}
