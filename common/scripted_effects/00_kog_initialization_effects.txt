kog_initialize_collegium_effect = {
	# First, place extant Catholic Devout leaders into the Cardinal list
	every_country = {
		limit = {
			kog_country_is_roman_catholic_trigger = yes
		}
		ig:ig_devout = {
			leader = {
				add_to_global_variable_list = {
					name = kog_cardinals_list
					target = this
				}
			}
		}
	}
	# Add Void Cardinals to List
	every_character_in_void = {
		limit = {
			has_trait = kog_cardinal
		}
		add_to_global_variable_list = {
			name = kog_cardinals_list
			target = this
		}
	}
	# Sort the list
	sort_global_variable_list = {
		name = kog_cardinals_list
		order = kog_character_gravitas_value
	}
}
