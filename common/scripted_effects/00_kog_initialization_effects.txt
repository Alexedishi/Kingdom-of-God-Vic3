kog_initialize_content_effect = {
	# If you are making a sub-mod: DO NOT override anything in this effect- override the individual effects downstream
	
	#### Step 1, Activate Contextless Journals
	kog_activate_contextless_journals_effect = yes
	
	#### Step 2, Initialize Denomination Script Containers
	kog_initialize_religion_script_containers_effect = yes
	
	#### Step 3, Create Patriarchs and Capture Heads of Faith
	if = {
		limit = {
			basileia_is_active = yes
		}
		kog_initialize_1736_patriarchs_effect = yes
	}
	else = {
		kog_initialize_1836_patriarchs_effect = yes
	}
	
	#### Step 4, Create Script Containers
	kog_initialize_rome_container_effect = yes
	kog_initialize_santiago_container_effect = yes
	kog_initialize_paris_container_effect = yes
	kog_initialize_koln_container_effect = yes
	kog_initialize_vienna_container_effect = yes
	kog_initialize_constantinople_container_effect = yes
	kog_initialize_antioch_container_effect = yes
	kog_initialize_jerusalem_container_effect = yes
	kog_initialize_alexandria_container_effect = yes
	kog_initialize_moscow_container_effect = yes
	if = {
		limit = {
			basileia_is_active = yes
		}
		kog_initialize_elysium_container_effect = yes
		kog_initialize_onondaga_container_effect = yes
		kog_initialize_tututepec_container_effect = yes
	}
	kog_initialize_container_lists_effect = yes
	
	#### Step 5, Create the College of Cardinals
	kog_initialize_collegium_effect = yes
	
	#### Step 6, Initialize the Ranking Lists
	kog_initialize_variable_lists_effect = yes
	
	#### Step 7, Initialize Journal and Variables
	every_country = {
		limit = {
			kog_content_scope_trigger = yes
		}
		kog_initialize_metropolis_effect = yes
	}
	
	#### Step 8, Apply Any Country-specific details
	kog_specific_country_setups_effect = yes
	
	#### Step 9, Create Mod Compatibility Maps & Apply Final Effects
	kog_initialize_basielia_romaion_variable_map_effect = yes
	kog_initialize_manaflow_variable_map_effect = yes
	kog_initialization_cleanup_effect = yes
}

		tags = { episcopal_seat western_christian roman_catholic papacy }

			c:PAP = {
				save_scope_as = kog_rome_controller
			}
			set_variable = {
				name = kog_seat_name
				value = flag:kog_rome_name
			}
			set_variable = {
				name = kog_seat_header
				value = flag:kog_rome_header
			}
			set_variable = {
				name = kog_seat_rank
				value = flag:kog_major_seat
			}
			set_variable = {
				name = kog_seat_religion
				value = rel:catholic
			}
			set_variable = {
				name = kog_seat_liturgical_language
				value = flag:kog_language_latin
			}
			set_variable = {
				name = kog_seat_liturgical_rite
				value = flag:kog_roman_rite
			}
			set_variable = {
				name = kog_seat_location
				value = s:STATE_LAZIO
			}
			set_variable = {
				name = kog_seat_province
				value = p:xB5726E
			}
			set_variable = {
				name = kog_seat_culture
				value = cu:south_italian
			}
			set_variable = {
				name = kog_seat_foundation
				value = 42 # This is a total shot in the dark
			}
			set_variable = {
				name = kog_seat_controller
				value = scope:kog_rome_controller
			}
			scope:kog_rome_controller = {
				ruler = {
					save_scope_as = kog_rome_patriarch
				}
			}

				name = kog_seat_patriarch
				value = scope:kog_rome_patriarch
			}
		}
	}
}

kog_initialize_container_lists_effect = {
	# Build Global Seat List
	every_container = {
		limit = {
			has_tag = episcopal_seat
		}
		add_to_global_variable_list = {
			name = kog_global_seat_list
			target = this
		}
	}
	
	# Build Catholic Seat List
	every_container = {
		limit = {
			has_tag = episcopal_seat
			has_tag = roman_catholic
		}
		add_to_global_variable_list = {
			name = kog_catholic_seat_list
			target = this
		}
	}
	
	# Build Orthodox Seat List
	every_container = {
		limit = {
			has_tag = episcopal_seat
			has_tag = eastern_orthodox
		}
		add_to_global_variable_list = {
			name = kog_orthodox_seat_list
			target = this
		}
	}
	
	# Build Oriental Seat List
	every_container = {
		limit = {
			has_tag = episcopal_seat
			has_tag = oriental_orthodox
		}
		add_to_global_variable_list = {
			name = kog_oriental_seat_list
			target = this
		}
	}
	
	# Build Elysian Seat List

kog_initialize_metropolis_effect = {
	# Step 0: Save Country Scope; this might not always have the country as root
	save_scope_as = kog_root_scope
	
	# Step 1: Create Script Container
	create_container = {
		name = kog_local
		tags = { local_metropolis }
		parent = scope:kog_root_scope
		save_scope_as = kog_local
		on_created = {
			scope:kog_root_scope = {
				set_variable = {
					name = kog_local
					value = scope:kog_local
				}
			}
		}
	}
	
	# Step 2: Initialize Liturgical Rites & Languages
	if = {
		limit = {
			kog_country_is_eastern_orthodox_trigger = yes
		}
		var:kog_local = {
			set_variable = {
				name = kog_liturgical_rite_var
				value = flag:kog_byzantine_rite
			}
			set_variable = {
				name = kog_mother_church_var
				value = flag:kog_constantinople
			}
		}
		if = {
			limit = {
				country_has_primary_culture = cu:russian
			}
			var:kog_local = {
				set_variable = {
					name = kog_liturgical_language_var
					value = flag:kog_language_slavonic
				}
			}
		}
		else = {
			var:kog_local = {
				set_variable = {
					name = kog_liturgical_language_var

				}
			}
		}
	}
	else_if = {
		limit = {
			kog_country_is_oriental_orthodox_trigger = yes
		}
		var:kog_local = {
			set_variable = {
				name = kog_liturgical_language_var

					name = kog_liturgical_rite_var
					value = flag:kog_syro_antiochian_rite
				}
			}
		}
		else_if = {
			limit = {
				country_has_primary_culture = cu:assyrian

				set_variable = {
					name = kog_liturgical_rite_var
					value = flag:kog_nestorian_rite
				}
			}
		}
		else_if = {
			limit = {
				country_has_primary_culture = cu:armenian
			}
			var:kog_local = {
				set_variable = {
					name = kog_liturgical_rite_var
					value = flag:kog_armenian_rite
				}
			}
		}
		else = {
			var:kog_local = {
				set_variable = {
					name = kog_liturgical_rite_var
					value = flag:kog_alexandrian_rite
				}
			}
		}
	}
	else_if = {
		# Basileia Romaion Rites
		limit = {
			kog_country_is_elysian_orthodox_trigger = yes
		}
		if = {
			limit = {
				country_has_primary_culture = cu:mixtec
			}

				set_variable = {
					name = kog_liturgical_rite_var
					value = flag:kog_mixtec_rite
				}

					name = kog_mother_church_var
					value = flag:kog_elysium

			}

			var:kog_local = {
				set_variable = {
					name = kog_liturgical_rite_var
					value = flag:kog_elysian_rite
				}
			}
		}
		var:kog_local = {
			set_variable = {
				name = kog_liturgical_language_var
				value = flag:kog_language_khoine # TODO replace this once Elysian Churches are added
			}
		}
	}
	else = {
		var:kog_local = {
			set_variable = {
				name = kog_liturgical_rite_var
				value = flag:kog_roman_rite
			}
			set_variable = {
				name = kog_liturgical_language_var

			}
			set_variable = {
				name = kog_mother_church_var
				value = flag:kog_rome

	save_scope_as = kog_country_scope
