kog_validate_global_journal_involvement_effect = {
	every_country = {
		limit = {
			OR = {
				AND = {
					is_involved_in_journal_entry = $type$
					$trigger$ = no
				}
				AND = {
					$trigger$ = yes
					NOT = {
						is_involved_in_journal_entry = $type$
					}
				}
			}
		}
		if = {
			limit = {
				is_involved_in_journal_entry = $type$
				$trigger$ = no
			}
			global_var:$type$ = {
				remove_involved_country =  prev
			}
		}
		else_if = {
			limit = {
				$trigger$ = yes
				NOT = {
					is_involved_in_journal_entry = $type$
				}
			}
			global_var:$type$ = {
				add_involved_country =  prev
			}
		}
	}
}

kog_update_journal_lists_effect = {
	scope:journal_entry = {
		save_scope_as = kog_global_faith_journal_scope
		clear_variable_list = kog_members_list
		clear_variable_list = kog_communion_list
		clear_variable_list = kog_schism_list
		clear_variable_list = kog_autocephalous_list
		every_scope_je_involved = {
			scope:kog_global_faith_journal_scope = {
				add_to_variable_list = {
					name = kog_ranking_list
					target = prev
				}
			}
		}
		every_scope_je_involved = {
			limit = {
				NOR = {
					kog_is_schismatic_trigger = yes
					kog_is_autocephalous_trigger = yes
				}
			}
			scope:kog_global_faith_journal_scope = {
				add_to_variable_list = {
					name = kog_communion_list
					target = prev
				}
			}
		}
		every_scope_je_involved = {
			limit = {
				kog_is_schismatic_trigger = yes
			}
			scope:kog_global_faith_journal_scope = {
				add_to_variable_list = {
					name = kog_schism_list
					target = prev
				}
			}
		}
		every_scope_je_involved = {
			limit = {
				kog_is_autocephalous_trigger = yes
			}
			scope:kog_global_faith_journal_scope = {
				add_to_variable_list = {
					name = kog_autocephalous_list
					target = prev
				}
			}
		}
		sort_variable_list = {
			name = kog_ranking_list
			order = prestige
		}
		sort_variable_list = {
			name = kog_communion_list
			order = prestige
		}
		sort_variable_list = {
			name = kog_schism_list
			order = prestige
		}
		sort_variable_list = {
			name = kog_autocephalous_list
			order = prestige
		}
	}
}

kog_update_devout_member_traits_effect = {
	ig:ig_devout = {
		leader = {
			if = {
				limit = {
					NOR = {
						has_trait = kog_patriarch
						has_trait = kog_cardinal
					}
				}
				add_trait = kog_patriarch
			}
		}
	}
	every_scope_character = {
		limit = {
			is_interest_group_type = ig_devout
			has_trait = kog_patriarch
			has_trait = kog_cardinal
		}
		remove_trait = kog_patriarch
	}
}

kog_update_national_piety_effect = {
	var:kog_local = {
		set_variable = {
			name = kog_national_piety_target_var
			value = parent.kog_national_piety_target_value
		}
		set_variable = {
			name = kog_national_piety_progress_var
			value = parent.kog_national_piety_progress_value
		}
		set_variable = {
			name = kog_national_piety_var
			value = {
				add = var:kog_national_piety_var
				add = var:kog_national_piety_progress_var
				round = yes
				max = 500
				min = -500
			}
		}
	}
	if = {
		limit = {
			has_modifier = modifier_kog_national_piety
		}
		remove_modifier = modifier_kog_national_piety
	}
	add_modifier = {
		name = modifier_kog_national_piety
		multiplier = kog_piety_multiplier_value
	}
}
