kog_change_seat_relations_effect = {
	# NOTE: The $seat$ parameter MUST point to one of the seat container scopes!
	custom_tooltip = {
		text = kog_change_seat_relations
		save_scope_as = kog_origin_scope
		set_local_variable = {
			name = kog_seat_scope
			value = $seat$
		}
		save_temporary_scope_value_as = {
			name = kog_seat_relations_change_value
			value = $value$
		}
		save_temporary_scope_value_as = {
			name = kog_seat_relations_math_value
			value = {
				value = kog_relations_start_value
				local_var:kog_seat_scope = {
					add = "variable_map(kog_relations_map|scope:kog_origin_scope)"
				}
				add = $value$
				max = kog_relations_max_value
				min = kog_relations_min_value
			}
		}
		local_var:kog_seat_scope = {
			com_update_value_in_variable_map = {
				name = kog_relations_map
				key = scope:kog_origin_scope
				value = scope:kog_seat_relations_math_value
			}
		}
	}
}

kog_update_seat_patriarchs_effect = {
	every_in_global_list = {
		variable = kog_global_seat_list
		save_scope_as = kog_episcopal_seat_scope
		# Step 1, Ensure there is a valid scope saved to kog_patriarch_var
		if = {
			limit = {
				NOT = {
					# If this variable is unset, heavens help us all
					# This is intentionally left unasserted because we NEED the error to point here if there's an issue
					exists = var:kog_patriarch_var
				}
			}
			# First, sweep the void for the Patriarch
			if = {
				limit = {
					any_character_in_void = {
						var:kog_seat_var ?= scope:kog_episcopal_seat_scope
					}
				}
				random_character_in_void = {
					limit = {
						var:kog_seat_var ?= scope:kog_episcopal_seat_scope
					}
					save_scope_as = kog_patriarch_scope
				}
			}
			# Regenerate the Patriarch if no character is found
			else = {
				# Special handling for the Papacy
				if = {
					limit = {
						has_tag = papacy
						c:PAP ?= {
							has_law_or_variant = law_type:law_papal_see
							container:kog_rome.var:kog_controller_var ?= c:PAP
						}
					}
					c:PAP = {
						ruler = {
							if = {
								limit = {
									NOT = {
										has_trait = kog_patriarch
									}
								}
								add_trait = kog_patriarch
							}
							set_variable = {
								name = kog_seat_var
								value = scope:kog_episcopal_seat_scope
							}
						}
					}
				}
				# Next try using the seat controller
				else_if = {
					limit = {
						exists = var:kog_controller_var
					}
					var:kog_controller_var = {
						create_character = {
							template = kog_patriarch_template
							religion = scope:kog_episcopal_seat_scope.var:kog_religion_var
							culture = culture
							on_created = {
								set_variable = {
									name = kog_seat_var
									value = scope:kog_episcopal_seat_scope
								}
								place_character_in_void = 180 # 15 year shelf life
							}
						}
					}
				}
				# Next try using the home province
				else_if = {
					limit = {
						var:kog_seat_province_var = {
							owner = {
								kog_content_scope_trigger = yes
							}
						}
					}
					var:kog_seat_province_var = {
						owner = {
							create_character = {
								template = kog_patriarch_template
								religion = religion
								culture = scope:kog_episcopal_seat_scope.var:kog_culture_var
								on_created = {
									set_variable = {
										name = kog_seat_var
										value = scope:kog_episcopal_seat_scope
									}
									place_character_in_void = 180 # 15 year shelf life
								}
							}
						}
					}
				}
				# Lastly fall back to default settings
				else = {
					var:kog_seat_province_var = {
						owner = {
							create_character = {
								template = kog_patriarch_template
								religion = scope:kog_episcopal_seat_scope.var:kog_religion_var
								culture = scope:kog_episcopal_seat_scope.var:kog_culture_var
								on_created = {
									set_variable = {
										name = kog_seat_var
										value = scope:kog_episcopal_seat_scope
									}
									place_character_in_void = 180 # 15 year shelf life
								}
							}
						}
					}
				}
				random_character_in_void = {
					limit = {
						var:kog_seat_var ?= scope:kog_episcopal_seat_scope
					}
					save_scope_as = kog_patriarch_scope
				}
				every_key_in_variable_map = {
					variable = kog_relations_map
					kog_change_seat_relations_effect = {
						seat = scope:kog_episcopal_seat_scope
						value = -25
					}
					post_notification = kog_new_patriarch_same_religion
				}
			}
			# Update the Patriarch Variable
			set_variable = {
				name = kog_patriarch_var
				value = scope:kog_patriarch_scope
			}
		}
	}
}
