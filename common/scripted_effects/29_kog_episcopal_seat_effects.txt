kog_change_seat_relations_effect = {
	custom_tooltip = {
		text = kog_change_$seat$_relations
		save_temporary_scope_value_as = {
			name = kog_change_seat_relations_effect
			value = $value$
		}
		var:je_kog_local = {
			set_variable = {
				name = kog_$seat$_relations_var
				value = {
					add = kog_opinion_start_value # Fallback in case the variable was unset
					if = {
						limit = {
							has_variable = kog_$seat$_relations_var
						}
						add = var:kog_$seat$_relations_var
					}
					add = $value$
					max = kog_opinion_max_value
					min = kog_opinion_min_value
				}
			}
		}
	}
}

kog_change_seat_relations_event_effect = {
	# This effect is meant to be used in conjunction with kog_event_seat_var
	if = {
		limit = {
			has_variable = kog_seat_event_var
		}
		if = {
			limit = {
				var:kog_seat_event_var = flag:kog_rome
			}
			kog_change_seat_relations_effect = {
				seat = rome
				value = $value$

kog_complete_seat_integration_effect = {
	custom_tooltip = {
		text = kog_usurped_$seat$
		save_scope_as = kog_usurper_scope
		je:je_kog_local = {
			remove_variable = kog_integrating_$seat$
			add_modifier = modifier_kog_controls_$seat$
		}
		scope:kog_$seat$_controller_scope ?= {
			change_relations = {
				country = root
				value = -30
			}
			je:je_kog_local = {
				remove_modifier = modifier_kog_controls_$seat$
			}
		}
		remove_from_global_variable_map = {
			name = kog_$seat$
			key = flag:kog_seat_controller
		}
		add_to_global_variable_map = {
			name = kog_$seat$
			key = flag:kog_seat_controller
			value = scope:kog_usurper_scope
		}
	}
	if = {
		limit = {
			has_modifier = modifier_kog_integrating_$seat$
		}
		remove_modifier =  modifier_kog_integrating_$seat$
	}
	every_country = {
		limit = {
			kog_content_scope_trigger = yes
		}
		post_notification = notification_kog_$seat$_usurped
	}
}

kog_assert_seat_control_effect = {
	je:je_kog_local = {
		remove_variable = kog_schism_var
		remove_modifier = modifier_kog_schismatic
		if = {
			limit = {
				NOT = {
					has_modifier = modifier_kog_autocephalous
				}
			}
			add_modifier = modifier_kog_autocephalous
		}
	}
	kog_complete_seat_integration_effect = {
		seat = $seat$
	}
	add_modifier = {
		name = modifier_kog_asserted_seat
		years = 5
		is_decaying = yes
	}
}
