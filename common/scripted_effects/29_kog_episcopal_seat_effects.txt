kog_change_seat_relations_effect = {
	# NOTE: The $seat$ parameter MUST point to one of the seat container scopes!
	custom_tooltip = {
		text = kog_change_seat_relations
		save_scope_as = kog_origin_scope
		set_local_variable = {
			name = kog_seat_scope
			value = $seat$
		}
		save_temporary_scope_value_as = {
			name = kog_seat_relations_change_value
			value = $value$
		}
		save_temporary_scope_value_as = {
			name = kog_seat_relations_math_value
			value = {
				value = kog_relations_start_value
				local_var:kog_seat_scope = {
					add = "variable_map(kog_relations_map|scope:kog_origin_scope)"
				}
				add = $value$
				max = kog_relations_max_value
				min = kog_relations_min_value
			}
		}
		local_var:kog_seat_scope = {
			com_update_value_in_variable_map = {
				name = kog_relations_map
				key = scope:kog_origin_scope
				value = scope:kog_seat_relations_math_value
			}
		}
	}
}

kog_complete_seat_integration_effect = {
	custom_tooltip = {
		text = kog_usurped_$seat$
		save_scope_as = kog_usurper_scope
		je:je_kog_local = {
			remove_variable = kog_integrating_$seat$
			add_modifier = modifier_kog_controls_$seat$
		}
		scope:kog_$seat$_controller_scope ?= {
			change_relations = {
				country = root
				value = -30
			}
			je:je_kog_local = {
				remove_modifier = modifier_kog_controls_$seat$
			}
		}
		remove_from_global_variable_map = {
			name = kog_$seat$
			key = flag:kog_seat_controller
		}
		add_to_global_variable_map = {
			name = kog_$seat$
			key = flag:kog_seat_controller
			value = scope:kog_usurper_scope
		}
	}
	if = {
		limit = {
			has_modifier = modifier_kog_integrating_$seat$
		}
		remove_modifier =  modifier_kog_integrating_$seat$
	}
	every_country = {
		limit = {
			kog_content_scope_trigger = yes
		}
		post_notification = kog_$seat$_usurped
	}
}

kog_assert_seat_control_effect = {
	je:je_kog_local = {
		remove_variable = kog_schism_var
		remove_modifier = modifier_kog_schismatic
		if = {
			limit = {
				NOT = {
					has_modifier = modifier_kog_autocephalous
				}
			}
			add_modifier = modifier_kog_autocephalous
		}
	}
	kog_complete_seat_integration_effect = {
		seat = $seat$
	}
	add_modifier = {
		name = modifier_kog_asserted_seat
		years = 5
		is_decaying = yes
	}
}

kog_check_seat_controllers_effect = {
	every_in_global_list = {
		variable = kog_global_seat_list
		if = {
			limit = {
				NOT = {
					exists = var:kog_seat_controller
				}
			}
			remove_variable = kog_seat_controller
		}
	}
	every_country = {
		limit = {
			kog_content_scope_trigger = yes
		}
		kog_update_episcopal_seat_count_effect = yes
	}
}

####---------- Patriarch Upkeep Effects ---------- ####
# These update effects use the following flow control:
# Attempt to locate the relevant variable map; regenerate if missing
# Check if the character saved to flag:kog_seat_patriarch exists; If the character exists, do nothing
# If the character does not exist, determine if the seat has a controller and then generate a character from the appropriate scope
# Clear the dead character scope from the map and insert the new scope

kog_update_patriarch_variables_effect = {
	kog_update_seat_patriarchs_effect = yes
	every_country = {
		limit = {
			kog_content_scope_trigger = yes
		}
		kog_update_metropolitan_bishop_patriarch_effect = yes
	}
}

kog_update_metropolitan_bishop_patriarch_effect = {
	# Local Metropolitan
	var:kog_local = {
		set_variable = {
			name = kog_metropolitan_var
			value = parent.ig:ig_devout.leader
		}
	}
}

kog_update_seat_patriarchs_effect = {
	every_in_global_list = {
		variable = kog_global_seat_list
		save_scope_as = kog_episcopal_seat_scope
		# Step 1, Ensure there is a valid scope saved to kog_seat_patriarch
		if = {
			limit = {
				NOT = {
					# If this variable is unset, heavens help us all
					exists = var:kog_seat_patriarch
				}
			}
			# First, sweep the void for the Patriarch
			if = {
				limit = {
					any_character_in_void = {
						var:kog_patriarch_var ?= scope:kog_episcopal_seat_scope
					}
				}
				random_character_in_void = {
					limit = {
						var:kog_patriarch_var ?= scope:kog_episcopal_seat_scope
					}
					save_scope_as = kog_patriarch_scope
				}
			}
			# Regenerate the Patriarch if no character is found
			else = {
				# Special handling for the Papacy
				if = {
					limit = {
						has_tag = papacy
						c:PAP ?= {
							has_law_or_variant = law_type:law_papal_see
							has_modifier = modifier_kog_controls_rome
						}
					}
					c:PAP = {
						ruler = {
							if = {
								limit = {
									NOT = {
										has_trait = kog_patriarch
									}
								}
								add_trait = kog_patriarch
							}
							set_variable = {
								name = kog_patriarch_var
								value = scope:kog_episcopal_seat_scope
							}
						}
					}
				}
				# Next try using the seat controller
				else_if = {
					limit = {
						exists = var:kog_seat_controller
					}
					var:kog_seat_controller = {
						create_character = {
							template = kog_patriarch_template
							religion = scope:kog_episcopal_seat_scope.var:kog_seat_religion
							culture = culture
							on_created = {
								set_variable = {
									name = kog_patriarch_var
									value = scope:kog_episcopal_seat_scope
								}
								place_character_in_void = 180 # 15 year shelf life
							}
						}
					}
				}
				# Next try using the home province
				else_if = {
					limit = {
						var:kog_seat_province = {
							owner = {
								kog_content_scope_trigger = yes
							}
						}
					}
					var:kog_seat_province = {
						owner = {
							create_character = {
								template = kog_patriarch_template
								religion = religion
								culture = scope:kog_episcopal_seat_scope.var:kog_seat_culture
								on_created = {
									set_variable = {
										name = kog_patriarch_var
										value = scope:kog_episcopal_seat_scope
									}
									place_character_in_void = 180 # 15 year shelf life
								}
							}
						}
					}
				}
				# Lastly fall back to default settings
				else = {
					var:kog_seat_province = {
						owner = {
							create_character = {
								template = kog_patriarch_template
								religion = scope:kog_episcopal_seat_scope.var:kog_seat_religion
								culture = scope:kog_episcopal_seat_scope.var:kog_seat_culture
								on_created = {
									set_variable = {
										name = kog_patriarch_var
										value = scope:kog_episcopal_seat_scope
									}
									place_character_in_void = 180 # 15 year shelf life
								}
							}
						}
					}
				}
				random_character_in_void = {
					limit = {
						var:kog_patriarch_var ?= scope:kog_episcopal_seat_scope
					}
					save_scope_as = kog_patriarch_scope
				}
				every_key_in_variable_map = {
					variable = kog_relations_map
					kog_change_seat_relations_effect = {
						seat = scope:kog_episcopal_seat_scope
						value = -25
					}
					post_notification = kog_new_patriarch
				}
			}
			# Update the Patriarch Variable
			set_variable = {
				name = kog_seat_patriarch
				value = scope:kog_seat_patriarch
			}
		}
	}
}
