kog_change_seat_relations_effect = {
	custom_tooltip = {
		text = kog_change_$seat$_relations
		save_temporary_scope_value_as = {
			name = kog_change_seat_relations_effect
			value = $value$
		}
		je:je_kog_local = {
			set_variable = {
				name = kog_$seat$_relations_var
				value = {
					add = kog_opinion_start_value # Fallback in case the variable was unset
					if = {
						limit = {
							has_variable = kog_$seat$_relations_var
						}
						add = var:kog_$seat$_relations_var
					}
					add = $value$
					max = kog_opinion_max_value
					min = kog_opinion_min_value
				}
			}
		}
	}
}

kog_change_seat_relations_event_effect = {
	# This effect is meant to be used in conjunction with kog_event_seat_var
	if = {
		limit = {
			has_variable = kog_seat_event_var
		}
		if = {
			limit = {
				var:kog_seat_event_var = flag:kog_rome
			}
			kog_change_seat_relations_effect = {
				seat = rome
				value = $value$
			}

			limit = {
				var:kog_seat_event_var = flag:kog_santiago

				seat = santiago
				value = $value$

		}
		else_if = {
			limit = {
				var:kog_seat_event_var = flag:kog_koln
			}

				seat = koln
