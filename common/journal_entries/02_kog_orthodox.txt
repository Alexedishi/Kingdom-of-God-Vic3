####---------- ORTHODOX ENCYCLICAL JOURNALS ----------#####
je_kog_encyclical_orthodox_easterners = {
	group = je_group_kog_encyclicals
	icon = "gfx/interface/icons/event_icons/event_orthodox.dds"
	weight = 100
	should_be_pinned_by_default_involved = yes
	
	should_be_involved = {
		kog_country_is_eastern_orthodox_trigger = yes
	}
	
	should_show_when_not_involved = {}
	
	immediate = {
		com_save_journal_to_global_variable = {
			name = je_kog_encyclical_orthodox_easterners
		}
	}
	
	timeout = 3650
	
	on_timeout = {
		if = {
			limit = {
				global_var:je_kog_encyclical_orthodox_easterners ?= flag:modifier_kog_rejected_epistle
			}
		}
		else_if = {
			limit = {
				global_var:je_kog_encyclical_orthodox_easterners ?= flag:modifier_kog_accepted_epistle
			}
		}
		remove_global_variable = je_kog_encyclical_orthodox_easterners
	}
	
	on_timeout_all_involved = {
		hidden_effect = {
			custom_tooltip = kog_nothing_happens # TODO put real effect here
		}
	}
	
	invalid = {
		NOT = {
			has_global_variable = je_kog_encyclical_orthodox_easterners
		}
	}
	
	on_invalid = {
		remove_global_variable = je_kog_encyclical_orthodox_easterners
	}
	
	on_weekly_pulse = {
		effect = {
			if = {
				limit = {
					global_var:je_kog_encyclical_orthodox_easterners ?= flag:modifier_kog_rejected_epistle
				}
				every_country = {
					limit = {
						is_involved_in_journal_entry = je_kog_encyclical_orthodox_easterners
					}
					if = {
						limit = {
							## TODO Rejection Compliance Trigger
						}
						set_variable = kog_encyclical_orthodox_easterners_compliance_var
						remove_variable = kog_encyclical_orthodox_easterners_non_compliance_var
					}
					if = {
						limit = {
							## TODO Rejection Non-Compliance Trigger
						}
						set_variable = kog_encyclical_orthodox_easterners_non_compliance_var
						remove_variable = kog_encyclical_orthodox_easterners_compliance_var
					}
				}
			}
			else_if = {
				limit = {
					global_var:je_kog_encyclical_orthodox_easterners ?= flag:modifier_kog_accepted_epistle
				}
				every_country = {
					limit = {
						is_involved_in_journal_entry = je_kog_encyclical_orthodox_easterners
					}
					if = {
						limit = {
							## TODO Acceptance Compliance Trigger
						}
						set_variable = kog_encyclical_orthodox_easterners_compliance_var
						remove_variable = kog_encyclical_orthodox_easterners_non_compliance_var
					}
					if = {
						limit = {
							## TODO Acceptance Non-Compliance Trigger
						}
						set_variable = kog_encyclical_orthodox_easterners_non_compliance_var
						remove_variable = kog_encyclical_orthodox_easterners_compliance_var
					}
				}
			}
		}
	}
	
	status_desc = {
		desc = je_kog_encyclical_orthodox_easterners_status
	}
}

je_kog_encyclical_orthodox_fraternity = {
	group = je_group_kog_encyclicals
	icon = "gfx/interface/icons/event_icons/event_orthodox.dds"
	weight = 100
	should_be_pinned_by_default_involved = yes
	
	should_be_involved = {
		kog_country_is_eastern_orthodox_trigger = yes
	}
	
	immediate = {
		scope:journal_entry = {
			save_scope_as = je_kog_encyclical_orthodox_fraternity
		}
		set_global_variable = {
			name = je_kog_encyclical_orthodox_fraternity
			value = scope:je_kog_encyclical_orthodox_fraternity
		}
	}
	
	timeout = 3650
	
	on_timeout = {
		remove_global_variable = je_kog_encyclical_orthodox_fraternity
	}
	
	on_timeout_all_involved = {
		if = {
			limit = {
				has_variable = kog_encyclical_orthodox_fraternity_compliance_var
			}
			add_modifier = {
				name = modifier_kog_rejected_epistle
				days = kog_10_years
			}
			kog_add_conservative_loyalists_effect = yes
			kog_add_reformist_radicals_effect = yes
		}
		else = {
			add_modifier = {
				name = modifier_kog_failed_encyclical
				days = kog_10_years
			}
			kog_add_reformist_loyalists_effect = yes
			kog_add_conservative_radicals_effect = yes
		}
	}
	
	on_weekly_pulse = {
		effect = {
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_kog_encyclical_orthodox_fraternity
				}
				if = {
					limit = {
						calc_true_if = {
							amount >= 3
							any_in_global_list = {
								variable = kog_global_seat_list
								var:kog_religion_var = rel:orthodox
								"variable_map(kog_opinions_map|root)" >= 75
							}
						}
					}
					set_variable = kog_encyclical_orthodox_fraternity_compliance_var
				}
				else = {
					remove_variable = kog_encyclical_orthodox_fraternity_compliance_var
				}
			}
		}
	}
}
