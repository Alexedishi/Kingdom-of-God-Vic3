		kog_country_is_eastern_orthodox_trigger = yes
	}
	
	should_show_when_not_involved = {
		kog_content_scope_trigger = yes
	}
	
	immediate = {
		scope:journal_entry = {
			com_hide_all_journal_elements_for_interorg_effect = yes
			com_set_international_organization_headquaters_effect = {
				state_region = s:STATE_EASTERN_THRACE
			}
			set_variable = kog_enable_encyclicals_panel
			set_variable = kog_enable_councils_panel
			save_scope_as = je_kog_orthodox
		}
		set_global_variable = {
			name = je_kog_orthodox
			value = scope:je_kog_orthodox
		}
	}
	
	complete = {
		always = no
	}
	
	on_monthly_pulse = {
		effect = {
			every_country = {
				limit = {
					OR = {
						AND = {
							is_involved_in_journal_entry = je_kog_orthodox
							kog_country_is_eastern_orthodox_trigger = no
						}
						AND = {
							kog_country_is_eastern_orthodox_trigger = yes
							NOT = {
								is_involved_in_journal_entry = je_kog_orthodox
							}
						}
					}
				}
				if = {
					limit = {
						is_involved_in_journal_entry = je_kog_orthodox
						kog_country_is_eastern_orthodox_trigger = no
					}
					global_var:je_kog_orthodox = {
						remove_involved_country =  prev
					}
				}
				else_if = {
					limit = {
						kog_country_is_eastern_orthodox_trigger = yes
						NOT = {
							is_involved_in_journal_entry = je_kog_orthodox
						}
					}
					global_var:je_kog_orthodox = {
						add_involved_country =  prev
					}
				}
			}
		}
	}
}

je_kog_encyclical_orthodox_easterners_1 = { # Conservative Outcome
	icon = "gfx/interface/icons/event_icons/event_orthodox.dds"
	
	group = je_group_kog_content
	
	should_be_pinned_by_default_involved = yes
	
	immediate = {
	}
	
	complete = {
		custom_tooltip = {
			text = je_kog_encyclical_orthodox_easterners_complete
			scope:journal_entry = {
			
			}
		}
	}
	
	on_complete = {
		add_modifier = {
			name = modifier_kog_rejected_epistle
			months = long_modifier_time
		}
		kog_add_conservative_loyalists_effect = yes
		kog_add_reformist_radicals_effect = yes
	}
	
	fail = {
		custom_tooltip = {
			text = je_kog_encyclical_orthodox_easterners_fail
			scope:journal_entry = {
			
			}
		}
	}
	
	on_fail = {
		add_modifier = {
			name = modifier_kog_failed_encyclical
			months = long_modifier_time
		}
		kog_add_reformist_loyalists_effect = yes
		kog_add_conservative_radicals_effect = yes
	}
	
	timeout = 3650
	
	on_timeout = {
		hidden_effect = {
			if = {
				limit = {
					scope:journal_entry = {
					
					}
				}
				add_modifier = {
					name = modifier_kog_rejected_epistle
					months = long_modifier_time
				}
				kog_add_conservative_loyalists_effect = yes
				kog_add_reformist_radicals_effect = yes
			}
			else = {
				add_modifier = {
					name = modifier_kog_failed_encyclical
					months = long_modifier_time
				}
				kog_add_reformist_loyalists_effect = yes
				kog_add_conservative_radicals_effect = yes
			}
		}
	}
	
	on_weekly_pulse = {
		effect = {
			if = {
				limit = {
				}
				set_variable = kog_orthodox_easterners_var
			}
			else = {
				remove_variable = kog_orthodox_easterners_var
			}
		}
	}
	
	weight = 500
	
}
