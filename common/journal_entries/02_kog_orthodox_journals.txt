####---------- ORTHODOX GLOBAL JOURNAL ----------####
je_kog_orthodox = {
	group = je_group_com_religious_organizations
	icon = "gfx/interface/icons/event_icons/event_orthodox.dds"
	weight = 0
	
	should_be_pinned_by_default_involved = yes
	
	is_shown_in_lobby = {
		country_has_state_religion = rel:orthodox
	}
	
	scripted_button = kog_display_widget_button
	
	is_shown_when_inactive = {
		always = yes
	}
	
	should_be_involved = {
		country_has_state_religion = rel:orthodox
	}
	
	possible = {
		hidden_trigger = {
			always = yes
		}
	}
	
	immediate = {
		kog_assign_dlc_icon_effect = yes
	}
	
	complete = {
		hidden_trigger = {
			always = no
		}
	}
	
	on_monthly_pulse = {
		effect = {
			every_country = {
				limit = {
					OR = {
						AND = {
							is_involved_in_journal_entry = je_kog_orthodox
							NOT = {
								country_has_state_religion = rel:orthodox
							}
						}
						AND = {
							country_has_state_religion = rel:orthodox
							NOT = {
								is_involved_in_journal_entry = je_kog_orthodox
							}
						}
					}
				}
				if = {
					limit = {
						is_involved_in_journal_entry = je_kog_orthodox
						NOT = {
							country_has_state_religion = rel:orthodox
						}
					}
					scope:journal_entry = {
						remove_involved_country =  prev
					}
				}
				else_if = {
					limit = {
						country_has_state_religion = rel:orthodox
						NOT = {
							is_involved_in_journal_entry = je_kog_orthodox
						}
					}
					scope:journal_entry = {
						add_involved_country =  prev
					}
				}
			}
		}
	}
}

####---------- ORTHODOX ENCYCLICAL ENTRIES ----------####
je_kog_encyclical_orthodox_easterners_1 = { # Conservative Outcome
	icon = "gfx/interface/icons/event_icons/event_orthodox.dds"

	group = je_group_kog_content

	should_be_pinned_by_default_uninvolved_or_context = yes

	scripted_progress_bar = kog_easterners_progress_bar

	immediate = {
	}

	complete = {
		custom_tooltip = {
			text = je_kog_encyclical_orthodox_easterners_complete
			scope:journal_entry = {
				scripted_bar_progress = {
					name = kog_easterners_progress_bar
					value <= 10
				}
			}
		}
		kog_encyclical_timeout_trigger = yes
	}

	on_complete = {
		add_modifier = {
			name = modifier_kog_rejected_epistle
			months = long_modifier_time
		}
		kog_add_conservative_loyalists_effect = yes
		kog_add_reformist_radicals_effect = yes
	}

	fail = {
		custom_tooltip = {
			text = je_kog_encyclical_orthodox_easterners_fail
			scope:journal_entry = {
				scripted_bar_progress = {
					name = kog_easterners_progress_bar
					value > 10
				}
			}
		}
		kog_encyclical_timeout_trigger = yes
	}

	on_fail = {
		add_modifier = {
			name = modifier_kog_failed_encyclical
			months = long_modifier_time
		}
		kog_add_reformist_loyalists_effect = yes
		kog_add_conservative_radicals_effect = yes
	}

	timeout = 3650

	on_timeout = {
		hidden_effect = {
			if = {
				limit = {
					scope:journal_entry = {
						scripted_bar_progress = {
							name = kog_easterners_progress_bar
							value <= 10
						}
					}
				}
				add_modifier = {
					name = modifier_kog_rejected_epistle
					months = long_modifier_time
				}
				kog_add_conservative_loyalists_effect = yes
				kog_add_reformist_radicals_effect = yes
			}
			else = {
				add_modifier = {
					name = modifier_kog_failed_encyclical
					months = long_modifier_time
				}
				kog_add_reformist_loyalists_effect = yes
				kog_add_conservative_radicals_effect = yes
			}
		}
	}

	on_weekly_pulse = {
		effect = {
			if = {
				limit = {
					scripted_bar_progress = {
						name = kog_easterners_progress_bar
						value <= 10
					}
				}
				set_variable = kog_orthodox_easterners_var
			}
			else = {
				remove_variable = kog_orthodox_easterners_var
			}
		}
	}

	weight = 500

}
