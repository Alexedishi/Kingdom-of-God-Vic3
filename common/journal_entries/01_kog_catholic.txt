	group = je_group_kog_encyclicals
	icon = "gfx/interface/icons/event_icons/event_catholic.dds"
	weight = 100
	
	activation_description = {
		always = yes
	}
	
	should_be_pinned_by_default_involved = yes
	can_revolution_inherit = yes
	transferable = no
	can_deactivate = no
	
	is_shown_in_lobby = {
		c:POL ?= this
	}
	
	should_be_involved = {
		OR = {
			c:PAP ?= this
			c:POL ?= this
		}
	}
	
	should_show_when_not_involved = {
		kog_country_is_roman_catholic_trigger = yes
	}
	
	immediate = {
		com_save_journal_to_global_variable = {
			name = je_kog_encyclical_cum_primum
		}
		global_var:je_kog_catholic = {
			add_to_variable_list = {
				name = kog_encyclicals_list
				target = scope:journal_entry
			}
		}
		scope:journal_entry = {
			set_variable = {
				name = kog_congress_poland_scope
				value = c:POL
			}
		}
	}
	
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
		color = gold_positive
		
		trigger = {
			c:POL ?= {
				is_subject = no
			}
		}
		effect = {
			global_var:je_kog_catholic = {
				remove_list_variable = {
					name = kog_encyclicals_list
					target = scope:journal_entry
				}
			}
		}
	}
	
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		conclusion_desc = je_standard_failure_conclusion_desc
		color = blue_negative
		
		trigger = {
			c:POL ?= {
				is_subject = yes
				is_at_war_with_overlord = yes
			}
		}
		effect = {
			global_var:je_kog_catholic = {
				remove_list_variable = {
					name = kog_encyclicals_list
					target = scope:journal_entry
				}
			}
		}
	}
	
	outcome = {
		conclusion_desc = je_standard_failure_conclusion_desc
		flags = { invalid_outcome hidden }
		
		trigger = {
			NOT = {
				exists = c:POL
			}
		}
		effect = {
			global_var:je_kog_catholic = {
				remove_list_variable = {
					name = kog_encyclicals_list
					target = scope:journal_entry
				}
			}
		}
	}
	
	status_desc = {
		desc = je_kog_encyclical_cum_primum_status
	}
}

je_kog_commissum_divinitus = {
	group = je_group_kog_encyclicals
	icon = "gfx/interface/icons/event_icons/event_catholic.dds"
	weight = 100
	
	activation_description = {
		always = yes
	}
	
	should_be_pinned_by_default_involved = yes
	can_revolution_inherit = yes
	transferable = no
	can_deactivate = no
	
	is_shown_in_lobby = {
		c:SWI ?= this
	}
	
	should_be_involved = {
		OR = {
			c:PAP ?= this
			c:SWI ?= this
		}
	}
	
	should_show_when_not_involved = {
		kog_country_is_roman_catholic_trigger = yes
	}
	
	immediate = {
		com_save_journal_to_global_variable = {
			name = je_kog_commissum_divinitus
		}
		global_var:je_kog_catholic = {
			add_to_variable_list = {
				name = kog_encyclicals_list
				target = scope:journal_entry
			}
		}
		scope:journal_entry = {
			set_variable = {
				name = kog_switzerland_scope
				value = c:SWI
			}
		}
		scope:journal_entry = {
			set_variable = {
				name = kog_swiss_catholic_radicalism_var_
				value = {
					c:SWI = {
						every_scope_pop = {
							limit = {
								has_pop_religion = catholic
							}
							add = total_size
							multiply = pop_radical_fraction
						}
						divide = {
							every_scope_pop = {
								limit = {
									has_pop_religion = catholic
								}
								add = total_size
							}
						}
					}
					round_to = 0.01
					max = 100
					min = 0
				}
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
        color = gold_positive

		trigger = {
			c:SWI ?= {
				has_law = law_type:law_total_separation
				has_law = law_type:law_monastic_secularization
				OR = {
					has_law = law_type:law_concordat_of_investment
					has_law = law_type:law_oath_of_supremacy
				}
				radical_fraction = {
					religion = rel:catholic
					value < 0.25
				}
				NOR = {
					has_revolution = yes
					has_any_secessionists_broken_out = yes
				}
			}
		}
		effect = { }
	}
	
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		conclusion_desc = je_standard_failure_conclusion_desc
		color = blue_negative
		
		trigger = {
			c:SWI ?= {
				any_political_movement = {
					is_political_movement_type = movement_religious_minority
					religion ?= rel:catholic
					OR = {
						is_revolutionary = yes
						is_secessionist = yes
					}
				}
			}
		}
		effect = { }
	}
	
	outcome = {
		conclusion_desc = je_standard_failure_conclusion_desc
		flags = { invalid_outcome hidden }
		
		trigger = {
			c:SWI ?= {
				has_law_or_variant = law_type:law_theocracy
			}
		}
		effect = { }
	}
	
	on_weekly_pulse = {
		effect = {
			scope:journal_entry = {
				set_variable = {
					name = kog_swiss_catholic_radicalism_var_
					value = {
