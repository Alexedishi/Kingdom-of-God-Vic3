####---------- CATHOLIC GLOBAL JOURNAL ----------#####
je_kog_catholic = {
	group = je_group_com_religious_organizations
	icon = "gfx/interface/icons/event_icons/event_catholic.dds"
	weight = 0
	
	should_be_pinned_by_default_involved = yes
	
	is_shown_in_lobby = {
		kog_country_is_roman_catholic_trigger = yes
	}
	
	should_be_involved = {
		kog_country_is_roman_catholic_trigger = yes
	}
	
	should_show_when_not_involved = {
		kog_content_scope_trigger = yes
	}
	
	immediate = {
		scope:journal_entry = {
			com_hide_all_journal_elements_for_interorg_effect = yes
			set_variable = kog_enable_encyclicals_panel
			set_variable = kog_enable_councils_panel
			save_scope_as = je_kog_catholic
		}
		set_global_variable = {
			name = je_kog_catholic
			value = scope:je_kog_catholic
		}
	}
	
	complete = {
		always = no
	}
	
	on_monthly_pulse = {
		effect = {
			every_country = {
				limit = {
					OR = {
						AND = {
							is_involved_in_journal_entry = je_kog_catholic
							kog_country_is_roman_catholic_trigger = no
						}
						AND = {
							kog_country_is_roman_catholic_trigger = yes
							NOT = {
								is_involved_in_journal_entry = je_kog_catholic
							}
						}
					}
				}
				if = {
					limit = {
						is_involved_in_journal_entry = je_kog_catholic
						kog_country_is_roman_catholic_trigger = no
					}
					global_var:je_kog_catholic = {
						remove_involved_country =  prev
					}
				}
				else_if = {
					limit = {
						kog_country_is_roman_catholic_trigger = yes
						NOT = {
							is_involved_in_journal_entry = je_kog_catholic
						}
					}
					global_var:je_kog_catholic = {
						add_involved_country =  prev
					}
				}
			}
		}
	}
}

####---------- CATHOLIC ENCYCLICAL JOURNALS ----------#####
je_kog_ecyclical_mirari_vos = {
	icon = "gfx/interface/icons/event_icons/event_catholic.dds"
	
	group = je_group_kog_content
	
	should_be_pinned_by_default = yes
	
	immediate = {}
	
	complete = {
		calc_true_if = {
			amount >= 3
			has_law = law_type:law_state_religion
			has_law = law_type:law_religious_schools
			has_law = law_type:law_charitable_health_system
			OR = {
				has_law = law_type:law_censorship
				has_law = law_type:law_outlawed_dissent
			}
		}
		kog_encyclical_timeout_trigger = yes
	}
	
	on_complete = {
		add_modifier = {
			name = modifier_kog_rebuked_lamennais
			months = long_modifier_time
		}
		kog_add_conservative_loyalists_effect = yes
		kog_add_reformist_radicals_effect = yes
	}
	
	fail = {
		calc_true_if = {
			amount < 3
			has_law = law_type:law_state_religion
			has_law = law_type:law_religious_schools
			has_law = law_type:law_charitable_health_system
			OR = {
				has_law = law_type:law_censorship
				has_law = law_type:law_outlawed_dissent
			}
		}
		kog_encyclical_timeout_trigger = yes
	}
	
	on_fail = {
		add_modifier = {
			name = modifier_kog_failed_encyclical
			months = long_modifier_time
		}
		kog_add_reformist_loyalists_effect = yes
		kog_add_conservative_radicals_effect = yes
	}
	
	timeout = 2404
	
	on_timeout = {
		hidden_effect = {
			if = {
				limit = {
					calc_true_if = {
						amount >= 3
						has_law = law_type:law_state_religion
						has_law = law_type:law_religious_schools
						has_law = law_type:law_charitable_health_system
						OR = {
							has_law = law_type:law_censorship
							has_law = law_type:law_outlawed_dissent
						}
					}
				}
				add_modifier = {
					name = modifier_kog_rebuked_lamennais
					months = long_modifier_time
				}
				kog_add_conservative_loyalists_effect = yes
				kog_add_reformist_radicals_effect = yes
			}
			else = {
				add_modifier = {
					name = modifier_kog_rebuked_lamennais
					months = long_modifier_time
				}
				kog_add_reformist_loyalists_effect = yes
				kog_add_conservative_radicals_effect = yes
			}
		}
	}
	
	on_weekly_pulse = {
		effect = {
			if = {
				limit = {
					calc_true_if = {
						amount >= 3
						has_law = law_type:law_state_religion
						has_law = law_type:law_religious_schools
						has_law = law_type:law_charitable_health_system
						OR = {
							has_law = law_type:law_censorship
							has_law = law_type:law_outlawed_dissent
						}
					}
				}
				set_variable = kog_catholic_mirari_vos_var
			}
			else = {
				remove_variable = kog_catholic_mirari_vos_var
			}
		}
	}
	
	weight = 500
	
}
