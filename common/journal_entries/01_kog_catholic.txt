			set_variable = kog_enable_encyclicals_panel
			set_variable = kog_enable_councils_panel
			save_scope_as = je_kog_catholic
		}
		set_global_variable = {
			name = je_kog_catholic
			value = scope:je_kog_catholic
		}
	}
	
	complete = {
		always = no
	}
	
	on_monthly_pulse = {
		effect = {
			every_country = {
				limit = {
					OR = {
						AND = {
							is_involved_in_journal_entry = je_kog_catholic
							kog_country_is_roman_catholic_trigger = no
						}
						AND = {
							kog_country_is_roman_catholic_trigger = yes
							NOT = {
								is_involved_in_journal_entry = je_kog_catholic
							}
						}
					}
				}
				if = {
					limit = {
						is_involved_in_journal_entry = je_kog_catholic
						kog_country_is_roman_catholic_trigger = no
					}
					global_var:je_kog_catholic = {
						remove_involved_country =  prev
					}
				}
				else_if = {
					limit = {
						kog_country_is_roman_catholic_trigger = yes
						NOT = {
							is_involved_in_journal_entry = je_kog_catholic
						}
					}
					global_var:je_kog_catholic = {
						add_involved_country =  prev
					}
				}
			}
		}
	}
}

####---------- CATHOLIC ENCYCLICAL JOURNALS ----------#####
je_kog_ecyclical_mirari_vos = {
	group = je_group_kog_encyclicals
	icon = "gfx/interface/icons/event_icons/event_catholic.dds"
	weight = 100
	
	should_be_pinned_by_default_involved = yes
	
	should_be_involved = {
		kog_country_is_roman_catholic_trigger = yes
	}
	
	should_show_when_not_involved = {
		kog_content_scope_trigger = yes
	}
	
	immediate = {
		scope:journal_entry = {
			save_scope_as = je_kog_ecyclical_mirari_vos
		}
		set_global_variable = {
			name = je_kog_ecyclical_mirari_vos
			value = scope:je_kog_ecyclical_mirari_vos
		}
	}
	
	timeout = 2404 # 6.5 years
	
	on_timeout = {
		custom_tooltip = je_kog_ecyclical_mirari_vos_success_effect
		custom_tooltip = je_kog_ecyclical_mirari_vos_failure_effect
	}
	
	on_weekly_pulse = {
		effect = {
			scope:journal_entry = {
				clear_variable_list = kog_ecyclical_compliant_list
				clear_variable_list = kog_ecyclical_non_compliant_list
			}
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_kog_ecyclical_mirari_vos
					calc_true_if = {
						amount >= 3
						has_law = law_type:law_state_religion
						has_law = law_type:law_religious_schools
						has_law = law_type:law_charitable_health_system
						OR = {
							has_law = law_type:law_censorship
							has_law = law_type:law_outlawed_dissent
						}
					}
				}
				global_var:je_kog_ecyclical_mirari_vos = {
					add_to_variable_list = {
						name = kog_ecyclical_compliant_list
						target = prev
					}
				}
			}
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_kog_ecyclical_mirari_vos
					calc_true_if = {
						amount < 3
						has_law = law_type:law_state_religion
						has_law = law_type:law_religious_schools
						has_law = law_type:law_charitable_health_system
						OR = {
							has_law = law_type:law_censorship
							has_law = law_type:law_outlawed_dissent
						}
					}
				}
				global_var:je_kog_ecyclical_mirari_vos = {
					add_to_variable_list = {
						name = kog_ecyclical_non_compliant_list
						target = prev
					}
				}
			}
			scope:journal_entry = {
				sort_variable_list = {
					name = kog_ecyclical_compliant_list
					order = prestige
				}
				sort_variable_list = {
					name = kog_ecyclical_non_compliant_list
					order = prestige
				}
			}
		}
	}
}
