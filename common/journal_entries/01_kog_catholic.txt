je_kog_ecyclical_mirari_vos = {
	group = je_group_kog_encyclicals
	icon = "gfx/interface/icons/event_icons/event_catholic.dds"
	weight = 100
	
	should_be_pinned_by_default_involved = yes
	
	should_be_involved = {
		kog_country_is_roman_catholic_trigger = yes
	}
	
	should_show_when_not_involved = {
		kog_content_scope_trigger = yes
	}
	
	immediate = {
		com_save_journal_to_global_variable = {
			name = je_kog_ecyclical_mirari_vos
		}
		global_var:je_kog_catholic = {
			add_to_variable_list = {
				name = kog_encyclicals_list
				target = scope:journal_entry
			}
		}
	}
	
	timeout = 2435 # Published in August 1832
	
	on_timeout = {
		custom_tooltip = {
			text = je_kog_ecyclical_mirari_vos_success_effect
			scope:journal_entry = {
				every_in_list = {
					variable = kog_ecyclical_compliant_list
				}
			}
		}
		custom_tooltip = {
			text = je_kog_ecyclical_mirari_vos_failure_effect
			every_in_list = {
				variable = kog_ecyclical_non_compliant_list
			}
		}
	}
	
	on_weekly_pulse = {
		effect = {
			scope:journal_entry = {
				clear_variable_list = kog_ecyclical_compliant_list
				clear_variable_list = kog_ecyclical_non_compliant_list
			}
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_kog_ecyclical_mirari_vos
					has_law = law_type:law_state_religion
					has_law = law_type:law_ecclesiastic_investiture
				}
				global_var:je_kog_ecyclical_mirari_vos = {
					add_to_variable_list = {
						name = kog_ecyclical_compliant_list
						target = prev
					}
				}
			}
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_kog_ecyclical_mirari_vos
					NOR = {
						has_law = law_type:law_state_religion
						has_law = law_type:law_ecclesiastic_investiture
					}
				}
				global_var:je_kog_ecyclical_mirari_vos = {
					add_to_variable_list = {
						name = kog_ecyclical_non_compliant_list
						target = prev
					}
				}
			}
			scope:journal_entry = {
				sort_variable_list = {
					name = kog_ecyclical_compliant_list
					order = prestige
				}
				sort_variable_list = {
					name = kog_ecyclical_non_compliant_list
					order = prestige
				}
			}
		}
	}
}

je_kog_commissum_divinitus = {
	group = je_group_kog_content
	icon = gfx/interface/icons/event_icons/event_catholic.dds
	weight = 500
	
	should_be_pinned_by_default_uninvolved_or_context = yes
	
	immediate = {
	}
	
	complete = {
		has_law = law_type:law_total_separation
		has_law = law_type:law_monastic_secularization
		OR = {
			has_law = law_type:law_concordat_of_investment
			has_law = law_type:law_oath_of_supremacy
		}
		radical_fraction = {
			religion = rel:catholic
			value < 0.25
		}
		NOR = {
			has_revolution = yes
			has_any_secessionists_broken_out = yes
		}
	}
	
	fail = {
		any_political_movement = {
			is_political_movement_type = movement_religious_minority
			religion = rel:catholic
			OR = {
				is_revolutionary = yes
				is_secessionist = yes
			}
		}
	}
	
	invalid = {
		has_law_or_variant = law_type:law_theocracy
	}
}
