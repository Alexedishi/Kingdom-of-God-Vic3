COUNTRIES = {
	every_country = {
		if = {
			limit = {
				kog_content_scope_trigger = yes
				NOT = {
					has_law_or_variant = law_type:law_theocracy
				}
			}
			activate_law = law_type:law_oath_of_supremacy
		}
		else_if = {
			limit = {
				kog_content_scope_trigger = yes
				has_law_or_variant = law_type:law_theocracy
			}
			activate_law = law_type:law_theocratic_investiture
		}
	}
}
