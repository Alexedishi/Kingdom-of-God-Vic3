# Focus Actions
kog_emphasize_unity_sgui = {
	scope = country
	
	is_shown = {
		always = yes
	}
	
	is_valid = {
		NOR = {
			has_modifier = modifier_kog_emphasize_unity
			custom_tooltip = {
				text = kog_focus_timeout_NOT
				has_variable = kog_focus_timeout_var
			}
		}
	}
	
	effect = {
		kog_remove_focus_modifiers_effect = yes
		add_modifier = modifier_kog_emphasize_unity
		custom_tooltip = {
			text = kog_focus_timeout_GET
			set_variable = {
				name = kog_focus_timeout_var
				years = 5
			}
		}
	}
}

kog_commission_icon_sgui = {
	scope = country
	
	is_shown = {
		always = yes
	}
	
	is_valid = {
		custom_tooltip = {
			text = kog_is_selecting_icon_NOT
			hidden_trigger = {
				NOT = {
					has_variable = kog_icon_var
				}
			}
		}
		trigger_if = {
			limit = {
				kog_has_icon_trigger = no
			}
			has_technology_researched = romanticism
			custom_tooltip = {
				text = kog_need_200_caesaropapism
				hidden_trigger = {
					kog_caesaropapism_capacity_value >= 200
				}
			}
		}
	}
	
	effect = {
		set_variable = kog_icon_var
		trigger_event = {
			id = kog.2230 # Commissioning an Icon
			popup = yes
		}
	}
}

kog_dissolve_monasteries_sgui = {
	scope = country
	
	is_shown = {
		NOT = {
			has_modifier = modifier_kog_dissolved_monasteries
		}
	}
	
	is_valid = {
		ig:ig_devout = {
			is_powerful = no
			is_in_government = no
		}
		NOT = {
			has_modifier = modifier_kog_restored_monasteries
		}
	}
	
	effect = {
		if = {
			limit = {
				NOT = {
					has_variable = kog_dissolved_monasteries
				}
			}
#			trigger_event = {
#				id = kog.030 # Dissolution of the Monasteries
#			}
		}
		add_modifier = modifier_kog_dissolved_monasteries
		set_variable = kog_dissolved_monasteries
	}
}

kog_reform_synod_sgui = {
	scope = country
	
	is_shown = {
		NOT = {
			has_modifier = modifier_kog_reformed_synod
		}
	}
	
	is_valid = {
		has_technology_researched = tech_bureaucracy
		ig:ig_devout = {
			is_in_government = no
			ig_approval > 0
		}
		NOR = {
			has_modifier = modifier_kog_reformed_synod
			has_modifier = modifier_kog_restored_synod
			has_modifier = modifier_kog_clung_to_tradition
		}
	}
	
	effect = {
		add_modifier = modifier_kog_reformed_synod
		add_radicals = {
			pop_type = clergymen
			value = small_radicals
		}
	}
}

kog_restore_synod_sgui = {
	scope = country
	
	is_shown = {
		has_modifier = modifier_kog_reformed_synod
	}
	
	is_valid = {
		has_modifier = modifier_kog_reformed_synod
		custom_tooltip = {
			text = kog_need_100_caesaropapism
			hidden_trigger = {
				kog_caesaropapism_capacity_value >= 100
			}
		}
	}
	
	effect = {
		remove_modifier = modifier_kog_reformed_synod
		add_modifier = {
			name = modifier_kog_restored_synod
			months = long_modifier_time
			is_decaying = yes
		}
		add_loyalists = {
			pop_type = clergymen
			value = small_radicals
		}
		remove_variable = kog_caesaropapism_questioned
	}
}
