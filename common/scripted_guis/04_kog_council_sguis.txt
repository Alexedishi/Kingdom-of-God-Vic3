## General SGUIs
kog_support_council_sgui = {
	scope = country
	
	is_valid = {
		NOR = {
			has_variable = kog_council_support_var
			has_variable = kog_council_timeout_var
			has_variable = kog_council_skipped
		}
	}
	effect = {
		set_variable = kog_council_support_var
		set_variable = {
			name = kog_council_timeout_var
			days = 30
		}
	}
}

kog_abstain_council_sgui = {
	scope = country
	
	is_valid = {
		OR = {
			has_variable = kog_council_support_var
			has_variable = kog_council_oppose_var
		}
		NOR = {
			has_variable = kog_council_timeout_var
			has_variable = kog_council_skipped
		}
	}
	effect = {
		remove_variable = kog_council_support_var
		remove_variable = kog_council_oppose_var
		set_variable = {
			name = kog_council_timeout_var
			days = 30
		}
	}
}

## Catholic SGUIs
kog_catholic_council_sgui = {
	scope = country
	
	is_shown = {
		has_global_variable = kog_catholic_convocation_var
	}
}

kog_catholic_convoke_sgui = {
	scope = country
	
	is_shown = {
		kog_is_autocephalous_trigger = yes
	}
	is_valid = {
		is_subject = no
		is_at_war = no
		kog_controls_any_seat_trigger = yes
		kog_has_1000_capacity_trigger = yes
#		kog_is_autocephalous_trigger = yes # TODO replace this
		# TODO add can_host_council trigger variant
		custom_tooltip = {
			text = kog_council_active_NOT
			hidden_trigger = {
				NOT = {
					has_global_variable = kog_catholic_matter_var
				}
			}
		}
	}
	effect = {
		#TODO Add re-factored event

		show_as_tooltip = {
			add_modifier = modifier_kog_hosting_council
		}
		hidden_effect = { # This is to present the player clean information in the UI, but has a time limit to avoid shenanigans
			add_modifier = {
				name = modifier_kog_hosting_council
				months = 12
			}
		}
		set_global_variable = kog_catholic_matter_var
		set_global_variable = kog_catholic_convocation_var # Error suppression- remove when event is implemented
	}
}
