####---------- Council Convocation SGUIs ----------####
kog_convoke_council_sgui = {
	scope = country
	
	is_shown = {
		kog_is_autocephalous_trigger = yes
	}
	is_valid = {
		kog_has_ongoing_council_trigger = no
		kog_capacity_trigger = {
			value = 1000
		}
		kog_controls_any_seat_trigger = yes
		is_subject = no
	}
	effect = {
		#TODO Add re-factored event
		add_modifier = modifier_kog_hosting_council
		global_var:je_kog_catholic = {
			set_variable = {
				name = kog_council_var
			}
		}
	}
}

####---------- Council Action SGUIs ----------####
kog_votes_support_sgui = {
	scope = country
	
	is_valid = {
		NOR = {
			has_variable = kog_council_support_var
			has_variable = kog_council_timeout_var
			has_variable = kog_council_skipped
		}
	}
	effect = {
		set_variable = kog_council_support_var
		set_variable = {
			name = kog_council_timeout_var
			days = 30
		}
	}
}

kog_votes_abstain_sgui = {
	scope = country
	
	is_valid = {
		OR = {
			has_variable = kog_council_support_var
			has_variable = kog_council_oppose_var
		}
		NOR = {
			has_variable = kog_council_timeout_var
			has_variable = kog_council_skipped
		}
	}
	effect = {
		remove_variable = kog_council_support_var
		remove_variable = kog_council_oppose_var
		set_variable = {
			name = kog_council_timeout_var
			days = 30
		}
	}
}

kog_votes_oppose_sgui = {
	scope = country
	
	is_valid = {
		NOR = {
			has_variable = kog_council_oppose_var
			has_variable = kog_council_timeout_var
			has_variable = kog_council_skipped
		}
	}
	effect = {
		set_variable = kog_council_oppose_var
		set_variable = {
			name = kog_council_timeout_var
			days = 30
		}
	}
}
