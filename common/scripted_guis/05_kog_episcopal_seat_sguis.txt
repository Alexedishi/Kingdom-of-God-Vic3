####---------- Episcopal Seat Action SGUIs ----------####
kog_start_sway_sgui = {
	scope = country
	saved_scopes = {
		kog_episcopal_seat_scope
	}
	
	is_shown = {
		scope:kog_episcopal_seat_scope = {
			"variable_map(kog_diplomacy_map|root)" != 1
		}
	}
	is_valid = {
		kog_capacity_trigger = {
			value = 100
		}
		custom_description = {
			text = kog_can_sway_seat_SGUI_trigger
			subject = this
			object = scope:kog_episcopal_seat_scope
			scope:kog_episcopal_seat_scope = {
				"variable_map(kog_relations_map|root)" < 50
			}
		}
	}
	effect = {
		show_as_tooltip = {
			add_modifier = modifier_kog_swaying_seats
		}
		scope:kog_episcopal_seat_scope = {
			com_update_value_in_variable_map = {
				name = kog_relations_map
				key = root
				value = 1
			}
		}
	}
}

kog_integrate_seat_sgui = {
	scope = country
	saved_scopes = {
		kog_episcopal_seat_scope
	}
	
	is_shown = {
		scope:kog_episcopal_seat_scope = {
			OR = {
				NOT = {
					has_variable = kog_controller_var
				}
				AND = {
					has_variable = kog_controller_var
					var:kog_controller_var != root
				}
			}
		}
	}
	is_valid = {
		kog_is_autocephalous_trigger = yes
		kog_capacity_trigger = {
			value = 200
		}
		custom_tooltip = {
			text = "kog_owns_seat_province_SGUI_trigger"
			scope:kog_episcopal_seat_scope = {
				var:kog_seat_province_var.owner = root
			}
		}
	}
	effect = {
		scope:kog_episcopal_seat_scope = {
			set_variable = {
				name = kog_integration_var
				value = 0
			}
			set_variable = {
				name = kog_integrator_var
				value = root
			}
		}
		show_as_tooltip = {
			add_modifier = modifier_kog_integrating_seat_placeholder
		}
		add_to_variable_list = {
			name = kog_seat_integration_list
			target = scope:kog_episcopal_seat_scope
		}
	}
}

kog_enforce_seat_sgui = {
	scope = country
	saved_scopes = {
		kog_episcopal_seat_scope
	}
	
	is_shown = {
		var:kog_local.var:kog_liturgical_rite_var != scope:kog_episcopal_seat_scope.var:kog_liturgical_rite_var
	}
	is_valid = {
		custom_tooltip = {
			text = kog_enforce_seat_sgui_trigger
			scope:kog_episcopal_seat_scope.var:kog_controller_var ?= root
		}
		kog_capacity_trigger = {
			value = 200
		}
		bureaucracy >= 50
	}
	effect = {
		custom_tooltip = {
			text = kog_enforce_seat_sgui_effect
			scope:kog_episcopal_seat_scope = {
				set_variable = {
					name = kog_enforcement_var
					value = 0
				}
				set_variable = {
					name = kog_enforcer_var
					value = root
				}
			}
		}
		show_as_tooltip = {
			add_modifier = modifier_kog_enforcing_seat_placeholder
		}
	}
}

kog_assert_seat_sgui = {
	scope = country
	saved_scopes = {
		kog_episcopal_seat_scope
	}
	
	is_shown = {
		kog_is_autocephalous_trigger = no
		var:kog_local.var:kog_mother_church_var = scope:kog_episcopal_seat_scope
	}
	is_valid = {
		custom_tooltip = {
			text = kog_assert_seat_sgui_trigger
			scope:kog_episcopal_seat_scope.var:kog_seat_province_var.owner = root
		}
	}
	effect = {
		custom_tooltip = {
			text = kog_assert_seat_sgui_effect
			scope:kog_episcopal_seat_scope = {
				set_variable = {
					name = kog_controller_var
					value = root
				}
				remove_variable = kog_integrator_var
				remove_variable = kog_integration_var
			}
		}
		add_modifier = {
			name = modifier_kog_asserted_seat
			days = kog_5_years
		}
	}
}

	}
}
