####---------- Episcopal Seat Opinion SGUIs ----------####
kog_opinion_friendly_sgui = {
	scope = country
	saved_scopes = {
		seat
	}
	
	is_shown = {
		scope:seat = {
			"variable_map(kog_opinion_map|root)" >= kog_opinion_threshold_friendly
		}
	}
}
