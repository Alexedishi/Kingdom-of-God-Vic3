####---------- Schism SGUIs ----------####
kog_declare_autocephaly_sgui = {
	scope = country

	is_shown = {
		kog_is_schismatic_trigger = no
		kog_is_autocephalous_trigger = no
	}
	is_valid = {
		is_subject = no
		kog_has_500_capacity_trigger = yes
		NOR = {
			has_law = law_type:law_theocracy
			has_modifier = modifier_kog_autocephaly_abandoned
		}
		trigger_if = {
			limit = {
				kog_country_is_roman_catholic_trigger = yes
			}
			always = no
		}
		trigger_else_if = {
			limit = {
				kog_country_is_eastern_orthodox_trigger = yes
			}
			trigger_if = {
				limit = {
					OR = {
						c:MOL ?= this
						c:WAL ?= this
					}
				}
				custom_tooltip = {
					text = kog_romania_united
					hidden_trigger = {
						c:ROM ?= this
					}
				}
			}
		}
	}
	effect = {
		add_modifier = {
			name = modifier_kog_schismatic
			months = 120
		}
		je:je_kog_local = {
			set_variable = {
				name = kog_schism_var
				value = 0
			}
		}
	}
}

kog_schism_progressbar_sgui = {
	scope = country

	is_shown = {
		je:je_kog_local = {
			has_variable = kog_schism_var
		}
	}
}
