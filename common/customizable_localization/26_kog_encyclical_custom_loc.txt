je_kog_encyclical_cum_primum_type = {
	type = country
	
	text = {
		localization_key = je_kog_encyclical_cum_primum_compliant
		trigger = {
			c:POL ?= {
				is_subject = yes
				is_at_war_with_overlord = no
				"c:POL.relations(c:POL.overlord)" >= relations_threshold:cordial
			}
		}
	}
	text = {
		localization_key = je_kog_encyclical_cum_primum_noncompliant
		trigger = {
			NOT = {
				c:POL ?= {
					is_subject = yes
					is_at_war_with_overlord = no
					"c:POL.relations(c:POL.overlord)" >= relations_threshold:cordial
				}
			}
		}
	}
}
