kog_national_piety_type = {
	type = country
	
	text = { 
		localization_key = kog_dutiful
		trigger = { 
			var:kog_local.var:kog_national_piety_var >= -100
			var:kog_local.var:kog_national_piety_var < 200
		}
	}
	text = {
		localization_key = kog_faithful
		trigger = {
			var:kog_local.var:kog_national_piety_var >= 200
			var:kog_local.var:kog_national_piety_var < 400
		}
	}
	text = {
		localization_key = kog_devoted
		trigger = {
			var:kog_local.var:kog_national_piety_var >= 400
		}
	}
	text = {
		localization_key = kog_sinful
		trigger = {
			var:kog_local.var:kog_national_piety_var < -100
		}
	}
}
