kog_council_stance_type = {
	type = country

	text = {
		trigger = {
			has_variable = kog_council_oppose_var
		}
		localization_key = kog_council_oppose
	}
	text = {
		trigger = {
			has_variable = kog_council_support_var
		}
		localization_key = kog_council_support
	}
	text = {
		trigger = {
			NOR = {
				has_variable = kog_council_oppose_var
				has_variable = kog_council_support_var
			}
		}
		localization_key = kog_council_abstain
	}
}

kog_catholic_matter_type = {
	type = country

	text = {
		trigger = {
			always = yes
		}
		localization_key = kog_no_effects
	}
}

kog_catholic_council_outcome_type = {
	type = country

	text = {
		trigger = {
			global_var:kog_catholic_support_var > global_var:kog_catholic_opposition_var
		}
		localization_key = kog_gui_support_winning
	}
	text = {
		trigger = {
			global_var:kog_catholic_support_var < global_var:kog_catholic_opposition_var
		}
		localization_key = kog_gui_oppose_winning
	}
	text = {
		trigger = {
			global_var:kog_catholic_support_var = global_var:kog_catholic_opposition_var
		}
		localization_key = kog_gui_noone_winning
	}
}
