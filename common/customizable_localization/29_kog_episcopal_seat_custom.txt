####--------- EPISCOPAL SEAT CONTROLLER ----------##
kog_controller_type = {
	type = character
	
	text = {
		trigger = {
			has_variable = kog_controller_var
			exists = var:kog_controller_var
		}
		localization_key = kog_active_owner
	}
	text = {
		trigger = {
			OR = {
				NOT = {
					has_variable = kog_controller_var
				}
				AND = {
					has_variable = kog_controller_var
					NOT = {
						exists = var:kog_controller_var
					}
				}
			}
		}
		localization_key = kog_no_owner
	}
}

kog_rome_opinion_type = {
	type = country
	
	text = {
		trigger = {
			je:je_kog_local.var:kog_rome_opinion_var >= 80
			global_var:kog_global.var:kog_rome.var:kog_patriarch_var.religion = root.religion
		}
		localization_key = kog_endorsing
	}
	text = {
		trigger = {
			je:je_kog_local.var:kog_rome_opinion_var >= 80
			global_var:kog_global.var:kog_rome.var:kog_patriarch_var.religion != root.religion
		}
		localization_key = kog_friendly
	}
	text = {
		trigger = {
			je:je_kog_local.var:kog_rome_opinion_var < 80
			je:je_kog_local.var:kog_rome_opinion_var >= 50
		}
		localization_key = kog_amicable
	}
	text = {
		trigger = {
			je:je_kog_local.var:kog_rome_opinion_var < 50
			je:je_kog_local.var:kog_rome_opinion_var >= 20
		}
		localization_key = kog_cordial
	}
	text = {
		trigger = {
			je:je_kog_local.var:kog_rome_opinion_var < 20
			je:je_kog_local.var:kog_rome_opinion_var > -20
		}
		localization_key = kog_neutral
	}
	text = {
		trigger = {
			je:je_kog_local.var:kog_rome_opinion_var <= -20
			je:je_kog_local.var:kog_rome_opinion_var > -50
		}
		localization_key = kog_poor
	}
	text = {
		trigger = {
			je:je_kog_local.var:kog_rome_opinion_var <= -50
			je:je_kog_local.var:kog_rome_opinion_var > -80
		}
		localization_key = kog_cold
	}
	text = {
		trigger = {
			je:je_kog_local.var:kog_rome_opinion_var <= -80
			global_var:kog_global.var:kog_rome.var:kog_patriarch_var.religion != root.religion
		}
		localization_key = kog_hostile
	}
	text = {
		trigger = {
			je:je_kog_local.var:kog_rome_opinion_var <= -80
			global_var:kog_global.var:kog_rome.var:kog_patriarch_var.religion = root.religion
		}
		localization_key = kog_condemning
	}
}
