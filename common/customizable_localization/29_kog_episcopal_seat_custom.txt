kog_rome_opinion_type = {
	type = country
	
	text = {
		trigger = {
			je:je_kog_local.var:kog_rome_opinion_var >= kog_opinion_threshold_friendly
			has_modifier = modifier_kog_endorsed_by_rome
		}
		localization_key = kog_friendly
	}
	text = {
		trigger = {
			je:je_kog_local.var:kog_rome_opinion_var >= kog_opinion_threshold_friendly
			NOT = {
				has_modifier = modifier_kog_endorsed_by_rome
			}
		}
		localization_key = kog_friendly
	}
	text = {
		trigger = {
			je:je_kog_local.var:kog_rome_opinion_var < kog_opinion_threshold_friendly
			je:je_kog_local.var:kog_rome_opinion_var >= kog_opinion_threshold_amicable
		}
		localization_key = kog_amicable
	}
	text = {
		trigger = {
			je:je_kog_local.var:kog_rome_opinion_var < kog_opinion_threshold_amicable
			je:je_kog_local.var:kog_rome_opinion_var >= kog_opinion_threshold_cordial
		}
		localization_key = kog_cordial
	}
	text = {
		trigger = {
			je:je_kog_local.var:kog_rome_opinion_var < kog_opinion_threshold_cordial
			je:je_kog_local.var:kog_rome_opinion_var > kog_opinion_threshold_poor
			
		}
		localization_key = kog_neutral
	}
	text = {
		trigger = {
			je:je_kog_local.var:kog_rome_opinion_var <= kog_opinion_threshold_poor
			je:je_kog_local.var:kog_rome_opinion_var > kog_opinion_threshold_cold
		}
		localization_key = kog_poor
	}
	text = {
		trigger = {
			je:je_kog_local.var:kog_rome_opinion_var <= kog_opinion_threshold_cold
			je:je_kog_local.var:kog_rome_opinion_var > kog_opinion_threshold_hostile
		}
		localization_key = kog_cold
	}
	text = {
		trigger = {
			je:je_kog_local.var:kog_rome_opinion_var <= kog_opinion_threshold_hostile
			NOT = {
				has_modifier = modifier_kog_condemned_by_rome
			}
		}
		localization_key = kog_hostile
	}
	text = {
		trigger = {
			je:je_kog_local.var:kog_rome_opinion_var <= kog_opinion_threshold_hostile
			has_modifier = modifier_kog_condemned_by_rome
		}
		localization_key = kog_condemning
	}
}
