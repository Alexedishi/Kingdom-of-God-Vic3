####---------- General Council Values ----------####
kog_votes_value = {
	value = 0
	
	# Catholic State Logic
	if = {
		limit = {
			kog_country_is_roman_catholic_trigger = yes
		}
		every_scope_state = {
			limit = {
				is_in_geographic_region = geographic_region_italy_old
			}
			add = {
				value = 4
				desc = kog_value_from_italian_states
			}
		}
		every_scope_state = {
			limit = {
				OR = {
					is_in_geographic_region = geographic_region_england_old
					is_in_geographic_region = geographic_region_iberia_old
					is_in_geographic_region = geographic_region_france_old
					is_in_geographic_region = geographic_region_occitania_old
					is_in_geographic_region = geographic_region_rhine_old
					is_in_geographic_region = geographic_region_north_germany_old
					is_in_geographic_region = geographic_region_south_germany_old
					is_in_geographic_region = geographic_region_poland_old
				}
			}
			add = {
				value = 2
				desc = kog_value_from_historically_catholic_states
			}
		}
		every_scope_state = {
			limit = {
				NOR = {
					is_in_geographic_region = geographic_region_italy_old
					is_in_geographic_region = geographic_region_england_old
					is_in_geographic_region = geographic_region_iberia_old
					is_in_geographic_region = geographic_region_france_old
					is_in_geographic_region = geographic_region_occitania_old
					is_in_geographic_region = geographic_region_rhine_old
					is_in_geographic_region = geographic_region_north_germany_old
					is_in_geographic_region = geographic_region_south_germany_old
					is_in_geographic_region = geographic_region_poland_old
				}
			}
			add = {
				value = 1
				desc = kog_value_from_other_states
			}
		}
	}
	
	# Orthodox State Logic
	if = {
		limit = {
			kog_country_is_eastern_orthodox_trigger = yes
		}
		every_scope_state = {
			limit = {
				OR = {
					is_in_geographic_region = geographic_region_balkans_old
					is_in_geographic_region = geographic_region_anatolia_old
				}
			}
			add = {
				value = 4
				desc = kog_value_from_byzantine_states
			}
		}
		every_scope_state = {
			limit = {
				OR = {
					is_in_geographic_region = geographic_region_danubia_old
					is_in_geographic_region = geographic_region_russia_old
					is_in_geographic_region = geographic_region_dnieper_old
					is_in_geographic_region = geographic_region_belarus_old
					is_in_geographic_region = geographic_region_caucasus_old
					is_in_geographic_region = geographic_region_arabic_old
					is_in_geographic_region = geographic_region_nile_basin
				}
			}
			add = {
				desc = kog_value_from_historically_orthodox_states
				value = 2
			}
		}
		every_scope_state = {
			limit = {
				NOR = {
					is_in_geographic_region = geographic_region_balkans_old
					is_in_geographic_region = geographic_region_anatolia_old
					is_in_geographic_region = geographic_region_danubia_old
					is_in_geographic_region = geographic_region_russia_old
					is_in_geographic_region = geographic_region_dnieper_old
					is_in_geographic_region = geographic_region_belarus_old
					is_in_geographic_region = geographic_region_caucasus_old
					is_in_geographic_region = geographic_region_arabic_old
					is_in_geographic_region = geographic_region_nile_basin
				}
			}
			add = {
				desc = kog_value_from_other_states
				value = 1
			}
		}
	}
	
	if = {
		limit = {
			c:PAP ?= this
			has_law = law_type:law_theocracy
		}
		add = {
			desc = kog_value_from_being_PAP
			value = 100
		}
	}
	else_if = {
		limit = {
			has_law = law_type:law_theocracy
		}
		add = {
			desc = kog_value_from_theocracy
			value = 50
		}
	}
	
	if = {
		limit = {
			kog_is_autocephalous_trigger = yes
		}
		add = {
			desc = kog_value_from_autocephaly
			value = 5
		}
	}
	
	add = {
		desc = kog_value_from_episcopal_seats
		value = var:kog_local.var:kog_episcopal_seats_var
		multiply = 10
	}
	
	if = {
		limit = {
			kog_is_schismatic_trigger = yes
		}
		multiply = 0.25
	}
	
	round_to = 1
	min = 0
}
