kog_capacity_value = {
	value = 0
	if = {
		limit = {
			kog_is_roman_catholic_trigger = yes
		}
		add = kog_papacy_capacity_value
	}
	else_if = {
		limit = {
			kog_is_eastern_christian_trigger = yes
		}
		add = kog_caesaropapism_capacity_value
	}
	else_if = {
		limit = {
			kog_is_oriental_orthodox_trigger = yes
		}
		add = kog_synodal_capacity_value
	}
	else_if = {
		limit = {
			kog_is_elysian_orthodox_trigger = yes
		}
		add = kog_synkletos_capacity_value
	}
	else_if = {
		limit = {
			kog_is_protestant_trigger = yes
		}
		add = kog_power_capacity_value
	}
}

kog_decree_expense_value = {
	value = 0
	every_scope_state = {
		limit = {
			always = no
			#has_decree = decree_kog_consecrate_metropolitan
		}
		subtract = {
			value = 100
			desc = kog_value_from_metropolitan_decree
		}
	}
	every_scope_state = {
		limit = {
			always = no
			#has_decree = decree_kog_encourage_almsgiving
		}
		subtract = {
			value = 100
			desc = kog_value_from_almsgiving_decree
		}
	}
}
