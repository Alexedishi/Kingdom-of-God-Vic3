	value = 0

kog_piety_target_value = {
	value = 0
}

kog_piety_progress_value = {
	value = 0
	if = {
		limit = {
			var:kog_local.var:kog_national_piety_var > kog_piety_target_value
		}
		add = kog_piety_target_value
		subtract = var:kog_local.var:kog_national_piety_var
	}
	if = {
		limit = {
			var:kog_local.var:kog_national_piety_var < kog_piety_target_value
		}
		add = kog_piety_target_value
		subtract = var:kog_local.var:kog_national_piety_var
	}
	round_to = 0.1
	max = 0.5
	min = -0.5
}

kog_cardinal_college_value = {
	value = 0
	every_scope_state = {
		limit = {
			region = sr:region_italy
		}
		add = {
			value = 4
			desc = kog_value_from_italian_states
		}
	}
	every_scope_state = {
		limit = {
			OR = {
				region = sr:region_england
				region = sr:region_iberia
				region = sr:region_france
				region = sr:region_occitania
				region = sr:region_rhine
				region = sr:region_north_germany
				region = sr:region_south_germany
				region = sr:region_poland
			}
		}
		add = {
			value = 2
			desc = kog_value_from_european_states
		}
	}
	every_scope_state = {
		limit = {
			NOR = {
				region = sr:region_italy
				region = sr:region_england
				region = sr:region_iberia
				region = sr:region_france
				region = sr:region_occitania
				region = sr:region_rhine
				region = sr:region_north_germany
				region = sr:region_south_germany
				region = sr:region_poland
			}
		}
		add = {
			value = 1
			desc = kog_value_from_other_states
		}
	}
}

kog_catholic_support_value = {
	value = 0
	every_country = {
		limit = {
			kog_country_is_roman_catholic_trigger = yes
			has_variable = kog_council_support_var
		}
		add = kog_cardinal_college_value
	}
}

kog_catholic_opposition_value = {
	value = 0
	every_country = {
		limit = {
			kog_country_is_roman_catholic_trigger = yes
			has_variable = kog_council_oppose_var
		}
		subtract = kog_cardinal_college_value
	}
}

kog_catholic_council_value = {
	value = 0
	add = {
		value = kog_catholic_support_value
		desc = kog_value_from_catholic_support
	}
	subtract = {
		value = kog_catholic_opposition_value
		desc = kog_value_from_catholic_opposition
	}
}

kog_character_gravitas_value = {
	value = 0
	if = {
		limit = {
			is_in_void = no
		}
		add = {
			value = owner.prestige
			multiply = 0.1
			round = 0.1
			desc = kog_value_from_home_prestige
		}
	}
	if = {
		limit = {
			OR = {
				has_culture = cu:north_italian
				has_culture = cu:south_italian
			}
		}
		add = {
			value = 200
			desc = kog_value_from_italian_culture
		}
	}
	add = {
		value = kog_character_piety_value
		desc = kog_value_from_piety
	}
	add = {
		value = age
		multiply = 2
		desc = kog_value_from_age
	}
	if = {
		limit = {
			has_trait = trait_child
		}
		add = {
			value = -500
			desc = kog_value_from_being_underage
		}
	}
	round = 0.01
	min = 0
}
