kog_synkletos_capacity_value = {
	value = 0

	# SOURCES
	add = {
		value = 100
		desc = kog_value_from_base
	}

	## EXPENSES
	# Seat Diplomacy Check
	subtract = {
		value = kog_seat_diplomacy_value
		desc = kog_value_from_diplomacy
	}
}

kog_elysian_synkletos_value = {
	value = 0
	if = {
		limit = {
			has_variable = kog_is_autocephalous
		}
		add = {
			value = 1
			desc = kog_value_from_having_autocephaly
		}
	}
	if = {
		limit = {
			has_variable = kog_controls_elysium
		}
		add = {
			value = 1
			desc = kog_value_from_controls_elysium
		}
	}
	if = {
		limit = {
			has_variable = kog_controls_mixtec
		}
		add = {
			value = 1
			desc = kog_value_from_controls_mixtec
		}
	}
	if = {
		limit = {
			has_variable = kog_controls_cahokia
		}
		add = {
			value = 1
			desc = kog_value_from_controls_cahokia
		}
	}
}

kog_elysian_support_value = {
	value = 0
	every_country = {
		limit = {
			kog_is_eastern_orthodox_trigger = yes
			has_variable = kog_council_support_var
		}
		add = kog_elysian_synkletos_value
	}
}

kog_elysian_opposition_value = {
	value = 0
	every_country = {
		limit = {
			kog_is_eastern_orthodox_trigger = yes
			has_variable = kog_council_oppose_var
		}
		subtract = kog_elysian_synkletos_value
	}
}

kog_elysian_council_value = {
	value = 0
	add = {
		value = kog_elysian_support_value
		desc = kog_value_from_elysian_support
	}
	subtract = {
		value = kog_elysian_opposition_value
		desc = kog_value_from_elysian_opposition
	}
}
