kog_capacity_value = {
	value = 0
	add = modifier:country_faith_capacity_add
	multiply = {
		add = 1
		add = modifier:country_faith_capacity_mult
	}
}

kog_capacity_multiplier_value = {
	value = 0
	add = kog_capacity_value
	divide = 1000
	round_to = 0.1
}

kog_national_piety_target_value = {
	value = 0
	
	# Church and State Country Law; this is the largest single contributor and serves as a base
	add = {
		desc = kog_value_from_active_religion_law
		value = {
			if = {
				limit = {
					has_law_or_variant = law_type:law_state_religion
				}
				add = 200
			}
			else_if = {
				limit = {
					has_law_or_variant = law_type:law_freedom_of_conscience
				}
				add = 100
			}
			else_if = {
				limit = {
					has_law_or_variant = law_type:law_total_separation
				}
				add = -100
			}
			else_if = {
				limit = {
					has_law_or_variant = law_type:law_state_atheism
				}
				add = -1000
			}
		}
	}
	
	# Canon Law Compliance
	add = {
		desc = kog_value_from_canon_law_compliance
		value = {
			var:kog_local ?= {
				if = {
					limit = {
						has_variable = kog_canon_law_compliance_var
					}
					add = var:kog_canon_law_compliance_var
				}
			}
		}
	}
	
	round_to = 0.01
	max = 500
	min = -500
}

kog_national_piety_progress_value = {
	value = 0
	add = var:kog_local.var:kog_national_piety_target_var
	subtract = var:kog_local.var:kog_national_piety_var
	round_to = 0.1
	max = 0.5
	min = -0.5
}

kog_piety_multiplier_value = {
	value = 0
	add = var:kog_local.var:kog_national_piety_var
	divide = 500
	round_to = 0.01
	max = 1
	min = -1
}
