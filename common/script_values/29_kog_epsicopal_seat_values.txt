kog_rome_monthly_relations_change_value = {
	value = 0
	if = {
		limit = { 
			has_modifier = modifier_kog_swaying_rome
		}
		add = modifier:country_monthly_seat_relations_progress_add
	}
	else_if = { 
		limit = { 
			has_modifier = modifier_kog_antagonizing_rome
		}
		subtract = modifier:country_monthly_seat_relations_progress_add
	}
	else = { 
		if = {
			limit = {
				je:je_kog_local.var:kog_rome_relations_var > kog_seat_relations_decay_target_value
			}
			subtract = modifier:country_monthly_seat_relations_decay_add
		}
		else_if = {
			limit = {
				je:je_kog_local.var:kog_rome_relations_var < kog_seat_relations_decay_target_value
			}
			add = modifier:country_monthly_seat_relations_decay_add
		}
	}
	multiply = {
		value = 1
		add = modifier:country_monthly_seat_relations_progress_mult
		min = 0.01
	}
	round_to = 0.01
}
