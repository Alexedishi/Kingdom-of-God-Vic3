kog_episcopal_seat_monthly_change_value = {
	value = 0
	add = "variable_map(kog_relations_map|scope:kog_root_scope)"
	if = {
		limit = {
			"variable_map(kog_diplomacy_map|scope:kog_root_scope)" = flag:kog_improving
		}
		add = scope:kog_root_scope.modifier:country_monthly_seat_relations_progress_add
	}
	else_if = {
		limit = {
			"variable_map(kog_diplomacy_map|scope:kog_root_scope)" = flag:kog_antagonizing
		}
		subtract = scope:kog_root_scope.modifier:country_monthly_seat_relations_progress_add
	}
	else_if = {
		limit = {
			"variable_map(kog_relations_map|scope:kog_root_scope)" > scope:kog_root_scope.modifier:country_seat_relations_base_add
		}
		subtract = scope:kog_root_scope.modifier:country_monthly_seat_relations_decay_add
	}
	else_if = {
		limit = {
			"variable_map(kog_relations_map|scope:kog_root_scope)" < scope:kog_root_scope.modifier:country_seat_relations_base_add
		}
		add = scope:kog_root_scope.modifier:country_monthly_seat_relations_decay_add
	}
	multiply = {
		add = 1
		add = modifier:country_monthly_seat_relations_progress_mult
		min = 0.01
	}
	max = 50
	min = -50
	round_to = 0.01
}
