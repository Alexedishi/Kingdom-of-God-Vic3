kog_gui_establish_schools = {
	name = "kog_gui_establish_schools"
	desc = "kog_gui_establish_schools_desc"
	
	visible = {
		has_institution = institution_schools
	}
	selected = {
		has_modifier = modifier_kog_sunday_schools
	}
	possible = {
		trigger_if = {
			limit = {
				has_modifier = modifier_kog_sunday_schools
			}
			NOT = {
				has_law = law_type:law_state_religion
			}
			ig:ig_devout = {
				is_in_government = no
			}
		}
		trigger_else = {
			NOR = {
				has_law = law_type:law_total_separation
				has_law = law_type:law_state_atheism
			}
		}
		custom_tooltip = {
			text = kog_placeholder_text
			NOT = {
				has_variable = kog_sunday_schools_cooldown_var
			}
		}
	}
	effect = {
		if = {
			limit = {
				has_modifier = modifier_kog_sunday_schools
			}
			remove_modifier = modifier_kog_sunday_schools
			add_modifier = {
				name = modifier_kog_disbanded_sunday_schools
				months = long_modifier_time
			}
		}
		else = {
			add_modifier = modifier_kog_sunday_schools
		}
		set_variable = {
			name = kog_sunday_schools_cooldown_var
			months = long_modifier_time
		}
	}
	
	ai_chance = {
		value = 0
	}
}
