kog_gui_reform_synod = {
	name = "kog_gui_reform_synod"
	desc = "kog_gui_reform_synod_desc"
	
	visible = {
		NOR = {
			has_law = law_type:law_state_church
			has_modifier = modifier_kog_restored_synod
		}
	}
	possible = {
		has_technology_researched = tech_bureaucracy
		ig:ig_devout = {
			is_in_government = no
			ig_approval > 0
		}
		NOR = {
			has_modifier = modifier_kog_reformed_synod
			has_modifier = modifier_kog_restored_synod
			has_modifier = modifier_kog_clung_to_tradition
		}
	}
	effect = {
		activate_law = law_type:law_state_church
		add_modifier = {
			name = modifier_kog_reformed_synod
			years = 10
			is_decaying = yes
		}
		add_radicals = {
			pop_type = clergymen
			value = large_radicals
		}
	}
	
	ai_chance = {
		value = 0
	}
}

kog_gui_restore_synod = {
	name = "kog_gui_restore_synod"
	desc = "kog_gui_restore_synod_desc"
	
	visible = {
		has_law = law_type:law_state_church
		NOR = {
			has_modifier = modifier_kog_reformed_synod
		}
	}
	possible = {
		ig:ig_devout = {
			is_insurrectionary = no
			OR = {
				is_marginal = no
				is_in_government = yes
			}
		}
		kog_capacity_trigger = {
			value = 500
		}
	}
	effect = {
		activate_law = law_type:law_state_religion
		add_modifier = {
			name = modifier_kog_restored_synod
			years = 5
			is_decaying = yes
		}
		add_loyalists = {
			pop_type = clergymen
			value = medium_radicals
		}
		remove_variable = kog_constantinianism_questioned
	}
	
	ai_chance = {
		value = 0
	}
}

kog_gui_establish_schools = {
	name = "kog_gui_establish_schools"
	desc = "kog_gui_establish_schools_desc"
	
	visible = {
		NOT = {
			active_law:lawgroup_education_system = {
				has_amendment = amendment_type:amendment_kog_sunday_schools
			}
		}
	}
	possible = {
		has_law_or_variant = law_type:law_state_religion
		has_institution = institution_schools
		NOT = {
			has_modifier = modifier_kog_disbanded_sunday_schools
		}
	}
	effect = {
		active_law:lawgroup_education_system = {
			add_amendment = {
				type = amendment_kog_sunday_schools
				sponsor = ig:ig_devout
			}
		}
		add_modifier = {
			name = modifier_kog_established_sunday_schools
			years = 5
			is_decaying = yes
		}
	}
	
	ai_chance = {
		value = 0
	}
}

kog_gui_disband_schools = {
	name = kog_gui_disband_schools
	desc = kog_gui_disband_schools_desc
	
	visible = {
		active_law:lawgroup_education_system = {
			has_amendment = amendment_type:amendment_kog_sunday_schools
		}
	}
	possible = {
		has_law_or_variant = law_type:law_state_religion
		has_institution = institution_schools
		NOT = {
			has_modifier = modifier_kog_established_sunday_schools
		}
	}
	effect = {
		active_law:lawgroup_education_system = {
			random_scope_amendment = {
				limit = {
					amendment_type:amendment_kog_sunday_schools ?= this.type
				}
				remove_amendment = yes
			}
		}
		add_modifier = {
			name = modifier_kog_disbanded_sunday_schools
			years = 5
			is_decaying = yes
		}
	}
	
	ai_chance = {
		value = 0
	}
}
