####---------- Apostolic Succession Law Group Debate Buttons ----------####
kog_gui_debate_apostolic_succession = {
	name = "kog_gui_debate_apostolic_succession"
	desc = "kog_gui_debate_canon_law_desc"
	
	visible = {
		kog_council_button_validity_trigger = yes
	}
	selected = {
		kog_canon_lawgroup_debate_trigger = {
			flag = apostolic_succession
		}
	}
	possible = {
		kog_country_is_ecumenical_council_participant_trigger = yes
		NOT = {
			kog_canon_lawgroup_debate_trigger = {
				flag = apostolic_succession
			}
		}
		
	}
	effect = {
		kog_add_council_matter_effect = {
			flag = apostolic_succession
		}
	}
	
	ai_chance = {
		value = 0
	}
}

kog_gui_support_papal_primacy = {
	name = "kog_gui_support_papal_primacy"
	desc = "kog_gui_support_papal_primacy_desc"
	
	visible = {
		kog_country_matches_journal_religion_trigger = yes
		kog_canon_lawgroup_debate_trigger = {
			flag = apostolic_succession
		}
	}
	selected = {
		"variable_map(kog_ecumenical_council_map|flag:kog_canon_lawgroup_apostolic_succession)" ?= flag:kog_canon_law_papal_primacy
	}
	possible = {
		kog_country_is_ecumenical_council_participant_trigger = yes
		custom_tooltip = { 
			text = kog_gui_support_papal_primacy_trigger
			NOT = {
				"variable_map(kog_ecumenical_council_map|flag:kog_canon_lawgroup_apostolic_succession)" ?= flag:kog_canon_law_papal_primacy
			}
		}
	}
	effect = {
		custom_tooltip = {
			text = kog_gui_support_papal_primacy_effect
			com_update_value_in_variable_map = {
				name = kog_ecumenical_council_map
				key = flag:kog_canon_lawgroup_apostolic_succession
				value = flag:kog_canon_law_papal_primacy
			}
		}
	}
}
