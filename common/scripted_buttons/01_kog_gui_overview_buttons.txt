kog_gui_declare_autocephaly = {
	name = "kog_gui_declare_autocephaly"
	desc = "kog_gui_declare_autocephaly_desc"
	
	visible = {
		kog_is_schismatic_trigger = no
		kog_is_autocephalous_trigger = no
	}
	selected = {
		kog_is_schismatic_trigger = yes
	}
	possible = {
		is_subject = no
		kog_capacity_trigger = {
			value = 500
		}
		NOR = {
			has_law = law_type:law_theocracy
			has_modifier = modifier_kog_autocephaly_abandoned
		}
		trigger_if = {
			limit = {
				kog_country_is_roman_catholic_trigger = yes
			}
			always = no
		}
		trigger_else_if = {
			limit = {
				kog_country_is_eastern_orthodox_trigger = yes
			}
			trigger_if = {
				limit = {
					OR = {
						c:MOL ?= this
						c:WAL ?= this
					}
				}
				custom_tooltip = {
					text = kog_romania_united
					hidden_trigger = {
						c:ROM ?= this
					}
				}
			}
		}
	}
	effect = {
		add_modifier = {
			name = modifier_kog_schismatic
			days = kog_10_years
		}
		var:kog_local = {
			set_variable = {
				name = kog_schism_var
				value = 0
			}
		}
		trigger_event = {
			id = kog_schism_events.001
			popup = yes
		}
	}
	
	ai_chance = {
		value = 0
	}
}
