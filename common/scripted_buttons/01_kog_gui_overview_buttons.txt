kog_gui_declare_autocephaly = {
	name = "kog_gui_declare_autocephaly"
	desc = "kog_gui_declare_autocephaly_desc"
	
	tags = { major_decision }
	
	visible = {
		kog_is_schismatic_trigger = no
		kog_is_autocephalous_trigger = no
	}
	selected = {
		kog_is_schismatic_trigger = yes
	}
	possible = {
		is_subject = no
		custom_description = {
			text = kog_capacity_trigger
			subject = this
			value = kog_schism_base_cost
			kog_capacity_value >= kog_schism_base_cost
		}
		NOR = {
			has_law = law_type:law_theocracy
			has_modifier = modifier_kog_autocephaly_abandoned
		}
		custom_description = {
			text = kog_seat_opinion_hostile_trigger
			subject = this
			object = var:kog_local.var:kog_mother_church_var
			value = kog_opinion_threshold_hostile
			var:kog_local.var:kog_mother_church_var = {
				"variable_map(kog_opinions_map|prev)" <= kog_opinion_threshold_hostile
			}
		}
		trigger_if = {
			limit = {
				OR = {
					c:MOL ?= this
					c:WAL ?= this
				}
			}
			custom_tooltip = {
				text = kog_romania_united
				hidden_trigger = {
					c:ROM ?= this
				}
			}
		}
	}
	effect = {
		add_modifier = {
			name = modifier_kog_schismatic
			days = kog_10_years
		}
		var:kog_local = {
			set_variable = {
				name = kog_schism_var
				value = 0
			}
		}
		trigger_event = {
			id = kog_schism_events.001
			popup = yes
		}
	}
	
	ai_chance = {
		value = -10
		if = {
			limit = {
				ruler = {
					law_stance = {
						law = law_type:law_state_religion
						value >= approve
					}
					OR = {
						has_trait = imperious
						has_ideology = ideology:ideology_fascist
						has_ideology = ideology:ideology_ethno_nationalist
						has_ideology = ideology:ideology_ethno_nationalist
						has_ideology = ideology:ideology_ethno_nationalist
					}
				}
			}
			add = 50
		}
		if = {
			limit = {
				ruler = {
					OR = {
						has_trait = reserved
						has_ideology = ideology:ideology_traditionalist
						has_ideology = ideology:ideology_integralist
						has_ideology = ideology:ideology_theocrat
						has_ideology = ideology:ideology_legitimist
						character_has_carlist_ideology = yes
						character_has_miguelist_ideology = yes
					}
				}
			}
			add = -100
		}
	}
}
