kog_gui_sway_rome = {
	name = "kog_gui_sway_rome"
	desc = "kog_gui_sway_seat_desc"
	
	visible = {
		always = yes
	}
	selected = {
		has_modifier = modifier_kog_swaying_rome
	}
	possible = {
		influence >= 50
		kog_capacity_trigger = {
			value = 100
		}
	}
	effect = {
		if = {
			limit = {
				has_modifier = modifier_kog_swaying_rome
			}
			remove_modifier = modifier_kog_swaying_rome
		}
		else = {
			add_modifier = modifier_kog_swaying_rome
		}
	}
	
	ai_chance = {
		value = 0
	}
}
