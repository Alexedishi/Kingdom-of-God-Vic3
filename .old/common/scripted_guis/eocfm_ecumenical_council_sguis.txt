## General SGUIs
eocfm_support_council_sgui = {
	scope = country
	
	is_valid = {
		NOR = {
			has_variable = eocfm_council_support_var
			has_variable = eocfm_council_timeout_var
			has_variable = eocfm_council_skipped
		}
	}
	effect = {
		set_variable = eocfm_council_support_var
		set_variable = {
			name = eocfm_council_timeout_var
			days = 30
		}
	}
}

eocfm_abstain_council_sgui = {
	scope = country
	
	is_valid = {
		OR = {
			has_variable = eocfm_council_support_var
			has_variable = eocfm_council_oppose_var
		}
		NOR = {
			has_variable = eocfm_council_timeout_var
			has_variable = eocfm_council_skipped
		}
	}
	effect = {
		remove_variable = eocfm_council_support_var
		remove_variable = eocfm_council_oppose_var
		set_variable = {
			name = eocfm_council_timeout_var
			days = 30
		}
	}
}

## Catholic SGUIs
eocfm_catholic_council_sgui = {
	scope = country
	
	is_shown = {
		has_global_variable = eocfm_catholic_convocation_var
	}
}

eocfm_catholic_convoke_sgui = {
	scope = country
	
	is_shown = {
		has_variable = eocfm_is_autocephalous
	}
	is_valid = {
		is_subject = no
		is_at_war = no
		eocfm_controls_any_seat_trigger = yes
		eocfm_has_1000_caesaropapism_trigger = yes
#		eocfm_is_autocephalous_trigger = yes # TODO replace this
		# TODO add can_host_council trigger variant
		custom_tooltip = {
			text = eocfm_council_active_NOT
			hidden_trigger = {
				NOT = {
					has_global_variable = eocfm_catholic_matter_var
				}
			}
		}
	}
	effect = {
#		trigger_event = {
#			id = eocfm.2320 # Convoking the Ecumenical Council # TODO replace with variant event
#			popup = yes
#		}
		show_as_tooltip = {
			add_modifier = modifier_eocfm_hosting_council
		}
		hidden_effect = { # This is to present the player clean information in the UI, but has a time limit to avoid shenanigans
			add_modifier = {
				name = modifier_eocfm_hosting_council
				months = 12
			}
		}
		set_global_variable = eocfm_catholic_matter_var
		set_global_variable = eocfm_catholic_convocation_var # Error suppression- remove when event is implemented
	}
}

## Orthodox SGUIs
eocfm_orthodox_council_sgui = {
	scope = country
	
	is_shown = {
		has_global_variable = eocfm_orthodox_convocation_var
	}
}

eocfm_orthodox_convoke_sgui = {
	scope = country
	
	is_shown = {
		has_variable = eocfm_is_autocephalous
	}
	is_valid = {
		is_subject = no
		is_at_war = no
		eocfm_controls_any_seat_trigger = yes
		eocfm_has_1000_caesaropapism_trigger = yes
		eocfm_is_autocephalous_trigger = yes
		custom_tooltip = {
			text = eocfm_theological_matter
			hidden_trigger = {
				eocfm_can_host_council_trigger = yes
			}
		}
		custom_tooltip = {
			text = eocfm_council_active_NOT
			hidden_trigger = {
				NOT = {
					has_global_variable = eocfm_orthodox_matter_var
				}
			}
		}
	}
	effect = {
		trigger_event = {
			id = eocfm.2320 # Convoking the Ecumenical Council
			popup = yes
		}
		show_as_tooltip = {
			add_modifier = modifier_eocfm_hosting_council
		}
		hidden_effect = { # This is to present the player clean information in the UI, but has a time limit to avoid shenanigans
			add_modifier = {
				name = modifier_eocfm_hosting_council
				months = 12
			}
		}
		set_global_variable = eocfm_orthodox_matter_var
	}
}
