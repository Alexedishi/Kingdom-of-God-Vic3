# Ecumenical Council Convoked (Catholic Variant)
kog_council_events.021 = {
	type = country_event
	placement = scope:kog_council_location_scope
	icon = "gfx/interface/icons/generic_icons/kog_council.dds"
	duration = 1
	
	title = kog_council_events.020.t
	desc = kog_council_events.020.d
	flavor = kog_council_events.020.f
	
	event_image = {
		video = "southamerica_christianity"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	trigger = {
		# triggered by je_kog_catholic_council
	}
	
	immediate = {
		random_country = { 
			limit = { 
				kog_country_is_roman_catholic_trigger = yes
				has_modifier = modifier_kog_convoked_council
			}
			save_scope_as = kog_council_caller
		}
		if = {
			limit = { 
				scope:kog_council_caller = {
					kog_is_autocephalous_trigger = yes
				}
			}
			scope:kog_council_caller = {
				capital = {
					save_scope_as = kog_council_location_scope
				}
			}
		}
		else = {
			global_var:je_kog_catholic = {
				var:kog_location_var = {
					save_scope_as = kog_council_location_scope
				}
			}
		}
	}
	
	option = {
		name = kog_council_events.020.a
		default_option = yes
	}
	
	option = {
		name = kog_council_events.020.b
	}
}
