# Oriental Orthodox Opener Events
kog_oriental.001 = { # Oriental Orthodox Opener Monitor
	type = country_event
	hidden = yes

	trigger = {
		NOT = {
			has_variable = kog_starter_var
		}
	}

	immediate = {
		set_variable = kog_starter_var
		if = {
			limit = {
				c:EGY ?= this
			}
			trigger_event = {
				id = kog_oriental.003 # The Heirs to St. Mark (EGY)
			}
		}
		else_if = {
			limit = {
				c:ARM ?= this
			}
			trigger_event = {
				id = kog_oriental.004 # ARM
			}
		}
		else_if = {
			limit = {
				c:ETH ?= this
			}
			trigger_event = {
				id = kog_oriental.005 # The Church in Isolation (ETH)
			}
		}
		else = {
			trigger_event = {
				id = kog_oriental.002 # The Church in the Orient (Oriental Generic)
			}
		}
	}

}

## ORIENTAL SCHISM NARRATIVE EVENTS ##
kog_oriental.300 = { # Declaration of Schism with Coptic Alexandria
	type = country_event
	placement = root

	title = kog_oriental.300.t
	desc = kog_oriental.300.d
	flavor = kog_oriental.300.f

	event_image = {
		video = "southamerica_christianity"
	}

	icon = "gfx/interface/icons/event_icons/event_oriental_orthodox.dds"

	on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_kog

	trigger = {
		# triggered by kog_declare_autocephaly_sgui
	}

	cancellation_trigger = {
		NOT = {
			has_variable = kog_schism_var
		}
	}

	immediate = {
		set_variable = {
			name = kog_schism_var
			value = 0
		}
		kog_update_patriarch_variables_effect = yes
	}

	option = {
		name = kog_oriental.300.a
		default_option = yes
		show_as_tooltip = {
			add_modifier = modifier_kog_schismatic
		}
		every_country = {
			limit = {
				OR = {
					AND = {
						kog_is_oriental_orthodox_trigger = yes
						NOT = {
							this = ROOT
						}
					}
					c:TUR ?= this
				}
			}
			ROOT = {
				save_scope_as = kog_schism_scope
			}
			post_notification = notification_kog_autocephaly_declared
		}
	}

}
