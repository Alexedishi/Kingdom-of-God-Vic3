kog_orthodox.001 = { # Eastern Orthodox Opener Monitor
	type = country_event
	hidden = yes

	trigger = {
		NOT = {
			has_variable = kog_starter_var
		}
	}

	immediate = {
		set_variable = kog_starter_var
		if = {
			limit = {
				c:RUS ?= this
			}
			trigger_event = {
				id = kog_orthodox.003 # The Most Holy Synod (RUS)
			}
		}
		else_if = {
			limit = {
				c:GRE ?= this
			}
			trigger_event = {
				id = kog_orthodox.004 # The Holy Synod of Greece (GRE)
			}
		}
		else_if = {
			limit = {
				c:MON ?= this
			}
			trigger_event = {
				id = kog_orthodox.005 # The Union of State and Church (MON)
			}
		}
		else_if = {
			limit = {
				country_has_primary_culture = cu:serb
				NOT = {
					c:MON ?= this
				}
			}
			trigger_event = {
				id = kog_orthodox.006 # Church of Saint Sava (SER)
			}
		}
		else_if = {
			limit = {
				country_has_primary_culture = cu:bulgarian
			}
			trigger_event = {
				id = kog_orthodox.007 # BUL
			}
		}
		else_if = {
			limit = {
				country_has_primary_culture= cu:romanian
			}
			trigger_event = {
				id = kog_orthodox.008 # ROM
			}
		}
		else_if = {
			limit = {
				c:ALK ?= this
			}
			trigger_event = {
				id = kog_orthodox.009 # ALK
			}
		}

		else = {
			trigger_event = {
				id = kog_orthodox.002 # The Bishops of the East (Orthodox Generic)
			}
		}
	}

}

kog_orthodox.101 = { # A Heretical Ruler
	type = country_event
	placement = root

	title = kog_orthodox.101.t
	desc = kog_orthodox.101.d
	flavor = kog_orthodox.101.f

	event_image = {
		video = "votp_monarch_holding_court"
	}

	icon = "gfx/interface/icons/event_icons/event_orthodox.dds"

	on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_kog

	trigger = {
		NOT = {
			has_variable = kog_ruler_conversion_considered
		}
		has_law = law_type:law_monarchy
		ruler = {
			NOT = {
				has_religion = orthodox
			}
		}
		heir ?= {
			NOT = {
				has_religion = orthodox
			}
		}
		ig:ig_devout = {
			ig_approval <= 5
		}
	}

	immediate = {
		set_variable = kog_ruler_conversion_considered
		ruler = {
			save_scope_as = kog_ruler_scope
		}
		heir = {
			save_scope_as = kog_heir_scope
		}
	}

	option = { # Both will convert
		name = kog_orthodox.101.a
		default_option = yes
		scope:kog_ruler_scope = {
			change_character_religion = rel:orthodox
			add_modifier = modifier_kog_accepted_conversion
		}
		scope:kog_heir_scope = {
			change_character_religion = rel:orthodox
			add_modifier = modifier_kog_accepted_conversion
		}
		add_modifier = {
			name = modifier_kog_ruler_accepted_conversion
			days = normal_modifier_time
		}
		add_loyalists = {
			religion = rel:orthodox
			value = medium_radicals
		}
	}

	option = { # Heir will convert
		name = kog_orthodox.101.b
		scope:kog_ruler_scope = {
			add_modifier = {
				name = modifier_kog_refused_conversion
				days = very_long_modifier_time
			}
		}
		scope:kog_heir_scope = {
			change_character_religion = rel:orthodox
			add_modifier = modifier_kog_accepted_conversion
		}
		add_loyalists = {
			religion = rel:orthodox
			value = small_radicals
		}
	}

	option = { # 99 THESES GO BRRRRRRRRR
		name = kog_orthodox.101.c
		scope:kog_ruler_scope = {
			add_modifier = {
				name = modifier_kog_refused_conversion
				days = very_long_modifier_time
			}
		}
		scope:kog_heir_scope = {
			add_modifier = {
				name = modifier_kog_refused_conversion
				days = very_long_modifier_time
			}
		}
		add_modifier = {
