####---------- Episcopal Seat Integration Completion Events ----------####
kog_episcopal_events.100 = {
	type = country_event
	placement = scope:kog_episcopal_seat_scope.var:kog_seat_province_var
	icon = gfx/interface/icons/generic_icons/kog_episcopal_seat.dds
	duration = 3
	dlc = dlc_kog
	
	title = kog_episcopal_events.100.t
	desc = kog_episcopal_events.100.d
	flavor = kog_episcopal_events.100.f
	
	gui_window = event_window_com_base
	
	event_image = {
		video = "ep1_transfer_of_authority"
	}
	on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	trigger = {
		has_variable_list = kog_seat_integration_list
		any_in_list = {
			variable = kog_seat_integration_list
			var:kog_integration_var >= 60
		}
	}
	
	cooldown = {
		days = kog_3_months
	}
	
	immediate = {
		save_scope_as = kog_seat_integrator_scope
		random_in_list = {
			variable = kog_seat_integration_list
			limit = {
				var:kog_integration_var >= 60
			}
			save_scope_as = kog_episcopal_seat_scope
			
		}
		scope:kog_episcopal_seat_scope = {
			remove_from_list = kog_seat_integration_list
			remove_variable = kog_integrator_var
			remove_variable = kog_integration_var
			set_variable = {
				name = kog_controller_var
				value = scope:kog_seat_integrator_scope
			}
		}
		set_variable = {
			name = kog_seat_var
			value = scope:kog_episcopal_seat_scope
			months = 3
		}
		every_country = {
			limit = {
				kog_content_scope_trigger = yes
			}
			if = {
				limit = {
					religion = scope:kog_seat_integrator_scope.religion
				}
				post_notification = kog_new_seat_integrated_same_religion
			}
			else = {
				post_notification = kog_new_seat_integrated_different_religion
			}
		}
	}
	
	option = {
		name = kog_episcopal_events.100.a
		default_option = yes
		kog_complete_seat_integration_effect = yes
	}
}

kog_episcopal_events.190 = {
	type = country_event
	placement = scope:kog_integrator_scope
	icon = gfx/interface/icons/generic_icons/kog_episcopal_seat.dds
	duration = 3
	dlc = dlc_kog
	
	title = kog_episcopal_events.190.t
	desc = kog_episcopal_events.190.d
	flavor = kog_episcopal_events.190.f
	
	gui_window = event_window_com_base
	
	event_image = {
		video = southamerica_christianity
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	trigger = {
		NOT = {
			has_global_variable = kog_episcopal_seat_dispute_cooldown
		}
		any_country = {
			religion = root.religion
			relations:root <= relations_threshold:poor
			kog_is_integrating_any_seat_trigger = yes
		}
	}
	
	cancellation_trigger = {
		NOT = {
			exists = scope:kog_integrator_scope
		}
	}
	
	immediate = {
		set_global_variable = {
			name = kog_episcopal_seat_dispute_cooldown
			years = 5
		}
		random_country = {
			limit = {
				religion = root.religion
				relations:root <= relations_threshold:poor
				kog_is_integrating_any_seat_trigger = yes
			}
			save_scope_as = kog_integrator_scope
		}
	}
	
	option = {
		name = kog_episcopal_events.190.a
		default_option = yes
		scope:kog_integrator_scope = {
			change_relations = {
				country = root
				value = -20
			}
			create_diplomatic_catalyst = {
				type = catalyst_event_negative
				target = root
			}
			set_variable = {
				name = kog_seat_integration_opposed
				years = 2
			}
		}
	}
	
	option = {
		name = kog_episcopal_events.190.b
		scope:kog_integrator_scope = {
			change_relations = {
				country = root
				value = 20
			}
			create_diplomatic_catalyst = {
				type = catalyst_event_positive
				target = root
			}
		}
	}
	
	after = {
		save_scope_as = kog_disputer_scope
		scope:kog_integrator_scope = {
			trigger_event = kog_episcopal_events.191
		}
	}
}
