kog_episcopal_events.100 = {
	type = country_event
	placement = p:xB040A0
	icon = gfx/interface/icons/generic_icons/kog_episcopal_seat.dds
	duration = 3
	dlc = dlc_kog
	
	title = kog_episcopal_events.100.t
	desc = kog_episcopal_events.100.d
	flavor = kog_episcopal_events.100.f
	
	gui_window = event_window_com_base
	
	event_image = {
		video = "ep1_transfer_of_authority"
	}
	on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	trigger = {
		var:kog_local = {
			has_variable = kog_integrating_rome
			var:kog_integrating_rome >= 60
		}
	}
	
	cooldown = {
		days = kog_6_months
	}
	
	immediate = {
		if = {
			limit = {
				is_key_in_global_variable_map = {
					name = kog_rome
					target = flag:kog_controller_var
				}
				exists = "global_variable_map(kog_rome|flag:kog_controller_var)"
			}
			"global_variable_map(kog_rome|flag:kog_controller_var)" = {
				save_scope_as = kog_rome_controller_scope
			}
		}
	}
	
	option = {
		name = kog_episcopal_events.100.a
		default_option = yes
		kog_complete_seat_integration_effect = {
			seat = kog_rome
		}
	}
}
