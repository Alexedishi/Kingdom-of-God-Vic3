eocfm.0200 = { # State Authority Asserted Over Rome
	type = country_event
	placement = root
	
	title = eocfm.0200.t
	desc = eocfm.0200.d
	flavor = eocfm.0200.f
	
	event_image = {
		video = "ep1_transfer_of_authority"
	}
	
	icon = "gfx/interface/icons/event_icons/event_orthodox.dds"
	
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	duration = 3
	
	dlc = dlc_eocfm
	
	trigger = {
		# triggered by eocfm_assert_rome_sgui
	}
	
	immediate = {
		eocfm_update_patriarch_variables_effect = yes
		set_variable = eocfm_rome_assert_var
		random_country = {
			limit = {
				has_variable = eocfm_controls_rome
			}
			save_scope_as = eocfm_controller_scope
		}
		every_country = {
			limit = {
				OR = {
					eocfm_is_eastern_christian_trigger = yes
					has_variable = eocfm_controls_rome
				}
				NOT = {
					has_variable = eocfm_rome_assert_var
				}
			}
			random_country = {
				limit = {
					has_variable = eocfm_rome_assert_var
				}
				save_scope_as = eocfm_usurper_scope
			}
			post_notification = notification_eocfm_rome_usurped
		}
		scope:eocfm_controller_scope ?= {
			remove_variable = eocfm_controls_rome
		}
		set_variable = eocfm_controls_rome
		set_variable = eocfm_is_autocephalous
		remove_variable = eocfm_rome_assert_var
		remove_variable = eocfm_schism_var
	}
	
	option = {
		name = eocfm.0200.a
		default_option = yes
		add_modifier = {
			name = modifier_eocfm_autocephaly_granted
			months = long_modifier_time
		}
		custom_tooltip = eocfm_usurped_rome
		scope:eocfm_controller_scope ?= {
			change_relations = {
				country = root
				value = -30
			}
		}
	}
	
}
