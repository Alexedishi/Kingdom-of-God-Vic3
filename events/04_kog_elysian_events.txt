## Elysian Orthodox Opener Events (Basileia Romaion) ##

kog_elysian.001 = { # Elysian Orthodox Opener Monitor
	type = country_event
	hidden = yes

	trigger = {
		NOT = {
			has_variable = kog_starter_var
		}
	}

	immediate = {
		set_variable = kog_starter_var
		else_if = {
			limit = {
				has_variable = kog_controls_elysium
			}
			trigger_event = {
				id = kog_elysian.002 # The Bishops of the Blessed Land (BR_ELY)
			}
		}
		else_if = {
			limit = {
				has_variable = kog_controls_mixtec
			}
			trigger_event = {
				id = kog_elysian.003 # The Bishops of the Bird's Hill (MIX)
			}
		}
		else_if = {
			limit = {
				has_variable = kog_controls_cahokia
			}
			trigger_event = {
				id = kog_elysian.004 # The Patriarchate of the Holy Mound (CAH)
			}
		}
		else_if = {
			limit = {
				kog_is_elysian_orthodox_trigger = yes
			}
			trigger_event = {
				id = kog_elysian.001 # The Ecumene Amongst the Barbarians (Elysian Generic)
			}
		}
	}
}

## ELYSIAN SCHISM NARRATIVE EVENTS ##
kog_elysian.300 = { # Declaration of Schism with Elysium
	type = country_event
	placement = root

	title = kog_elysian.300.t
	desc = kog_elysian.300.d
	flavor = kog_elysian.300.f

	event_image = {
		video = "southamerica_christianity"
	}

	icon = "gfx/interface/icons/event_icons/event_orthodox.dds"

	on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_kog

	trigger = {
		# triggered by kog_declare_autocephaly_sgui
	}

	cancellation_trigger = {
		NOT = {
			has_variable = kog_schism_var
		}
	}

	immediate = {
		set_variable = {
			name = kog_schism_var
			value = 0
		}
		kog_update_patriarch_variables_effect = yes
	}

	option = {
		name = kog_elysian.300.a
		default_option = yes
		show_as_tooltip = {
			add_modifier = modifier_kog_schismatic
		}
		every_country = {
			limit = {
				OR = {
					AND = {
						kog_is_elysian_orthodox_trigger = yes
						NOT = {
							this = ROOT
						}
					}
					c:TUR ?= this
				}
			}
			ROOT = {
				save_scope_as = kog_schism_scope
			}
			post_notification = notification_kog_autocephaly_declared
		}
	}

}
