# Roman Catholic Opener Events I
kog_catholic.001 = { # Catholic Opener Monitor
	type = country_event
	hidden = yes

	trigger = {
		NOT = {
			has_variable = kog_starter_var
		}
	}

	immediate = {
		set_variable = kog_starter_var
		if = {
			limit = {
				c:PAP ?= this
			}
			trigger_event = {
				id = kog_catholic.002 # The Throne of St. Peter (PAP)
			}
		}
		else_if = {
			limit = {
				NOT = {
					c:PAP ?= this
				}
				OR = {
					country_has_primary_culture = cu:north_italian
					country_has_primary_culture = cu:south_italian
				}
			}
			trigger_event = {
				id = kog_catholic.003 # The Homelands of the Church
			}
		}
		else_if = {
			limit = {
				c:SPA ?= this
			}
			trigger_event = {
				id = kog_catholic.004 # St. James and Strike for Spain! (SPA)
			}
		}
		else_if = {
			limit = {
				c:FRA ?= this
			}
			trigger_event = {
				id = kog_catholic.005 # The Eldest Daughter of the Church (FRA)
			}
		}
		else_if = {
			limit = {
				NOT = {
					c:AUS ?= this
				}
				OR = {
					country_has_primary_culture = cu:north_german
					country_has_primary_culture = cu:south_german
				}
			}
			trigger_event = {
				id = kog_catholic.006 # Cuius Regio, Eius Religio (German)
			}
		}
		else_if = {
			limit = {
				c:AUS ?= this
			}
			trigger_event = {
				id = kog_catholic.007 # The Guardians of the Church (AUS)
			}
		}
		else_if = {
			limit = {
				c:POR ?= this
			}
			trigger_event = {
				id = kog_catholic.008 # POR
			}
		}
		else_if = {
			limit = {
				c:MEX ?= this
			}
			trigger_event = {
				id = kog_catholic.010 # MEX
			}
		}
		else_if = {
			limit = {
				c:UCA ?= this
			}
			trigger_event = {
				id = kog_catholic.011 # UCA
			}
		}
		else_if = {
			limit = {
				c:CLM ?= this
			}
			trigger_event = {
				id = kog_catholic.012 # CLM
			}
		}
		else_if = {
			limit = {
				c:BRZ ?= this
			}
			trigger_event = {
				id = kog_catholic.013 # BRZ
			}
		}
		else_if = {
			limit = {
				c:ARG ?= this
			}
			trigger_event = {
				id = kog_catholic.014 # ARG
			}
		}
		else_if = {
			limit = {
				c:CHL ?= this
			}
			trigger_event = {
				id = kog_catholic.015 # CHL
			}
		}
		else_if = {
			limit = {
				c:PHI ?= this
			}
			trigger_event = {
				id = kog_catholic.016 # PHI
			}
		}
		else_if = {
			limit = {
				c:VNZ ?= this
			}
			trigger_event = {
				id = kog_catholic.017 # VNZ
			}
		}
		else_if = {
			limit = {
				c:BOL ?= this
			}
			trigger_event = {
				id = kog_catholic.018 # BOL
			}
		}
		else_if = {
			limit = {
				c:KRA ?= this
			}
			trigger_event = {
				id = kog_catholic.019 # KRA
			}
		}
		else = {
			trigger_event = {
				id = kog_catholic.009 # The Bishops of Rome (Catholic Generic)
			}
		}
	}

}

## CATHOLIC SCHISM EVENTS ##
kog_catholic.300 = { # Declaration of Schism with Rome
	type = country_event
	placement = root

	title = kog_catholic.300.t
	desc = kog_catholic.300.d
	flavor = kog_catholic.300.f

	event_image = {
		video = "southamerica_christianity"
	}

	icon = "gfx/interface/icons/event_icons/event_catholic.dds"

	on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_kog

	trigger = {
		# triggered by kog_declare_autocephaly_sgui
	}

	cancellation_trigger = {
		NOT = {
			has_variable = kog_schism_var
		}
	}

	immediate = {
		set_variable = {
			name = kog_schism_var
			value = 0
		}
		kog_update_patriarch_variables_effect = yes
	}

	option = {
		name = kog_catholic.300.a
		default_option = yes
		show_as_tooltip = {
			add_modifier = modifier_kog_schismatic
		}
		every_country = {
			limit = {
				OR = {
					AND = {
						kog_is_roman_catholic_trigger = yes
						NOT = {
							this = ROOT
						}
					}
					c:TUR ?= this
				}
			}
			ROOT = {
				save_scope_as = kog_schism_scope
			}
			post_notification = notification_kog_autocephaly_declared
		}
	}

}
