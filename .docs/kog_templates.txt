####---------- Seat Patriarch Update Effect ----------####
kog_update_tempkey_patriatch_effect = {
	# Step 1, Ensure the relevant variable map exists
	if = {
		limit = {
			NOT = {
				has_global_variable_map = kog_tempkey
			}
		}
		kog_initialize_tempkey_variable_map_effect = yes # Regenerate if missing
	}
	# Step 2, Ensure that there is a valid scope saved to the key flag:kog_seat_patriarch
	if = {
		limit = { # If this evaluates to true, proceed to patriarch regeneration
			OR = {
				AND = {
					is_key_in_global_variable_map = {
						name = kog_tempkey
						target = flag:kog_seat_patriarch
					}
					NOT = {
						exists = "global_variable_map(kog_tempkey|flag:kog_seat_patriarch)"
					}
				}
				NOT = {
					is_key_in_global_variable_map = {
						name = kog_tempkey
						target = flag:kog_seat_patriarch
					}
				}
			}
		}
		# First, attempt to find the character in the void
		if = {
			limit = {
				any_character_in_void = {
					has_modifier = modifier_kog_tempkey_patriarch
				}
			}
			random_character_in_void = {
				limit = {
					has_modifier = modifier_kog_tempkey_patriarch
				}
				save_scope_as = kog_tempkey_patriarch_scope
			}
		}
		# Failing that, generate a new character
		else = {
			# Attempt to generate using the Seat controller
			if = {
				limit = {
					AND = {
						is_key_in_global_variable_map = {
							name = kog_tempkey
							target = flag:kog_seat_controller
						}
						exists = "global_variable_map(kog_tempkey|flag:kog_seat_controller)"
					}
				}
				"global_variable_map(kog_tempkey|flag:kog_seat_controller)" = {
					create_character = {
						template = kog_patriarch_template
						on_created = {
							add_modifier = modifier_kog_tempkey_patriarch
							place_character_in_void = 180 # 15 year shelf life
						}
					}
				}
			}
			# Attempt to generate using the Province owner
			else_if = {
				limit = {
					p: = {
						owner = {
							kog_content_scope_trigger = yes
						}
					}
				}
				p: = {
					owner = {
						create_character = {
							template = kog_patriarch_template
							religion = rel:
							on_created = {
								add_modifier = modifier_kog_tempkey_patriarch
								place_character_in_void = 180 # 15 year shelf life
							}
						}
					}
				}
			}
			# Fall back to a --; create from scope of province owner
			else = {
				p: = {
					owner = {
						create_character = {
							template = kog_patriarch_template
							culture = cu:
							religion = rel:
							on_created = {
								add_modifier = modifier_kog_tempkey_patriarch
								place_character_in_void = 180 # 15 year shelf life
							}
						}
					}
				}
			}
			# Capture the new character
			random_character_in_void = {
				limit = {
					has_modifier = modifier_kog_tempkey_patriarch
				}
				save_scope_as = kog_tempkey_patriarch_scope
			}
		}
		# Update Relations Variables
		every_country = {
			limit = {
				kog_content_scope_trigger = yes
			}
			je:je_kog_local = {
				# Sanity check in case initialization had a wobble
				if = {
					limit = {
						NOT = {
							has_variable = kog_tempkey_relations_var
						}
					}
					set_variable = {
						name = kog_tempkey_relations_var
						value = 0
					}
				}
				# Do not reduce relations if already at or below 0
				else_if = {
					limit = {
						var:kog_tempkey_relations_var > 0
					}
					change_variable = {
						name = kog_tempkey_relations_var
						add = -25
					}
					clamp_variable = {
						name = kog_tempkey_relations_var
						min = 0
						max = 50
					}
				}
			}
			# Notify all Christians of a new Pope
			post_notification = notification_kog_tempkey_replaced
		}
	}
	# Step 3, Update the relevant variable map
	remove_from_global_variable_map = {
		name = kog_tempkey
		key = flag:kog_seat_patriarch
	}
	add_to_global_variable_map = {
		name = kog_tempkey
		key = flag:kog_seat_patriarch
		value = scope:kog_tempkey_patriarch_scope
	}
}

####---------- Seat Controller Validation Effect ----------####
kog_validate_tempkey_controller_effect = {
	# Clear the controller from the variable map; do NOT regenerate the map, this is handled by Patriarch validation
	if = {
		limit = {
			has_global_variable_map = kog_tempkey
			is_key_in_variable_map = {
				name = kog_tempkey
				target = flag:kog_seat_controller
			}
		}
		remove_from_global_variable_map = {
			name = kog_tempkey
			key = flag:kog_seat_controller
		}
	}
	# Attempt to find the controller with this terrible iterator
	random_country = {
		limit = {
			kog_content_scope_trigger = yes
			je:je_kog_local = {
				has_variable = kog_controls_tempkey
			}
		}
		save_scope_as = kog_tempkey_controller
	}
	# If the controller exists, save them to the map; overwise, LEAVE OUT the controller key
	if = {
		limit = {
			exists = scope:kog_tempkey_controller
		}
		add_to_global_variable_map = {
			name = kog_tempkey
			key = flag:kog_seat_controller
			value = scope:kog_tempkey_controller
		}
	}
}
