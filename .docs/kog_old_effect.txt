




kog_update_santiago_effect = {
	global_var:kog_global = {
		var:kog_santiago = {
			if = {
				limit = {
					NOT = {
						exists = var:kog_patriarch_var
					}
				}
				# First, try to find character in void
				if = {
					limit = {
						any_character_in_void = {
							has_template = template_kog_santiago_patriarch
						}
					}
					random_character_in_void = {
						limit = {
							has_template = template_kog_santiago_patriarch
						}
						save_scope_as = kog_santiago_patriarch_scope
					}
				}
				# Failing that, generate a new character
				else = {
					# Attempt to generate using the Santiago controller
					if = {
						limit = {
							has_variable = kog_controller_var
							exists = var:kog_controller_var
						}
						var:kog_controller_var = {
							create_character = {
								template = template_kog_santiago_patriarch
								on_created = {
									place_character_in_void = 180 # 15 year shelf life
								}
							}
						}
					}
					# Fallback to a Spanish Catholic
					else = {
						create_character = {
							template = template_kog_santiago_patriarch
							culture = cu:spanish
							religion = rel:catholic
							on_created = {
								place_character_in_void = 180 # 15 year shelf life
							}
						}
					}
				}
				# Save the new character
				random_character_in_void = {
					limit = {
						has_template = template_kog_santiago_patriarch
					}
					save_scope_as = kog_santiago_patriarch_scope
				}
				set_variable = {
					name = kog_patriarch_var
					value = scope:kog_santiago_patriarch_scope
				}
				
				# Update Relations Variables
				every_country = {
					limit = {
						kog_content_scope_trigger = yes
					}
					je:je_kog_local = {
						# Sanity check in case initialization had a wobble
						if = {
							limit = {
								NOT = {
									has_variable = kog_santiago_relations_var
								}
							}
							set_variable = {
								name = kog_santiago_relations_var
								value = 0
							}
						}
						# Do not reduce relations if already at or below 0
						else_if = {
							limit = {
								var:kog_santiago_relations_var > 0
							}
							change_variable = {
								name = kog_santiago_relations_var
								add = -25
							}
							clamp_variable = {
								name = kog_santiago_relations_var
								min = 0
								max = 50
							}
						}
					}
					if = {
						limit = {
							country_has_state_religion = rel:catholic
						}
						post_notification = notification_kog_santiago_replaced
					}
				}
			}
		}
	}
}

					}
					if = {
						limit = {
							country_has_state_religion = rel:catholic
						}
						post_notification = notification_kog_alexandria_replaced
					}
				}
			}
		}
	}
}

kog_update_moscow_effect = {
	global_var:kog_global = {
		var:kog_moscow = {
			if = {
				limit = {
					NOT = {
						exists = var:kog_patriarch_var
					}
				}
				# First, try to find character in void
				if = {
					limit = {
						any_character_in_void = {
							has_template = template_kog_moscow_patriarch
						}
					}
					random_character_in_void = {
						limit = {
							has_template = template_kog_moscow_patriarch
						}
						save_scope_as = kog_moscow_patriarch_scope
					}
				}
				# Failing that, generate a new character
				else = {
					# Attempt to generate using the Moscow controller
					if = {
						limit = {
							has_variable = kog_controller_var
							exists = var:kog_controller_var
						}
						var:kog_controller_var = {
							create_character = {
								template = template_kog_moscow_patriarch
								on_created = {
									place_character_in_void = 180 # 15 year shelf life
								}
							}
						}
					}
					# Fallback to a Greek Orthodox
					else = {
						create_character = {
							template = template_kog_moscow_patriarch
							culture = cu:greek
							religion = rel:orthodox
							on_created = {
								place_character_in_void = 180 # 15 year shelf life
							}
						}
					}
				}
				# Save the new character
				random_character_in_void = {
					limit = {
						has_template = template_kog_moscow_patriarch
					}
					save_scope_as = kog_moscow_patriarch_scope
				}
				set_variable = {
					name = kog_patriarch_var
					value = scope:kog_moscow_patriarch_scope
				}
				
				# Update Relations Variables
				every_country = {
					limit = {
						kog_content_scope_trigger = yes
					}
					je:je_kog_local = {
						# Sanity check in case initialization had a wobble
						if = {
							limit = {
								NOT = {
									has_variable = kog_moscow_relations_var
								}
							}
							set_variable = {
								name = kog_moscow_relations_var
								value = 0
							}
						}
						# Do not reduce relations if already at or below 0
						else_if = {
							limit = {
								var:kog_moscow_relations_var > 0
							}
							change_variable = {
								name = kog_moscow_relations_var
								add = -25
							}
							clamp_variable = {
								name = kog_moscow_relations_var
								min = 0
								max = 50
							}
						}
					}
					if = {
						limit = {
							country_has_state_religion = rel:catholic
						}
						post_notification = notification_kog_moscow_replaced
					}
				}
			}
		}
	}
}
