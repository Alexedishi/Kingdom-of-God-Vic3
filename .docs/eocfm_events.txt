				OR = {

						eocfm_is_roman_catholic_trigger = yes

							this = ROOT
						}

			ROOT = {
				save_scope_as = eocfm_schism_scope
			}
			post_notification = notification_eocfm_autocephaly_declared
		}	
	}

}
