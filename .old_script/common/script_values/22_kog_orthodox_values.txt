kog_ecumenical_weight_value = {
	value = 0
	if = {
		limit = {
			kog_is_autocephalous_trigger = yes
		}
		add = {
			value = 1
			desc = kog_value_from_having_autocephaly
		}
	}
	if = {
		limit = {
			has_variable = kog_controls_constantinople
		}
		add = {
			value = 1
			desc = kog_value_from_controls_constantinople
		}
	}
	if = {
		limit = {
			has_variable = kog_controls_antioch
		}
		add = {
			value = 1
			desc = kog_value_from_controls_antioch
		}
	}
	if = {
		limit = {
			has_variable = kog_controls_jerusalem
		}
		add = {
			value = 1
			desc = kog_value_from_controls_jerusalem
		}
	}
	if = {
		limit = {
			has_variable = kog_controls_alexandria
		}
		add = {
			value = 1
			desc = kog_value_from_controls_alexandria
		}
	}
	if = {
		limit = {
			has_variable = kog_controls_moscow
		}
		add = {
			value = 1
			desc = kog_value_from_controls_moscow
		}
	}
}

kog_orthodox_support_value = {
	value = 0
	every_country = {
		limit = {
			kog_is_eastern_orthodox_trigger = yes
			has_variable = kog_council_support_var
		}
		add = kog_ecumenical_weight_value
	}
}

kog_orthodox_opposition_value = {
	value = 0
	every_country = {
		limit = {
			kog_is_eastern_orthodox_trigger = yes
			has_variable = kog_council_oppose_var
		}
		subtract = kog_ecumenical_weight_value
	}
}

kog_orthodox_council_value = {
	value = 0
	add = {
		value = kog_orthodox_support_value
		desc = kog_value_from_orthodox_support
	}
	subtract = {
		value = kog_orthodox_opposition_value
		desc = kog_value_from_orthodox_opposition
	}
}
